def _get_pool() -> psycopg2.pool.ThreadedConnectionPool:
    global _pg_pool
    if _pg_pool is None or _pg_pool.closed:
        _pg_pool = psycopg2.pool.ThreadedConnectionPool(
            minconn=2, maxconn=max(SQL_MAX_WORKERS + 2, 10), dsn=PG_DSN
        )
    return _pg_pool


def _pg_conn():
    return _get_pool().getconn()


def _put_conn(conn):
    try:
        _get_pool().putconn(conn)
    except Exception:
        pass


class _pg_cursor:
    """Borrows a connection from the pool, yields a cursor, returns it on exit."""

    def __init__(self, dict_cursor: bool = False):
        self.dict_cursor = dict_cursor
        self._conn = None
        self._cur = None

    def __enter__(self):
        self._conn = _pg_conn()
        factory = psycopg2.extras.RealDictCursor if self.dict_cursor else None
        self._cur = self._conn.cursor(cursor_factory=factory) if factory else self._conn.cursor()
        return self._cur

    def __exit__(self, exc_type, exc_val, exc_tb):
        if exc_type:
            try:
                self._conn.rollback()
            except Exception:
                pass
        else:
            try:
                self._conn.commit()
            except Exception:
                pass
        try:
            if self._cur:
                self._cur.close()
        except Exception:
            pass
        if self._conn:
            _put_conn(self._conn)
        return False


# INTERNAL UTILITIES
# Reconciled exports built BEFORE they were given their own containers are still
# sitting in the feed containers. They are recognisable by name, and must be
# excluded from the "pick a file to reconcile" lists — both the file itself and
# the date folder it created.
_RECONCILED_NAME_MARKERS = ("_all_", "_iwpb_", "_noniwpb_", "_master_",
                            "_active_", "_cleared_")


def _is_reconciled_export(file_name: str) -> bool:
    name = (file_name or "").lower()
    if not name.startswith(("fri_", "mnrl_")):
        return False
    # feed downloads are fri_data_… / mnrl_normal_… / mnrl_reactivated_…
    if name.startswith(("fri_data_", "mnrl_normal_", "mnrl_reactivated_")):
        return False
    return any(m in name for m in _RECONCILED_NAME_MARKERS)


def _ensure_container(name: str) -> None:
    try:
        cc = _blob_service().get_container_client(name)
        cc.create_container()
        print(f"[BLOB] Created Container '{name}'")
    except ResourceExistsError:
        pass
    except HttpResponseError as e:
        raise RuntimeError(f"Failed to ensure container '{name}': {e.message}") from e

def _chunks(lst: list, n: int) -> Generator:
    for i in range(0, len(lst), n):
        yield lst[i : i + n]

def _canon_header(name) -> str:
    return re.sub(r"[^a-z0-9]+", "_", str(name).strip().lower()).strip("_")


# Header spellings that all mean "the mobile number"
_MOB_COL_ALIASES = (
    "mob_num", "mobile_no", "mobile_number", "mobile", "mob_no",
    "msisdn", "phone_no", "final_mob_num",
)


def _resolve_mob_col(requested: str, headers: list[str]) -> str:
    """Find the mobile-number column, whatever the file calls it.

    The UI sends 'mob_num', but export_and_store writes the header as
    'Mobile No'. Matching on the canonical form means either spelling works, and
    a file that uses neither gets a message naming its actual columns.
    """
    by_canon = {_canon_header(h): h for h in headers}

    if requested in headers:
        return requested

    wanted = _canon_header(requested)
    if wanted in by_canon:
        return by_canon[wanted]

    for alias in _MOB_COL_ALIASES:
        if alias in by_canon:
            return by_canon[alias]

    raise ValueError(
        f"No mobile number column in this file. Looked for '{requested}'. "
        f"The file has: {', '.join(headers)}"
    )


def _detect_mnrl_data_type(file_name: str, headers: list[str]) -> str:
    """Normal vs Reactivated, from the file name with a header fallback."""
    name = (file_name or "").lower()
    if "reactivat" in name:
        return "Reactivated"
    if "normal" in name:
        return "Normal"
    canon = {str(h).strip().lower().replace(" ", "_") for h in headers}
    if "date_of_reactivation" in canon or "reactivation_id" in canon:
        return "Reactivated"
    return "Normal"


# EXCEL BUILDERS
# Sensible widths for columns whose content is routinely long. Excel CLIPS a
# cell's text when the neighbouring cell is occupied, so a full 30-character
# email in a default-width column looks truncated even though the value is
# intact — the commonest false report of "the export cut my data".
COLUMN_WIDTHS: dict[str, int] = {
    "Internet Address": 42,
    "Name & Address 1": 34, "Name & Address 2": 34, "Name & Address 3": 34,
    "Name & Address 4": 34, "Name & Address 5": 34,
    "Mobile Number": 16, "Customer Id": 16, "Address ID": 16,
    "Telephone No": 18, "Telephone No 2": 18, "Postal Code": 12,
    "Reported Status": 26, "Current Status": 26, "Cleared Reason": 16,
    "First Reported On": 16, "Last Reported On": 16, "Cleared On": 14,
    "Investigation Details": 42, "Response Message": 38, "C-DOT Request Id": 20,
    "Data Date": 14, "Sent At": 20, "Why Sent": 24,
}
DEFAULT_COLUMN_WIDTH = 18


def _apply_column_widths(ws, columns: list, labels: dict | None = None) -> None:
    """Size columns before any row is written.

    A write-only worksheet accepts column_dimensions only up front, which is why
    this is called immediately after the sheet is created.
    """
    from openpyxl.utils import get_column_letter
    labels = labels or {}
    for idx, col in enumerate(columns, start=1):
        heading = labels.get(col) or str(col).replace("_", " ").title()
        width = COLUMN_WIDTHS.get(heading, DEFAULT_COLUMN_WIDTH)
        ws.column_dimensions[get_column_letter(idx)].width = width


def _make_header_cells(ws, columns: list, labels: dict | None = None) -> list:
    """Header row. `labels` maps column -> heading; without it a raw column name
    is title-cased, which turns 'zbnad1' into 'Zbnad1'."""
    labels = labels or {}
    header_font = Font(bold=True, color="FFFFFF")
    header_fill = PatternFill("solid", fgColor="1D4ED8")
    header_align = Alignment(horizontal="center")
    cells = []
    for col in columns:
        heading = labels.get(col) or str(col).replace("_", " ").title()
        cell = WriteOnlyCell(ws, value=heading)
        cell.font = header_font
        cell.fill = header_fill
        cell.alignment = header_align
        cells.append(cell)
    return cells


def build_excel(records: list, columns: list) -> io.BytesIO:
    buf = io.BytesIO()
    wb = Workbook(write_only=True)
    ws = wb.create_sheet("Data")
    _apply_column_widths(ws, columns)
    ws.append(_make_header_cells(ws, columns))
    for chunk in _chunks(records, 5_000):
        for rec in chunk:
            ws.append([rec.get(c, "") for c in columns])
    wb.save(buf)
    buf.seek(0)
    return buf


def build_excel_parts(
    records: list,
    base_file_name: str,
    columns: list,
    max_rows: int = EXCEL_MAX_ROWS,
) -> list[tuple[io.BytesIO, str, int]]:
    total_parts = -(-len(records) // max_rows)
    results: list[tuple[io.BytesIO, str, int]] = []

    for part_idx, chunk in enumerate(_chunks(records, max_rows), start=1):
        fname = base_file_name if total_parts == 1 else (
            f"{base_file_name.replace('.xlsx', '')}_part{part_idx}.xlsx"
        )
        buf = io.BytesIO()
        wb = Workbook(write_only=True)
        ws = wb.create_sheet("Data")
        _apply_column_widths(ws, columns)
        ws.append(_make_header_cells(ws, columns))
        for row_chunk in _chunks(chunk, 5_000):
            for rec in row_chunk:
                ws.append([rec.get(c, "") for c in columns])
        wb.save(buf)
        buf.seek(0)
        results.append((buf, fname, len(chunk)))

    return results


def build_excel_parts_stream(
    row_iter: Iterable[dict],
    base_file_name: str,
    columns: list,
    max_rows: int = EXCEL_MAX_ROWS,
    labels: dict | None = None,
) -> Iterator[tuple[io.BytesIO, str, int, int]]:
    """Consume an iterator of row dicts and yield finished workbook parts.

    Generator, not a list: each part is uploaded and released before the next is
    built, so peak memory is one workbook rather than the whole result set. This
    is what makes a 10 lakh row download survivable.

    Yields (buffer, file_name, row_count, part_index).
    """
    stem = base_file_name.replace(".xlsx", "")
    part_idx = 0
    wb = ws = buf = None
    rows_in_part = 0

    def _open_part():
        nonlocal wb, ws, buf, rows_in_part, part_idx
        part_idx += 1
        buf = io.BytesIO()
        wb = Workbook(write_only=True)
        ws = wb.create_sheet("Data")
        _apply_column_widths(ws, columns, labels)
        ws.append(_make_header_cells(ws, columns, labels))
        rows_in_part = 0

    def _close_part():
        wb.save(buf)
        buf.seek(0)
        return buf, rows_in_part

    for row in row_iter:
        if ws is None:
            _open_part()
        ws.append([row.get(c, "") for c in columns])
        rows_in_part += 1

        if rows_in_part >= max_rows - 1:  # -1 leaves room for the header
            done_buf, count = _close_part()
            yield done_buf, f"{stem}_part{part_idx}.xlsx", count, part_idx
            wb = ws = buf = None

    if ws is not None:
        done_buf, count = _close_part()
        fname = base_file_name if part_idx == 1 else f"{stem}_part{part_idx}.xlsx"
        yield done_buf, fname, count, part_idx


def build_matched_excel(
    input_rows: list[dict],
    headers: list[str],
    mob_col: str,
    matched_map: dict[str, dict],
    hsbc_columns: list[str] | None = None,
) -> tuple[io.BytesIO, int]:
    hsbc_cols = hsbc_columns or MOBILECLEANUP_FETCH_COLUMNS
    output_headers = headers + [
        MOBILECLEANUP_COLUMN_LABELS.get(c, c.replace("_", " ").title()) for c in hsbc_cols
    ]
    buf = io.BytesIO()
    wb = Workbook(write_only=True)
    ws = wb.create_sheet("Matched Data")
    _apply_column_widths(ws, output_headers)
    ws.append(_make_header_cells(ws, output_headers))

    matched_count = 0
    buffer: list = []

    for row in input_rows:
        mob_val = str(row.get(mob_col, "")).strip()
        if mob_val not in matched_map:
            continue
        data_row = [row.get(h, "") for h in headers]
        col_data = matched_map[mob_val]
        data_row += [col_data.get(col, "") for col in hsbc_cols]
        buffer.append(data_row)
        matched_count += 1
        if len(buffer) >= EXCEL_FLUSH_ROWS:
            for r in buffer:
                ws.append(r)
            buffer.clear()

    for r in buffer:
        ws.append(r)

    wb.save(buf)
    buf.seek(0)
    return buf, matched_count


# AZURE BLOB CORE
def upload_blob(container: str, blob_name: str, data: bytes | io.BytesIO) -> str:
    _ensure_container(container)
    blob_client = _blob_service().get_blob_client(container=container, blob=blob_name)

    if isinstance(data, (bytes, bytearray)):
        size = len(data)
        stream = io.BytesIO(data)
    else:
        data.seek(0, 2)
        size = data.tell()
        data.seek(0)
        stream = data

    if size > BLOB_CHUNK_THRESHOLD:
        blob_client.upload_blob(stream, overwrite=True, max_concurrency=4)
    else:
        blob_client.upload_blob(stream, overwrite=True)

    return blob_client.url


def get_sas_download_url(container: str, blob_name: str, expiry_hours: int = 1) -> str:
    sas_token = generate_blob_sas(
        account_name=AZURE_STORAGE_ACCOUNT_NAME,
        container_name=container,
        blob_name=blob_name,
        account_key=AZURE_STORAGE_ACCOUNT_KEY,
        permission=BlobSasPermissions(read=True),
        expiry=datetime.now(timezone.utc) + timedelta(hours=expiry_hours),
    )
    return (
        f"https://{AZURE_STORAGE_ACCOUNT_NAME}.blob.core.windows.net/"
        f"{container}/{blob_name}?{sas_token}"
    )


# BLOB BROWSER
def list_source_date_folders(container: str) -> list[str]:
    if container not in VALID_SOURCE_CONTAINERS.values():
        raise ValueError(f"Container '{container}' is not a permitted source.")
    cc = _blob_service().get_container_client(container)
    seen = set()
    for blob in cc.list_blobs():
        parts = blob.name.split("/")
        if len(parts) >= 2 and parts[0]:
            seen.add(parts[0])
    return sorted(seen, reverse=True)


def list_source_files(container: str, date_folder: str) -> list[str]:
    if container not in VALID_SOURCE_CONTAINERS.values():
        raise ValueError(f"Container '{container}' is not a permitted source.")
    prefix = f"{date_folder}/"
    cc = _blob_service().get_container_client(container)
    files = []
    for blob in cc.list_blobs(name_starts_with=prefix):
        fname = blob.name[len(prefix) :]
        if (fname and fname.endswith(".xlsx") and "/" not in fname
                and not _is_reconciled_export(fname)):
            files.append(fname)
    return sorted(files)


def read_excel_from_blob(container: str, date_folder: str, file_name: str) -> tuple[list[dict], list[str]]:
    if container not in VALID_SOURCE_CONTAINERS.values():
        raise ValueError(f"Container '{container}' is not a permitted source.")

    blob_path = f"{date_folder}/{file_name}"
    bc = _blob_service().get_blob_client(container=container, blob=blob_path)

    try:
        stream = io.BytesIO()
        bc.download_blob().readinto(stream)
        stream.seek(0)
    except Exception as e:
        raise ValueError(f"Could not read blob '{blob_path}' from '{container}': {e}") from e

    wb = load_workbook(filename=stream, read_only=True, data_only=True)
    ws = wb.active
    rows_iter = ws.iter_rows(values_only=True)

    try:
        raw_headers = next(rows_iter)
    except StopIteration:
        wb.close()
        raise ValueError(f"Excel file '{file_name}' is empty.")

    headers = [str(h).strip() if h is not None else f"col_{i}" for i, h in enumerate(raw_headers)]
    rows = [dict(zip(headers, row)) for row in rows_iter]
    wb.close()

    if not rows:
        raise ValueError(f"Excel file '{file_name}' has a header row but no data rows.")

    return rows, headers


def upload_matchoff_output(buf: io.BytesIO, date_folder: str, file_name: str) -> str:
    _ensure_container(AZURE_CONTAINER_MATCHOFF_OUTPUT)
    blob_path = f"{date_folder}/{file_name}"
    bc = _blob_service().get_blob_client(container=AZURE_CONTAINER_MATCHOFF_OUTPUT, blob=blob_path)
    bc.upload_blob(buf, overwrite=True)
    return blob_path


# MOBILECLEANUP RECONCILE
def _resolve_fetch_columns(force: bool = False) -> list[str]:
    """Which of the wanted columns actually exist on mobilecleanup.

    Without this, adding a column to MOBILECLEANUP_COLUMN_LABELS that the live
    table doesn't have makes EVERY chunk query fail, which used to look exactly
    like "nothing matched". A missing optional column now degrades instead.
    """
    global _active_columns_cache
    if _active_columns_cache is not None and not force:
        return _active_columns_cache

    parts = PG_MATCHOFF_TABLE.split(".")
    table_name = parts[-1].strip('"')
    schema_name = parts[-2].strip('"') if len(parts) > 1 else None

    sql = "SELECT column_name FROM information_schema.columns WHERE table_name = %s"
    params: list = [table_name]
    if schema_name:
        sql += " AND table_schema = %s"
        params.append(schema_name)

    with _pg_cursor() as cur:
        cur.execute(sql, params)
        present = {r[0].lower() for r in cur.fetchall()}

    if not present:
        raise RuntimeError(
            f"Table '{PG_MATCHOFF_TABLE}' was not found. Check PG_MATCHOFF_TABLE "
            f"and that the HSBC data has been loaded."
        )

    if PG_MOB_COLUMN.lower() not in present:
        raise RuntimeError(
            f"Column '{PG_MOB_COLUMN}' does not exist on {PG_MATCHOFF_TABLE}. "
            f"Set PG_MOB_COLUMN to the column holding the cleaned mobile number."
        )

    wanted = list(dict.fromkeys([PG_MOB_COLUMN] + MOBILECLEANUP_FETCH_COLUMNS))
    active = [c for c in wanted if c.lower() in present]
    missing = [c for c in wanted if c.lower() not in present]

    if missing:
        print(
            f"[RECONCILE] {PG_MATCHOFF_TABLE} has no column(s): {', '.join(missing)} "
            f"— continuing without them."
        )
        if PG_GHO_COLUMN in missing:
            print(
                f"[RECONCILE] ! '{PG_GHO_COLUMN}' is missing, so every reconciled row "
                f"will fall back to the default classification. Add the column or set "
                f"PG_GHO_COLUMN to its real name."
            )

    _active_columns_cache = active
    return active


def gho_column_available() -> bool:
    try:
        return PG_GHO_COLUMN in _resolve_fetch_columns()
    except Exception:
        return False


def _fetch_chunk(mob_chunk: list[str], columns: list[str]) -> dict[str, dict]:
    if not mob_chunk:
        return {}

    sql = (
        f"SELECT {', '.join(columns)} FROM {PG_MATCHOFF_TABLE} "
        f"WHERE {PG_MOB_COLUMN} IN %s"
    )

    result = {}
    conn = _pg_conn()
    try:
        with conn.cursor() as cur:
            cur.execute(sql, (tuple(mob_chunk),))
            col_names = [desc[0] for desc in cur.description]
            for row in cur.fetchall():
                mob_val = str(row[0]).strip()
                result[mob_val] = {
                    col_names[i]: (str(val).strip() if val is not None else "")
                    for i, val in enumerate(row)
                }
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        _put_conn(conn)

    return result


def _mobile_variants(value: str) -> set[str]:
    """Every form of an Indian mobile number worth looking up.

    Covers the raw value plus the 10-digit and 91-prefixed forms, so the match
    succeeds whichever convention each side happens to use.
    """
    raw = str(value or "").strip()
    digits = re.sub(r"\D", "", raw)
    if not digits:
        return set()

    variants = {raw, digits}

    # Strip international padding down to a bare subscriber number
    core = digits
    for prefix in ("0091", "091", "0091", "91", "0"):
        if core.startswith(prefix) and len(core) - len(prefix) == 10:
            core = core[len(prefix):]
            break

    if len(core) == 10:
        variants.update({core, f"91{core}", f"091{core}", f"0{core}"})
    elif len(digits) == 12 and digits.startswith("91"):
        variants.update({digits[2:], digits})

    return {v for v in variants if v}


def match_mob_numbers(
    mob_numbers: list,
    chunk_size: int = SQL_CHUNK_SIZE,
    max_workers: int = SQL_MAX_WORKERS,
) -> dict[str, dict]:
    """Reconcile numbers against HSBC data.

    A query failure is raised, never absorbed. Swallowing it here would report a
    broken lookup as "zero matches", which silently empties the master tables.
    """
    originals = [str(m).strip() for m in mob_numbers if m is not None and str(m).strip()]
    if not originals:
        return {}

    # The DOT feeds and mobilecleanup do not always agree on the country prefix:
    # one side stores 9876543210, the other 919876543210. Matching the literal
    # string finds only the rows that happen to agree, which reads as a genuine
    # 0.04% match rate rather than the format mismatch it actually is. So look
    # up every plausible form of each number and map the hit back to the value
    # the caller passed in.
    variants_by_original: dict[str, set[str]] = {}
    lookup_to_originals: dict[str, set[str]] = {}

    for original in set(originals):
        variants = _mobile_variants(original)
        variants_by_original[original] = variants
        for variant in variants:
            lookup_to_originals.setdefault(variant, set()).add(original)

    all_variants = list(lookup_to_originals.keys())
    columns = _resolve_fetch_columns()
    chunks = list(_chunks(all_variants, chunk_size))
    raw_hits: dict[str, dict] = {}
    failures: list[Exception] = []

    with ThreadPoolExecutor(max_workers=max_workers) as pool:
        futures = {pool.submit(_fetch_chunk, chunk, columns): chunk for chunk in chunks}
        for future in as_completed(futures):
            try:
                raw_hits.update(future.result())
            except Exception as exc:
                failures.append(exc)

    if failures:
        raise RuntimeError(
            f"{len(failures)} of {len(chunks)} lookup batches failed against "
            f"{PG_MATCHOFF_TABLE}. First error: {failures[0]}"
        )

    # Re-key hits onto the values the caller supplied
    matched: dict[str, dict] = {}
    for hit_value, row in raw_hits.items():
        for original in lookup_to_originals.get(str(hit_value).strip(), ()):
            matched.setdefault(original, row)

    exact = sum(1 for k in matched if k in raw_hits)
    if matched and exact < len(matched):
        print(
            f"[RECONCILE] {len(matched) - exact:,} of {len(matched):,} matches needed "
            f"a prefix adjustment — the feed and {PG_MATCHOFF_TABLE} store numbers "
            f"in different formats"
        )

    return matched


def list_gho_codes_from_source() -> list[dict]:
    """Distinct GHO codes present in mobilecleanup, with row counts."""
    if not gho_column_available():
        return []

    with _pg_cursor(dict_cursor=True) as cur:
        cur.execute(
            f"""SELECT {PG_GHO_COLUMN} AS gho_code, COUNT(*) AS record_count
                FROM {PG_MATCHOFF_TABLE}
                WHERE {PG_GHO_COLUMN} IS NOT NULL AND {PG_GHO_COLUMN} <> ''
                GROUP BY {PG_GHO_COLUMN} ORDER BY {PG_GHO_COLUMN}"""
        )
        rows = cur.fetchall()

    return [
        {
            "gho_code": r["gho_code"],
            "record_count": r["record_count"],
            "segment": ms.segment_for(r["gho_code"]),
        }
        for r in rows
    ]


# METADATA CRUD
def save_export_metadata(
    module: str,
    file_name: str,
    blob_name: str,
    record_count: int,
    export_date: str,
    data_type: str = None,
    source: str = "feed",
    segment: str = None,
) -> str:
    """source='feed' for a raw FRI/MNRL download, 'reconciled' for a master
    export. The two archives are listed separately so the FRI and MNRL tabs
    show only what those tabs produced."""
    row_id = str(uuid.uuid4())
    container = _get_container_name(module, source)
    with _pg_cursor() as cur:
        cur.execute(
            """
            INSERT INTO export_files
            (id, module, file_name, blob_name, container, record_count,
             export_date, data_type, source, segment, created_at)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, NOW())
            """,
            (row_id, module.upper(), file_name, blob_name, container,
             record_count, export_date, data_type, source, segment),
        )
    return row_id


def list_export_metadata(module: str, source: str = "feed") -> list:
    """Files for one archive. 'feed' powers the FRI/MNRL tabs, 'reconciled' the
    Reconcile tab; passing None returns both."""
    sql = """
        SELECT id, file_name, blob_name, container, record_count, export_date,
               data_type, source, segment, created_at
        FROM export_files WHERE module = %s
    """
    params = [module.upper()]
    if source:
        sql += " AND source = %s"
        params.append(source)
    sql += " ORDER BY created_at DESC LIMIT 500"

    with _pg_cursor(dict_cursor=True) as cur:
        cur.execute(sql, params)
        rows = cur.fetchall()

    result = []
    for row in rows:
        d = dict(row)
        d["created_at"] = d["created_at"].isoformat()
        d["export_date"] = str(d["export_date"])
        result.append(d)
    return result


def get_export_metadata_row(file_id: str, module: str) -> dict:
    with _pg_cursor(dict_cursor=True) as cur:
        cur.execute(
            "SELECT * FROM export_files WHERE id = %s AND module = %s",
            (file_id, module.upper()),
        )
        row = cur.fetchone()
    if not row:
        raise ValueError(f"No export record found: id={file_id}, module={module}")
    return dict(row)


def save_matchoff_metadata(
    source_container: str,
    source_date_folder: str,
    source_file_name: str,
    output_blob_path: str | None,
    input_row_count: int,
    matched_count: int,
    mob_col: str,
    run_date: str,
    module: str | None = None,
    data_type: str | None = None,
    master: dict | None = None,
) -> str:
    row_id = str(uuid.uuid4())
    master = master or {}
    with _pg_cursor() as cur:
        cur.execute(
            """
            INSERT INTO matchoff_runs (
                id, source_container, source_date_folder, source_file_name,
                output_blob_path, output_container, input_row_count,
                matched_count, mob_column, run_date, module, data_type,
                master_inserted, master_updated, master_cleared, conflicts_ignored,
                created_at
            )
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, NOW())
            """,
            (
                row_id, source_container, source_date_folder, source_file_name,
                output_blob_path, AZURE_CONTAINER_MATCHOFF_OUTPUT, input_row_count,
                matched_count, mob_col, run_date, module, data_type,
                master.get("master_inserted", 0), master.get("master_updated", 0),
                master.get("cleared", 0), master.get("conflicts_ignored", 0),
            ),
        )
    return row_id

# EXPORT (raw feed -> blob) — used by the FRI / MNRL download tabs
def export_and_store(
    module: str,
    records: list,
    columns: list,
    file_name: str,
    export_date: str,
    data_type: str = None,
) -> dict:
    container = _get_container_name(module)
    total = len(records)

    parts_raw = build_excel_parts(
        records=records, columns=columns,
        base_file_name=file_name, max_rows=EXCEL_MAX_ROWS,
    )

    parts_meta = []
    for buf, part_fname, row_count in parts_raw:
        blob_name = f"{export_date}/{part_fname}"
        upload_blob(container, blob_name, buf)
        row_id = save_export_metadata(
            module=module, file_name=part_fname, blob_name=blob_name,
            record_count=row_count, export_date=export_date, data_type=data_type,
        )
        parts_meta.append({
            "id": row_id, "file_name": part_fname, "record_count": row_count,
            "export_date": export_date, "module": module.upper(), "data_type": data_type,
        })
        print(f"[EXPORT] Uploaded {module} part {blob_name} ({row_count:,} rows)")

    return {
        "parts": parts_meta,
        "total_records": total,
        "file_count": len(parts_meta),
        "export_date": export_date,
        "module": module.upper(),
        "data_type": data_type,
        "record_count": total,
        "file_name": parts_meta[0]["file_name"] if parts_meta else file_name,
    }


# EXPORT (master -> blob) — the reconcile tab's download
def export_master(
    module: str,
    feed_date: str | None = None,
    segment: str | None = None,
    record_status: str = "all",
    date_to: str | None = None,
) -> dict:
    """Stream the filtered master straight into workbook parts and upload each.

    record_status defaults to 'all' so a date's report carries both the numbers
    still requiring action and the ones that turned compliant that day.

    Nothing is materialised: rows arrive from a server-side cursor, go into a
    write-only worksheet, and each finished part is uploaded and dropped before
    the next begins.
    """
    module = module.upper()
    container = _get_container_name(module, "reconciled")

    total = ms.count_active(module, feed_date, segment, record_status, date_to)
    if total == 0:
        # An empty selection is a normal answer, not a failure. Returning a
        # result lets the UI say "nothing here" instead of showing a red error.
        return {
            "status": "empty",
            "module": module,
            "feed_date": feed_date,
            "segment": segment,
            "record_count": 0,
            "file_count": 0,
            "parts": [],
            "file_name": None,
            "sas_url": None,
            "message": (
                "No reconciled records match those filters yet. Reconcile a file "
                "for this date, or widen the classification."
            ),
        }

    columns = ms.EXPORT_COLUMNS[module]
    label_date = feed_date or datetime.now(timezone.utc).strftime("%Y-%m-%d")
    seg_label = (segment or "all").replace("-", "").lower()
    status_label = "" if record_status == "all" else f"_{record_status}"
    span = f"{feed_date}_to_{date_to}" if date_to and date_to != feed_date else label_date
    base_name = f"{module.lower()}_{seg_label}{status_label}_{span}_{total}_records.xlsx"

    row_iter = ms.iter_active(module, feed_date, segment, columns=columns,
                              record_status=record_status, date_to=date_to)

    parts_meta = []
    exported = 0
    for buf, part_fname, row_count, part_idx in build_excel_parts_stream(
        row_iter, base_name, columns, EXCEL_MAX_ROWS, labels=ms.EXPORT_LABELS
    ):
        blob_name = f"{label_date}/{part_fname}"
        upload_blob(container, blob_name, buf)
        buf.close()

        row_id = save_export_metadata(
            module=module, file_name=part_fname, blob_name=blob_name,
            record_count=row_count, export_date=label_date,
            data_type=(segment or "All segments"),
            source="reconciled", segment=segment,
        )
        parts_meta.append({
            "id": row_id, "file_name": part_fname, "record_count": row_count,
            "blob_name": blob_name,
        })
        exported += row_count
        print(f"[MASTER EXPORT] part {part_idx} -> {blob_name} ({row_count:,} rows)")

    sas_url = get_sas_download_url(container, parts_meta[0]["blob_name"], expiry_hours=24)

    return {
        "module": module,
        "feed_date": feed_date,
        "segment": segment,
        "record_status": record_status,
        "record_count": exported,
        "file_count": len(parts_meta),
        "parts": parts_meta,
        "file_name": parts_meta[0]["file_name"],
        "sas_url": sas_url,
        "export_date": label_date,
    }


# RECONCILE
def run_matchoff(
    source_container: str,
    date_folder: str,
    file_name: str,
    mob_col: str = "mob_num",
) -> dict:
    """Reconcile a source feed against HSBC data and write the master store.

    This is the ONLY writer of fri_master / mnrl_master: a number earns a row
    once it has matched mobilecleanup, and the match is also where cust_id and
    gho_code (hence IWPB / Non-IWPB) come from.

    No Excel is produced unless MATCHOFF_WRITE_EXCEL=1. Downloads come from
    export_master(), filtered by date and GHO code.
    """
    run_date = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    module = _module_for_container(source_container)
    is_fri = module == "FRI"

    print(f"[RECONCILE] Reading {source_container}/{date_folder}/{file_name}")
    input_rows, headers = read_excel_from_blob(source_container, date_folder, file_name)
    print(f"[RECONCILE] {len(input_rows):,} rows, columns: {headers}")

    mob_col = _resolve_mob_col(mob_col, headers)
    print(f"[RECONCILE] Reading mobile numbers from column '{mob_col}'")

    data_type = None if is_fri else _detect_mnrl_data_type(file_name, headers)
    # A clear-only feed legitimately matches nothing in HSBC data
    is_clear_feed = data_type == "Reactivated"

    if is_fri and FRI_STATUS_COLUMN in headers:
        eligible_rows = [
            row for row in input_rows
            if str(row.get(FRI_STATUS_COLUMN, "")).strip() != FRI_EXCLUDE_STATUS
        ]
        excluded_compliant = len(input_rows) - len(eligible_rows)
        print(
            f"[RECONCILE] {len(eligible_rows):,} actionable, "
            f"{excluded_compliant:,} already compliant"
        )
    else:
        eligible_rows = input_rows
        excluded_compliant = 0

    # A file with no eligible rows still has work to do: every row being
    # compliant means every row should CLEAR its master entry. Falling through
    # with an empty match set does exactly that — returning early here would
    # silently skip the clearing, which is the whole point of a compliant file.
    all_compliant = not eligible_rows and not is_clear_feed

    # Match EVERY row, compliant ones included. A compliant number has to be
    # reconciled too: without a cust_id it cannot be written to the master, and
    # the report would not be able to show what turned compliant today.
    mob_numbers = [row.get(mob_col) for row in input_rows]
    unique_count = len({str(m).strip() for m in mob_numbers if m})
    print(f"[RECONCILE] Matching {unique_count:,} unique numbers against mobilecleanup")

    matched_map = match_mob_numbers(mob_numbers)
    print(f"[RECONCILE] {len(matched_map):,} unique numbers found in HSBC data "
          f"(across all {len(input_rows):,} rows, compliant included)")

    # Master store gets ALL rows — including compliant ones — so it can spot a
    # number contradicted within the same feed date.
    try:
        if is_fri:
            master = ms.ingest_fri_reconciled(
                feed_date=date_folder, rows=input_rows, matched_map=matched_map,
                source_file=file_name, cust_id_key=PG_CUSTID_COLUMN,
                gho_key=PG_GHO_COLUMN,
            )
        else:
            master = ms.ingest_mnrl_reconciled(
                feed_date=date_folder, data_type=data_type, rows=input_rows,
                matched_map=matched_map, source_file=file_name,
                cust_id_key=PG_CUSTID_COLUMN, gho_key=PG_GHO_COLUMN,
            )
        print(
            f"[MASTER][{module}] inserted={master['master_inserted']:,} "
            f"updated={master['master_updated']:,} "
            f"cleared={master['cleared']:,} (new={master.get('cleared_new', 0):,}) "
            f"unreconciled={master['unreconciled']:,} "
            f"conflicts={master['conflicts_ignored']:,} "
            f"segments={master['by_segment']}"
        )
    except Exception as exc:
        print(f"[MASTER] X ingest failed for {file_name}: {exc}")
        raise

    matched_count = sum(
        1 for row in eligible_rows if str(row.get(mob_col, "")).strip() in matched_map
    )

    output_blob_path = None
    output_file_name = None
    sas_url = None

    if MATCHOFF_WRITE_EXCEL:
        base_name = file_name.replace(".xlsx", "")
        output_file_name = f"{base_name}_matched_{matched_count}_records.xlsx"
        hsbc_cols = [c for c in _resolve_fetch_columns() if c != PG_MOB_COLUMN]
        buf, _ = build_matched_excel(
            eligible_rows, headers, mob_col, matched_map, hsbc_columns=hsbc_cols
        )
        output_blob_path = upload_matchoff_output(buf, date_folder, output_file_name)
        sas_url = get_sas_download_url(
            AZURE_CONTAINER_MATCHOFF_OUTPUT, output_blob_path, expiry_hours=24
        )
        print(f"[RECONCILE] Wrote matchoff-output/{output_blob_path}")

    run_id = save_matchoff_metadata(
        source_container=source_container,
        source_date_folder=date_folder,
        source_file_name=file_name,
        output_blob_path=output_blob_path,
        input_row_count=len(input_rows),
        matched_count=matched_count,
        mob_col=mob_col,
        run_date=run_date,
        module=module,
        data_type=data_type,
        master=master,
    )

    # Two different denominators, so both are named explicitly:
    #   unique_matched  distinct numbers found in HSBC data across ALL rows,
    #                   compliant ones included
    #   actionable_matched  rows still requiring action that matched; these are
    #                   the ones written to the master as active
    # The gap between them is compliant numbers, which clear instead of adding.
    print(
        f"[RECONCILE] Done. input_rows={len(input_rows):,} "
        f"unique_matched={len(matched_map):,} "
        f"actionable_matched={matched_count:,} "
        f"compliant_rows={excluded_compliant:,} run_id={run_id}"
    )

    cleared = master.get("cleared", 0)

    if all_compliant:
        # Nothing to add, but the compliant rows still cleared their entries.
        status = "all_compliant"
        message = (
            f"Every one of the {len(input_rows):,} rows was already compliant. "
            f"{cleared:,} matching number(s) were cleared from the master; "
            f"there was nothing new to add."
        )
    elif matched_count == 0 and not is_clear_feed:
        status = "no_matches"
        message = (
            f"None of the {len(eligible_rows):,} numbers belong to an HSBC "
            f"customer, so nothing was added to the master. The file and the "
            f"lookup both worked."
            + (f" {cleared:,} compliant number(s) were still cleared." if cleared else "")
        )
    elif is_clear_feed and matched_count == 0:
        status = "cleared_only"
        message = (
            f"Reactivation file processed: {cleared:,} number(s) cleared from "
            f"the master. Reactivations don't add records, so no match was needed."
        )
    else:
        status = "done"
        message = None

    return {
        "id": run_id,
        "status": status,
        "message": message,
        "module": module,
        "data_type": data_type,
        "source_container": source_container,
        "date_folder": date_folder,
        "source_file": file_name,
        "input_row_count": len(input_rows),
        "excluded_compliant_count": excluded_compliant,
        "eligible_row_count": len(eligible_rows),
        "matched_count": matched_count,
        "output_file": output_file_name,
        "sas_url": sas_url,
        "run_date": run_date,
        "master": master,
    }


# ATR REPORT EXPORT
AZURE_CONTAINER_ATR_REPORTS = os.environ.get("AZURE_CONTAINER_ATR_REPORTS", "atr-reports")


def export_atr_report(
    run_id: str | None = None,
    module: str | None = None,
    date_from: str | None = None,
    date_to: str | None = None,
    outcome: str | None = None,
) -> dict:
    """Build the ATR submission report as Excel and upload it.

    One row per number per attempt, with every field that was sent to C-DOT and
    the response that came back — the same detail /api/atr/history/<mobile>
    returns, for a whole run or date range at once.

    Streamed from atr_events through the same part-splitting writer as the
    master export, so a fortnightly run covering lakhs of numbers stays within
    one workbook's worth of memory.
    """
    import atr as atr_service

    total = atr_service.count_report_rows(run_id, module, date_from, date_to, outcome)
    if total == 0:
        return {
            "status": "empty",
            "record_count": 0,
            "run_id": run_id,
            "module": module,
            "message": (
                "No submissions match those filters yet. A report appears here "
                "once reports have actually been sent."
            ),
        }

    label_date = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    if run_id:
        stem = f"atr_report_run_{str(run_id)[:8]}"
    else:
        parts = ["atr_report", (module or "all").lower()]
        if date_from:
            parts.append(str(date_from))
        stem = "_".join(parts)
    base_name = f"{stem}_{total}_records.xlsx"

    row_iter = atr_service.iter_report_rows(run_id, module, date_from, date_to, outcome)

    parts_meta = []
    exported = 0
    for buf, part_fname, row_count, part_idx in build_excel_parts_stream(
        row_iter, base_name, atr_service.REPORT_COLUMNS, EXCEL_MAX_ROWS,
        labels=atr_service.REPORT_LABELS,
    ):
        blob_name = f"{label_date}/{part_fname}"
        upload_blob(AZURE_CONTAINER_ATR_REPORTS, blob_name, buf)
        buf.close()
        parts_meta.append({
            "file_name": part_fname, "record_count": row_count, "blob_name": blob_name,
        })
        exported += row_count
        print(f"[ATR REPORT] part {part_idx} -> {blob_name} ({row_count:,} rows)")

    sas_url = get_sas_download_url(
        AZURE_CONTAINER_ATR_REPORTS, parts_meta[0]["blob_name"], expiry_hours=24
    )

    return {
        "status": "done",
        "run_id": run_id,
        "module": module,
        "record_count": exported,
        "file_count": len(parts_meta),
        "parts": parts_meta,
        "file_name": parts_meta[0]["file_name"],
        "sas_url": sas_url,
    }
