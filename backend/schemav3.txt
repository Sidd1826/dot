-- =============================================================================
-- DOT INTELLIGENCE PLATFORM — SCHEMA v3  (cumulative migration)
--
-- HOW TO RUN
--     psql "$AZURE_POSTGRES_DSN" -v ON_ERROR_STOP=1 -f schema_v3.sql
--
-- Safe to run repeatedly. Every statement is guarded, so re-running upgrades
-- whatever is missing and leaves existing rows untouched. Verified by running
-- it twice against a populated database: row counts and ATR state unchanged.
--
-- PREREQUISITE: the base schema.sql must already be applied (export_files,
-- mobilecleanup, matchoff_runs, jobs). The guard below stops with a clear
-- message rather than a confusing "relation does not exist" halfway through.
--
-- The whole file runs in ONE transaction: if anything fails, nothing is
-- applied and the database is exactly as it was.
-- =============================================================================

BEGIN;

DO $preflight$
DECLARE
    missing TEXT;
BEGIN
    SELECT string_agg(t, ', ') INTO missing
    FROM unnest(ARRAY['export_files', 'mobilecleanup', 'matchoff_runs', 'jobs']) AS t
    WHERE NOT EXISTS (
        SELECT 1 FROM information_schema.tables
        WHERE table_schema = 'public' AND table_name = t
    );

    IF missing IS NOT NULL THEN
        RAISE EXCEPTION
            'Base schema is missing these tables: %. Run schema.sql first, then re-run this file.',
            missing;
    END IF;
END
$preflight$;


CREATE EXTENSION IF NOT EXISTS "pgcrypto";


-- =============================================================================
-- mobilecleanup — GHO code (already implemented upstream; guard for safety)
-- =============================================================================
ALTER TABLE mobilecleanup ADD COLUMN IF NOT EXISTS gho_code VARCHAR(20);
CREATE INDEX IF NOT EXISTS idx_mobilecleanup_gho ON mobilecleanup (gho_code);


-- =============================================================================
-- TABLE: fri_master
-- One row per mobile_no. Populated ONLY by the reconcile stage.
-- =============================================================================
CREATE TABLE IF NOT EXISTS fri_master (
    id                  BIGSERIAL PRIMARY KEY,

    mobile_no           VARCHAR(20)  NOT NULL,
    cust_id             VARCHAR(100),
    gho_code            VARCHAR(20),           -- from mobilecleanup at reconcile
    segment             VARCHAR(15)  NOT NULL DEFAULT 'Non-IWPB'
                        CHECK (segment IN ('IWPB', 'Non-IWPB')),

    -- ── compliance lifecycle ─────────────────────────────────────────────────
    status              VARCHAR(10)  NOT NULL DEFAULT 'active'
                        CHECK (status IN ('active', 'cleared')),
    clear_reason        VARCHAR(15)  CHECK (clear_reason IN ('compliant')),
    cleared_on          DATE,
    cleared_at          TIMESTAMPTZ,

    -- ── rolling window ───────────────────────────────────────────────────────
    first_reported_on   DATE         NOT NULL,
    last_reported_on    DATE         NOT NULL,
    report_count        INTEGER      NOT NULL DEFAULT 1 CHECK (report_count >= 1),

    -- ── ATR lifecycle (per-number log lives here, not in a side table) ───────
    atr_status          VARCHAR(12)  NOT NULL DEFAULT 'pending'
                        CHECK (atr_status IN ('pending', 'sent', 'rejected', 'skipped')),
    atr_run_id          UUID,
    atr_action_type     VARCHAR(5),            -- BOA / BOT
    atr_action_taken    VARCHAR(5),
    atr_submitted_at    TIMESTAMPTZ,
    atr_error           TEXT,

    -- ── feed payload ─────────────────────────────────────────────────────────
    fri                 VARCHAR(50),
    fri_status          VARCHAR(50),
    lsa_id              VARCHAR(20),
    tsp_id              VARCHAR(20),

    payload             JSONB,
    created_at          TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ  NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_fri_master_mobile UNIQUE (mobile_no),
    CONSTRAINT chk_fri_master_clear CHECK (
        (status = 'active'  AND clear_reason IS NULL     AND cleared_on IS NULL)
     OR (status = 'cleared' AND clear_reason IS NOT NULL AND cleared_on IS NOT NULL)
    ),
    CONSTRAINT chk_fri_master_window CHECK (last_reported_on >= first_reported_on)
);

-- the download query: date + segment, active only
CREATE INDEX IF NOT EXISTS idx_fri_master_download
    ON fri_master (last_reported_on DESC, segment) WHERE status = 'active';
-- the ATR queue: keyset-paginated by id
CREATE INDEX IF NOT EXISTS idx_fri_master_atr_queue
    ON fri_master (segment, last_reported_on, id) WHERE status = 'active' AND atr_status = 'pending';
CREATE INDEX IF NOT EXISTS idx_fri_master_gho      ON fri_master (gho_code);
CREATE INDEX IF NOT EXISTS idx_fri_master_cleared  ON fri_master (cleared_on DESC) WHERE status = 'cleared';
CREATE INDEX IF NOT EXISTS idx_fri_master_purge    ON fri_master (last_reported_on);
CREATE INDEX IF NOT EXISTS idx_fri_master_run      ON fri_master (atr_run_id);


-- =============================================================================
-- TABLE: mnrl_master
-- =============================================================================
CREATE TABLE IF NOT EXISTS mnrl_master (
    id                      BIGSERIAL PRIMARY KEY,

    mobile_no               VARCHAR(20)  NOT NULL,
    cust_id                 VARCHAR(100),
    gho_code                VARCHAR(20),
    segment                 VARCHAR(15)  NOT NULL DEFAULT 'Non-IWPB'
                            CHECK (segment IN ('IWPB', 'Non-IWPB')),

    status                  VARCHAR(10)  NOT NULL DEFAULT 'active'
                            CHECK (status IN ('active', 'cleared')),
    clear_reason            VARCHAR(15)  CHECK (clear_reason IN ('reactivated')),
    cleared_on              DATE,
    cleared_at              TIMESTAMPTZ,
    reactivation_id         VARCHAR(50),

    first_reported_on       DATE         NOT NULL,
    last_reported_on        DATE         NOT NULL,
    report_count            INTEGER      NOT NULL DEFAULT 1 CHECK (report_count >= 1),

    atr_status              VARCHAR(12)  NOT NULL DEFAULT 'pending'
                            CHECK (atr_status IN ('pending', 'sent', 'rejected', 'skipped')),
    atr_run_id              UUID,
    atr_action_type         VARCHAR(5),
    atr_action_taken        VARCHAR(5),
    atr_submitted_at        TIMESTAMPTZ,
    atr_error               TEXT,

    date_of_disconnection   DATE,
    disconnection_reason_id VARCHAR(20),
    lsa_id                  VARCHAR(20),
    tsp_id                  VARCHAR(20),

    payload                 JSONB,
    created_at              TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    updated_at              TIMESTAMPTZ  NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_mnrl_master_mobile UNIQUE (mobile_no),
    CONSTRAINT chk_mnrl_master_clear CHECK (
        (status = 'active'  AND clear_reason IS NULL     AND cleared_on IS NULL)
     OR (status = 'cleared' AND clear_reason IS NOT NULL AND cleared_on IS NOT NULL)
    ),
    CONSTRAINT chk_mnrl_master_window CHECK (last_reported_on >= first_reported_on)
);

CREATE INDEX IF NOT EXISTS idx_mnrl_master_download
    ON mnrl_master (last_reported_on DESC, segment) WHERE status = 'active';
CREATE INDEX IF NOT EXISTS idx_mnrl_master_atr_queue
    ON mnrl_master (segment, last_reported_on, id) WHERE status = 'active' AND atr_status = 'pending';
CREATE INDEX IF NOT EXISTS idx_mnrl_master_gho     ON mnrl_master (gho_code);
CREATE INDEX IF NOT EXISTS idx_mnrl_master_cleared ON mnrl_master (cleared_on DESC) WHERE status = 'cleared';
CREATE INDEX IF NOT EXISTS idx_mnrl_master_purge   ON mnrl_master (last_reported_on);
CREATE INDEX IF NOT EXISTS idx_mnrl_master_run     ON mnrl_master (atr_run_id);


-- =============================================================================
-- TABLE: dot_feed_ledger
-- Records what each feed said, per date, so a number appearing on BOTH the add
-- side and the clear side of one date can be detected across separate jobs.
--   feed_kind 'add'   = FRI non-compliant | MNRL Normal
--   feed_kind 'clear' = FRI compliant     | MNRL Reactivated
-- =============================================================================
CREATE TABLE IF NOT EXISTS dot_feed_ledger (
    module      VARCHAR(10) NOT NULL CHECK (module IN ('FRI', 'MNRL')),
    feed_date   DATE        NOT NULL,
    feed_kind   VARCHAR(10) NOT NULL CHECK (feed_kind IN ('add', 'clear')),
    mobile_no   VARCHAR(20) NOT NULL,
    source_file TEXT,
    seen_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    PRIMARY KEY (module, feed_date, feed_kind, mobile_no)
);

CREATE INDEX IF NOT EXISTS idx_feed_ledger_conflict ON dot_feed_ledger (module, feed_date, mobile_no);
CREATE INDEX IF NOT EXISTS idx_feed_ledger_purge    ON dot_feed_ledger (feed_date);


-- =============================================================================
-- TABLE: dot_master_events — state changes only, never routine sightings
-- =============================================================================
CREATE TABLE IF NOT EXISTS dot_master_events (
    id          BIGSERIAL PRIMARY KEY,
    module      VARCHAR(10) NOT NULL CHECK (module IN ('FRI', 'MNRL')),
    mobile_no   VARCHAR(20) NOT NULL,
    event       VARCHAR(20) NOT NULL CHECK (event IN (
                    'compliant', 'reactivated', 'conflict_ignored', 'retracted'
                )),
    event_date  DATE        NOT NULL,
    source_file TEXT,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_dot_events_mobile  ON dot_master_events (module, mobile_no, event_date DESC);
CREATE INDEX IF NOT EXISTS idx_dot_events_conflict
    ON dot_master_events (module, event_date DESC) WHERE event = 'conflict_ignored';


-- =============================================================================
-- TABLE: atr_defaults
-- The ATR payload values applied to every number in a (module, segment).
-- Editable from the ATR tab — no redeploy to change an action code.
-- =============================================================================
CREATE TABLE IF NOT EXISTS atr_defaults (
    module                VARCHAR(10)  NOT NULL CHECK (module IN ('FRI', 'MNRL')),
    segment               VARCHAR(15)  NOT NULL CHECK (segment IN ('IWPB', 'Non-IWPB')),

    action_type           VARCHAR(5)   NOT NULL CHECK (action_type IN ('BOA', 'BOT')),
    action_taken          VARCHAR(5)   NOT NULL,
    investigation_details VARCHAR(500),
    grievance_received    VARCHAR(3)   NOT NULL DEFAULT 'No' CHECK (grievance_received IN ('Yes', 'No')),
    remarks               VARCHAR(100),

    -- BOT only; ignored when action_type = 'BOA'
    total_money_saved     NUMERIC(18, 2),
    transaction_declined  INTEGER,

    is_active             BOOLEAN      NOT NULL DEFAULT TRUE,
    updated_by            VARCHAR(100),
    updated_at            TIMESTAMPTZ  NOT NULL DEFAULT NOW(),

    PRIMARY KEY (module, segment),

    -- BOT mandates both amount fields per the C-DOT spec
    CONSTRAINT chk_atr_bot_fields CHECK (
        action_type <> 'BOT'
        OR (total_money_saved IS NOT NULL AND transaction_declined IS NOT NULL)
    )
);

-- =============================================================================
-- TABLE: atr_runs — one row per submission run
-- =============================================================================
CREATE TABLE IF NOT EXISTS atr_runs (
    id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    module            VARCHAR(10) NOT NULL CHECK (module IN ('FRI', 'MNRL')),
    segment           VARCHAR(15),                 -- NULL = all segments
    feed_date         DATE,                        -- NULL = all dates in window
    gho_code          VARCHAR(20),

    action_type       VARCHAR(5),
    action_taken      VARCHAR(5),

    total_selected    INTEGER NOT NULL DEFAULT 0,
    total_sent        INTEGER NOT NULL DEFAULT 0,
    total_rejected    INTEGER NOT NULL DEFAULT 0,
    total_skipped     INTEGER NOT NULL DEFAULT 0,  -- invalid mobile etc.
    batches_sent      INTEGER NOT NULL DEFAULT 0,
    batch_size        INTEGER NOT NULL DEFAULT 0,

    status            VARCHAR(12) NOT NULL DEFAULT 'running'
                      CHECK (status IN ('running', 'done', 'partial', 'failed')),
    dry_run           BOOLEAN NOT NULL DEFAULT FALSE,
    request_ids       JSONB,                       -- requestIds returned by C-DOT
    error             TEXT,

    started_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    finished_at       TIMESTAMPTZ,

    CONSTRAINT chk_atr_run_counts CHECK (
        total_sent + total_rejected + total_skipped <= total_selected
    )
);

CREATE INDEX IF NOT EXISTS idx_atr_runs_created ON atr_runs (started_at DESC);
CREATE INDEX IF NOT EXISTS idx_atr_runs_module  ON atr_runs (module, started_at DESC);


-- =============================================================================
-- TABLE: atr_rejects
-- ONLY rejected numbers. Accepted state is on the master row (atr_status='sent'),
-- so a million-row success doesn't get written twice.
-- =============================================================================
CREATE TABLE IF NOT EXISTS atr_rejects (
    id            BIGSERIAL PRIMARY KEY,
    run_id        UUID        NOT NULL REFERENCES atr_runs (id) ON DELETE CASCADE,
    module        VARCHAR(10) NOT NULL,
    mobile_no     VARCHAR(20) NOT NULL,
    cust_id       VARCHAR(100),
    batch_no      INTEGER,
    error_message TEXT,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_atr_rejects_run    ON atr_rejects (run_id);
CREATE INDEX IF NOT EXISTS idx_atr_rejects_mobile ON atr_rejects (module, mobile_no);

-- =============================================================================
-- MIGRATION — only does anything if an earlier build created a third segment.
-- There are exactly two classifications: IWPB, and Non-IWPB (which is CIB).
-- =============================================================================
DO $$
BEGIN
    -- move any parked rows into the default classification
    UPDATE fri_master  SET segment = 'Non-IWPB' WHERE segment NOT IN ('IWPB', 'Non-IWPB');
    UPDATE mnrl_master SET segment = 'Non-IWPB' WHERE segment NOT IN ('IWPB', 'Non-IWPB');
    DELETE FROM atr_defaults WHERE segment NOT IN ('IWPB', 'Non-IWPB');

    ALTER TABLE fri_master  DROP CONSTRAINT IF EXISTS fri_master_segment_check;
    ALTER TABLE mnrl_master DROP CONSTRAINT IF EXISTS mnrl_master_segment_check;
    ALTER TABLE atr_defaults DROP CONSTRAINT IF EXISTS atr_defaults_segment_check;

    ALTER TABLE fri_master
        ADD CONSTRAINT fri_master_segment_check CHECK (segment IN ('IWPB', 'Non-IWPB'));
    ALTER TABLE mnrl_master
        ADD CONSTRAINT mnrl_master_segment_check CHECK (segment IN ('IWPB', 'Non-IWPB'));
    ALTER TABLE atr_defaults
        ADD CONSTRAINT atr_defaults_segment_check CHECK (segment IN ('IWPB', 'Non-IWPB'));

    ALTER TABLE fri_master  ALTER COLUMN segment SET DEFAULT 'Non-IWPB';
    ALTER TABLE mnrl_master ALTER COLUMN segment SET DEFAULT 'Non-IWPB';
END $$;


-- =============================================================================
-- VIEWS
-- =============================================================================
CREATE OR REPLACE VIEW v_fri_active  AS SELECT * FROM fri_master  WHERE status = 'active';
CREATE OR REPLACE VIEW v_mnrl_active AS SELECT * FROM mnrl_master WHERE status = 'active';

-- what the ATR tab shows before you press send
CREATE OR REPLACE VIEW v_atr_queue AS
    SELECT 'FRI' AS module, segment, gho_code, last_reported_on AS feed_date,
           COUNT(*) AS pending
    FROM fri_master
    WHERE status = 'active' AND atr_status = 'pending'
    GROUP BY segment, gho_code, last_reported_on
    UNION ALL
    SELECT 'MNRL', segment, gho_code, last_reported_on, COUNT(*)
    FROM mnrl_master
    WHERE status = 'active' AND atr_status = 'pending'
    GROUP BY segment, gho_code, last_reported_on;

CREATE OR REPLACE VIEW v_master_conflicts AS
    SELECT module, mobile_no, event_date, source_file, created_at
    FROM dot_master_events
    WHERE event = 'conflict_ignored'
    ORDER BY event_date DESC, created_at DESC;

-- Rows reconciled BEFORE this migration have no snapshot. Backfill them from
-- mobilecleanup while that data is still loaded. Safe to re-run.
UPDATE fri_master m SET
    zbadid = COALESCE(m.zbadid, c.zbadid), zbnad1 = COALESCE(m.zbnad1, c.zbnad1),
    zbnad2 = COALESCE(m.zbnad2, c.zbnad2), zbnad3 = COALESCE(m.zbnad3, c.zbnad3),
    zbnad4 = COALESCE(m.zbnad4, c.zbnad4), zbnad5 = COALESCE(m.zbnad5, c.zbnad5),
    zbteno = COALESCE(m.zbteno, c.zbteno), zbten2 = COALESCE(m.zbten2, c.zbten2),
    zbpscd = COALESCE(m.zbpscd, c.zbpscd), zbiadd = COALESCE(m.zbiadd, c.zbiadd)
FROM mobilecleanup c
WHERE c.final_mob_num = m.mobile_no AND m.zbnad1 IS NULL;

UPDATE mnrl_master m SET
    zbadid = COALESCE(m.zbadid, c.zbadid), zbnad1 = COALESCE(m.zbnad1, c.zbnad1),
    zbnad2 = COALESCE(m.zbnad2, c.zbnad2), zbnad3 = COALESCE(m.zbnad3, c.zbnad3),
    zbnad4 = COALESCE(m.zbnad4, c.zbnad4), zbnad5 = COALESCE(m.zbnad5, c.zbnad5),
    zbteno = COALESCE(m.zbteno, c.zbteno), zbten2 = COALESCE(m.zbten2, c.zbten2),
    zbpscd = COALESCE(m.zbpscd, c.zbpscd), zbiadd = COALESCE(m.zbiadd, c.zbiadd)

-- ── 2. ATR state machine on the master rows ─────────────────────────────────
-- atr_state_hash is the duplicate guard. It fingerprints everything that
-- determines the report: classification, the feed status, and the action codes
-- in force. A number is eligible only when its stored hash differs from the
-- hash computed now, so re-running a cycle sends nothing, and a status change
-- makes it eligible again automatically.
DO $$
DECLARE t TEXT;
BEGIN
    FOREACH t IN ARRAY ARRAY['fri_master', 'mnrl_master'] LOOP
        EXECUTE format('ALTER TABLE %I ADD COLUMN IF NOT EXISTS atr_state_hash   CHAR(32)', t);
        EXECUTE format('ALTER TABLE %I ADD COLUMN IF NOT EXISTS atr_send_count   INTEGER NOT NULL DEFAULT 0', t);
        EXECUTE format('ALTER TABLE %I ADD COLUMN IF NOT EXISTS atr_last_sent_at TIMESTAMPTZ', t);
        EXECUTE format('ALTER TABLE %I ADD COLUMN IF NOT EXISTS atr_claimed_at   TIMESTAMPTZ', t);
        EXECUTE format('ALTER TABLE %I ADD COLUMN IF NOT EXISTS atr_last_trigger VARCHAR(24)', t);
    END LOOP;
END $$;

-- 'sending' is the in-flight claim; 'not_required' is a number that cleared
-- before its first report was ever due.
ALTER TABLE fri_master  DROP CONSTRAINT IF EXISTS fri_master_atr_status_check;
ALTER TABLE mnrl_master DROP CONSTRAINT IF EXISTS mnrl_master_atr_status_check;
ALTER TABLE fri_master  ADD CONSTRAINT fri_master_atr_status_check
    CHECK (atr_status IN ('pending', 'sending', 'sent', 'rejected', 'skipped', 'not_required'));
ALTER TABLE mnrl_master ADD CONSTRAINT mnrl_master_atr_status_check
    CHECK (atr_status IN ('pending', 'sending', 'sent', 'rejected', 'skipped', 'not_required'));

-- Claim queries filter on these
CREATE INDEX IF NOT EXISTS idx_fri_master_atr_claim
    ON fri_master (segment, id) WHERE status = 'active' AND atr_status <> 'sending';
CREATE INDEX IF NOT EXISTS idx_mnrl_master_atr_claim
    ON mnrl_master (segment, id) WHERE status = 'active' AND atr_status <> 'sending';
CREATE INDEX IF NOT EXISTS idx_fri_master_stuck
    ON fri_master (atr_claimed_at) WHERE atr_status = 'sending';
CREATE INDEX IF NOT EXISTS idx_mnrl_master_stuck
    ON mnrl_master (atr_claimed_at) WHERE atr_status = 'sending';


-- ── 3. atr_events — the single audit table ──────────────────────────────────
-- Replaces atr_rejects. One row per attempt per number, accepted or not,
-- holding exactly what was sent and exactly what came back. `trigger` and
-- `previous_state` answer "why was this sent again?".
CREATE TABLE IF NOT EXISTS atr_events (
    id              BIGSERIAL PRIMARY KEY,
    run_id          UUID        NOT NULL REFERENCES atr_runs (id) ON DELETE CASCADE,

    module          VARCHAR(10) NOT NULL CHECK (module IN ('FRI', 'MNRL')),
    mobile_no       VARCHAR(20) NOT NULL,
    cust_id         VARCHAR(100),
    segment         VARCHAR(15),

    outcome         VARCHAR(12) NOT NULL
                    CHECK (outcome IN ('sent', 'rejected', 'skipped', 'failed')),
    trigger         VARCHAR(24) NOT NULL DEFAULT 'first_send'
                    CHECK (trigger IN ('first_send', 'status_change', 'action_config_change',
                                       'retry_after_reject', 'manual')),

    action_type     VARCHAR(5),
    action_taken    VARCHAR(5),

    request_payload JSONB,        -- the exact atr_data entry sent to C-DOT
    response_code   VARCHAR(10),
    response_message TEXT,
    request_id      VARCHAR(100), -- C-DOT requestId for the batch

    state_hash      CHAR(32),     -- fingerprint at send time
    previous_state  JSONB,        -- what it was before, when this was a re-send
    current_state   JSONB,        -- what it is now

    attempt_no      INTEGER NOT NULL DEFAULT 1,
    batch_no        INTEGER,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_atr_events_run     ON atr_events (run_id, id);
CREATE INDEX IF NOT EXISTS idx_atr_events_mobile  ON atr_events (module, mobile_no, created_at DESC);

-- ── 5. atr_schedule — the fortnightly cycle ─────────────────────────────────
CREATE TABLE IF NOT EXISTS atr_schedule (
    module        VARCHAR(10) PRIMARY KEY CHECK (module IN ('FRI', 'MNRL')),
    enabled       BOOLEAN     NOT NULL DEFAULT TRUE,
    interval_days INTEGER     NOT NULL DEFAULT 14 CHECK (interval_days BETWEEN 1 AND 90),
    run_at_hour   INTEGER     NOT NULL DEFAULT 2  CHECK (run_at_hour BETWEEN 0 AND 23),
    next_run_at   TIMESTAMPTZ NOT NULL DEFAULT (date_trunc('day', NOW()) + INTERVAL '1 day' + INTERVAL '2 hours'),
    last_run_at   TIMESTAMPTZ,
    last_run_id   UUID,
    last_status   VARCHAR(12),
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ── 6. Audit views ──────────────────────────────────────────────────────────
-- Full history for one number: every report, why it was sent, what came back.
CREATE OR REPLACE VIEW v_atr_history AS
    SELECT e.module, e.mobile_no, e.cust_id, e.segment,
           e.outcome, e.trigger, e.attempt_no,
           e.action_type, e.action_taken,
           e.response_code, e.response_message, e.request_id,
           e.previous_state, e.current_state, e.request_payload,
           e.created_at,
           r.trigger_source, r.feed_date
    FROM atr_events e
    JOIN atr_runs r ON r.id = e.run_id
    ORDER BY e.module, e.mobile_no, e.created_at DESC;

-- Numbers reported more than once, i.e. where a status change forced a re-send
CREATE OR REPLACE VIEW v_atr_resends AS
    SELECT module, mobile_no, COUNT(*) AS send_count,
           MIN(created_at) AS first_sent, MAX(created_at) AS last_sent,
           ARRAY_AGG(trigger ORDER BY created_at) AS triggers
    FROM atr_events
    WHERE outcome = 'sent'
    GROUP BY module, mobile_no
    HAVING COUNT(*) > 1;
