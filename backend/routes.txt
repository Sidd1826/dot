FRI_COLUMNS = ["fri", "lsa_id", "mobile_no", "status", "tsp_id"]
MNRL_NORMAL_COLUMNS = [
    "date_of_disconnection", "disconnectionreason_id", "lsa_id", "mobile_no", "tsp_id",
]
MNRL_REACT_COLUMNS = ["date_of_reactivation", "mobile_no", "reactivation_id"]

_VALID_CONTAINERS = {"fri-exports", "mnrl-exports"}
_VALID_SEGMENTS = {"IWPB", "Non-IWPB", "UNKNOWN", "ALL"}


# ── VALIDATION ───────────────────────────────────────────────────────────────
def _validate_date(value: str) -> str:
    if not value:
        raise ValueError("Pick a date first.")
    try:
        return datetime.strptime(value, "%Y-%m-%d").strftime("%Y-%m-%d")
    except ValueError:
        raise ValueError("Dates use the format YYYY-MM-DD.")


def _optional_date(value: str | None) -> str | None:
    if not value or str(value).upper() == "ALL":
        return None
    return _validate_date(value)


def _validate_module(value: str) -> str:
    if not value or value.upper() not in ("FRI", "MNRL"):
        raise ValueError("Module must be FRI or MNRL.")
    return value.upper()


def _validate_data_type(value: str) -> str:
    if value not in ("Normal", "Reactivated"):
        raise ValueError("Data type must be Normal or Reactivated.")
    return value


def _validate_container(value: str) -> str:
    if not value or value not in _VALID_CONTAINERS:
        raise ValueError(f"Container must be one of: {', '.join(_VALID_CONTAINERS)}")
    return value


def _validate_segment(value: str | None) -> str | None:
    """Returns None for 'all', which means every segment."""
    if not value or value.upper() == "ALL":
        return None
    for seg in _VALID_SEGMENTS:
        if value.lower() == seg.lower():
            return None if seg == "ALL" else seg
    raise ValueError("Segment must be IWPB, Non-IWPB or All.")


def _optional_gho(value: str | None) -> str | None:
    if not value or value.upper() == "ALL":
        return None
    return value.strip()


def _fail(exc: Exception, status: int = 500):
    print(traceback.format_exc())
    return jsonify({"error": str(exc)}), status


# ── BACKGROUND WORK ──────────────────────────────────────────────────────────
def _do_fri_export(date: str, client_id, batch_size: int) -> dict:
    total = fri_count(key_id="FRI", count_date=date).get("count", 0)
    if total == 0:
        raise ValueError("No records published for that date.")

    records = fetch_and_decrypt_all(
        key_id="FRI", total_count=total, date=date,
        client_id=client_id, BATCH_SIZE=batch_size,
    )
    if not records:
        raise ValueError("The download returned no rows.")

    return export_and_store(
        module="FRI", records=records, columns=FRI_COLUMNS,
        file_name=f"fri_data_{date}_{len(records)}_records.xlsx",
        export_date=date,
    )


def _do_mnrl_export(date, data_type, bank_id, batch_size, max_workers) -> dict:
    if data_type == "Normal":
        total = mnrl_count(key_id="DIU", count_date=date).get("count", 0)
        if total == 0:
            raise ValueError("No records published for that date.")
        records = mnrl_data(
            key_id="DIU", date=date, total_count=total, BATCH_SIZE=batch_size,
            Bank_Id=bank_id, max_workers=max_workers,
        )
        columns = MNRL_NORMAL_COLUMNS
        file_name = f"mnrl_normal_{date}_{len(records)}_records.xlsx"
    else:
        total = mnrl_count_reactivated(key_id="DIU", count_date=date).get("count", 0)
        if total == 0:
            raise ValueError("No reactivated records published for that date.")
        records = mnrl_data_reactivated_excel(
            key_id="DIU", total_count=total, date=date, BATCH_SIZE=batch_size,
            Bank_Id=bank_id, max_workers=max_workers,
        )
        columns = MNRL_REACT_COLUMNS
        file_name = f"mnrl_reactivated_{date}_{len(records)}_records.xlsx"

    if not records:
        raise ValueError("The download returned no rows.")

    return export_and_store(
        module="MNRL", records=records, columns=columns, file_name=file_name,
        export_date=date, data_type=data_type,
    )


def _do_reconcile(container, date_folder, file_name, mob_col) -> dict:
    return run_matchoff(
        source_container=container, date_folder=date_folder,
        file_name=file_name, mob_col=mob_col,
    )


def _do_master_export(module, feed_date, segment, gho_code) -> dict:
    return export_master(
        module=module, feed_date=feed_date, segment=segment, gho_code=gho_code
    )


def _do_atr_run(module, feed_date, segment, gho_code, batch_size, limit, dry_run) -> dict:
    return atr_service.run_atr(
        module=module, feed_date=feed_date, segment=segment, gho_code=gho_code,
        batch_size=batch_size, limit=limit, dry_run=dry_run,
    )

@routes.route("/encrypt", methods=["POST"])
def encrypt():
    try:
        key_id = request.headers.get("X-PUBLIC-KEY-ID")
        if not key_id:

        return _fail(e)

# ── SHARED JOB STATUS ────────────────────────────────────────────────────────
@routes.route("/export/status/<job_id>", methods=["GET"])
def route_export_status(job_id):
    job = get_job(job_id)
    if not job:
        return jsonify({"error": "That job no longer exists."}), 404

    response = {"job_id": job_id, "status": job["status"]}
    if job["status"] == "done":
        response["result"] = job["result"]
    elif job["status"] == "failed":
        response["error"] = (job.get("error") or "Unknown error").strip().splitlines()[-1]
    return jsonify(response), 200

@routes.route("/atr/preview", methods=["GET"])
def route_atr_preview():
    """GET /atr/preview?type=FRI&date=...&segment=IWPB — what a run would send."""
    try:
        module = _validate_module(request.args.get("type", "FRI"))
        feed_date = _optional_date(request.args.get("date"))
        segment = _validate_segment(request.args.get("segment"))
        gho_code = _optional_gho(request.args.get("gho_code"))

        breakdown = []
        for seg in ([segment] if segment else [ms.SEGMENT_IWPB, ms.SEGMENT_NON_IWPB]):
            defaults = atr_service.get_defaults(module, seg)
            breakdown.append({
                "segment": seg,
                "pending": ms.count_atr_pending(module, feed_date, seg, gho_code),
                "defaults": defaults[0] if defaults else None,
            })

        return jsonify({
            "module": module,
            "date": feed_date,
            "gho_code": gho_code or "All",
            "total_pending": sum(b["pending"] for b in breakdown),
            "breakdown": breakdown,
        })
    except ValueError as e:
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        return _fail(e)
