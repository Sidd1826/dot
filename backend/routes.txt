_VALID_CONTAINERS = {"fri-exports", "mnrl-exports"}
_VALID_SEGMENTS = {"IWPB", "Non-IWPB", "ALL"}


# ── VALIDATION ───────────────────────────────────────────────────────────────
def _validate_date(value: str) -> str:
    if not value:
        raise ValueError("Pick a date first.")
    try:
        return datetime.strptime(value, "%Y-%m-%d").strftime("%Y-%m-%d")
    except ValueError:
        raise ValueError("Dates use the format YYYY-MM-DD.")


def _optional_date(value: str | None) -> str | None:
    if not value or str(value).upper() == "ALL":
        return None
    return _validate_date(value)


def _validate_module(value: str) -> str:
    if not value or value.upper() not in ("FRI", "MNRL"):
        raise ValueError("Module must be FRI or MNRL.")
    return value.upper()


def _validate_data_type(value: str) -> str:
    if value not in ("Normal", "Reactivated"):
        raise ValueError("Data type must be Normal or Reactivated.")
    return value


def _validate_container(value: str) -> str:
    if not value or value not in _VALID_CONTAINERS:
        raise ValueError(f"Container must be one of: {', '.join(_VALID_CONTAINERS)}")
    return value


def _validate_segment(value: str | None) -> str | None:
    """Returns None for 'all', which means every segment."""
    if not value or value.upper() == "ALL":
        return None
    for seg in _VALID_SEGMENTS:
        if value.lower() == seg.lower():
            return None if seg == "ALL" else seg
    raise ValueError("Segment must be IWPB, Non-IWPB or All.")


def _fail(exc: Exception, status: int = 500):
    print(traceback.format_exc())
    return jsonify({"error": str(exc)}), status


# ── BACKGROUND WORK ──────────────────────────────────────────────────────────
def _do_fri_export(date: str, client_id, batch_size: int) -> dict:
    total = fri_count(key_id="FRI", count_date=date).get("count", 0)
    if total == 0:
        raise ValueError("No records published for that date.")

    records = fetch_and_decrypt_all(
        key_id="FRI", total_count=total, date=date,
        client_id=client_id, BATCH_SIZE=batch_size,
    )
    if not records:
        raise ValueError("The download returned no rows.")

    return export_and_store(
        module="FRI", records=records, columns=FRI_COLUMNS,
        file_name=f"fri_data_{date}_{len(records)}_records.xlsx",
        export_date=date,
    )


def _do_mnrl_export(date, data_type, bank_id, batch_size, max_workers) -> dict:
    if data_type == "Normal":
        total = mnrl_count(key_id="DIU", count_date=date).get("count", 0)
        if total == 0:
            raise ValueError("No records published for that date.")
        records = mnrl_data(
            key_id="DIU", date=date, total_count=total, BATCH_SIZE=batch_size,
            Bank_Id=bank_id, max_workers=max_workers,
        )
        columns = MNRL_NORMAL_COLUMNS
        file_name = f"mnrl_normal_{date}_{len(records)}_records.xlsx"
    else:
        total = mnrl_count_reactivated(key_id="DIU", count_date=date).get("count", 0)
        if total == 0:
            raise ValueError("No reactivated records published for that date.")
        records = mnrl_data_reactivated_excel(
            key_id="DIU", total_count=total, date=date, BATCH_SIZE=batch_size,
            Bank_Id=bank_id, max_workers=max_workers,
        )
        columns = MNRL_REACT_COLUMNS
        file_name = f"mnrl_reactivated_{date}_{len(records)}_records.xlsx"

    if not records:
        raise ValueError("The download returned no rows.")

    return export_and_store(
        module="MNRL", records=records, columns=columns, file_name=file_name,
        export_date=date, data_type=data_type,
    )


def _do_reconcile(container, date_folder, file_name, mob_col) -> dict:
    return run_matchoff(
        source_container=container, date_folder=date_folder,
        file_name=file_name, mob_col=mob_col,
    )


def _do_master_export(module, feed_date, segment, record_status, date_to=None) -> dict:
    return export_master(module=module, feed_date=feed_date, segment=segment,
                         record_status=record_status, date_to=date_to)


def _do_atr_report(run_id, module, date_from, date_to, outcome) -> dict:
    return export_atr_report(run_id=run_id, module=module, date_from=date_from,
                             date_to=date_to, outcome=outcome)


def _do_atr_run(module, feed_date, segment, batch_size, limit,
                date_to=None, case_type=None) -> dict:
    return atr_service.run_atr(
        module=module, feed_date=feed_date, segment=segment,
        batch_size=batch_size, limit=limit, trigger_source="manual",
        date_to=date_to, case_type=case_type,
    )

@routes.route("/export/status/<job_id>", methods=["GET"])
def route_export_status(job_id):
    job = get_job(job_id)
    if not job:
        return jsonify({"error": "That job no longer exists."}), 404

    response = {"job_id": job_id, "status": job["status"]}
    if job["status"] == "done":
        response["result"] = job["result"]
    elif job["status"] == "failed":
        response["error"] = (job.get("error") or "Unknown error").strip().splitlines()[-1]
    return jsonify(response), 200

@routes.route("/atr/preview", methods=["GET"])
def route_atr_preview():
    """GET /atr/preview?type=FRI&date_from=…&date_to=…&segment=IWPB

    Broken down by classification AND case type, because an open case and a
    compliant one carry different action codes and are sent separately.
    'eligible' is what a run would actually send: numbers whose state differs
    from the one already reported.

    Deliberately three queries in total — all defaults, the counts, the
    schedule — rather than one per combination. The per-combination version
    issued a dozen and could exhaust the connection pool under load.
    """
    try:
        module = _validate_module(request.args.get("type", "FRI"))
        date_from = _optional_date(request.args.get("date_from") or request.args.get("date"))
        date_to = _optional_date(request.args.get("date_to")) or date_from
        segment = _validate_segment(request.args.get("segment"))

        all_defaults = atr_service.get_defaults(module)
        by_combo = {(d["case_type"], d["segment"]): d for d in all_defaults}

        counts = ms.atr_preview_counts(module, date_from, segment, date_to, by_combo)

        breakdown = []
        for row in counts:
            ct, seg = row["case_type"], row["segment"]
            d = by_combo.get((ct, seg))
            breakdown.append({
                "segment": seg,
                "case_type": ct,
                "case_label": atr_service.CASE_LABEL.get((module, ct), ct),
                "eligible": row["eligible"] if d and d.get("is_active", True) else 0,
                "never_sent": row["never_sent"],
                "in_window": row["in_window"],
                "defaults": d,
            })

        days = None
        if date_from and date_to:
            days = (datetime.strptime(date_to, "%Y-%m-%d")
                    - datetime.strptime(date_from, "%Y-%m-%d")).days + 1

        return jsonify({
            "module": module,
            "date_from": date_from,
            "date_to": date_to,
            "days_covered": days,
            "total_eligible": sum(b["eligible"] for b in breakdown),
            "breakdown": breakdown,
            "schedule": atr_service.get_schedule(module),
        })
    except ValueError as e:
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        return _fail(e)


@routes.route("/atr/submit", methods=["POST"])
def route_atr_submit():
    """POST /atr/submit {type, date, segment} — background job.

    Safe to call twice: eligibility is decided by the state fingerprint, so a
    second call finds nothing to send rather than duplicating reports.
    """
    req = request.get_json(silent=True) or {}
    try:
        module = _validate_module(req.get("type", "FRI"))
        feed_date = _optional_date(req.get("date_from") or req.get("date"))
        date_to = _optional_date(req.get("date_to")) or feed_date
        segment = _validate_segment(req.get("segment"))
        case_type = req.get("case_type")
        if case_type and case_type not in atr_service.CASE_TYPES:
            raise ValueError("case_type must be open or cleared")
        # None means "use the module's published cap"
        batch_size = int(req["batch_size"]) if req.get("batch_size") else None
        limit = int(req["limit"]) if req.get("limit") else None
    except (ValueError, TypeError) as e:
        return jsonify({"error": str(e)}), 400

    try:
        job_id = start_job(
            _do_atr_run, module, feed_date, segment, batch_size, limit,
            date_to, case_type
        )
        return jsonify({"job_id": job_id, "status": "pending"}), 202
    except Exception as e:
        return _fail(e)
