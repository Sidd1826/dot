"""
atr.py
======

Action Taken Report submission to C-DOT.

Flow
----
    master rows whose state differs from the one last reported
        -> claim a batch atomically (FOR UPDATE SKIP LOCKED -> 'sending')
        -> apply atr_defaults for (module, segment)
        -> build atr_data records per the C-DOT spec
        -> encrypt {client_id, atr_data} with the ATR RSA public key
        -> POST
        -> record every attempt in atr_events, store the new fingerprint

Sent once per state
-------------------
Eligibility is "the state fingerprint on the row differs from the one computed
now", not "the row is marked pending". Pressing Send twice therefore sends once.
A number is reported again only when its status, classification or action code
actually changed — and atr_events records which of those it was.

Cleared numbers are never selected, so a number that turned compliant during the
fortnight simply is not in the cycle. No special case, no cut-off date.

Volume handling
---------------
Batches are claimed and released one at a time, so a 10 lakh run costs one batch
of memory. Batch size adapts downward on HTTP 413.

FRI and MNRL differ only by URL, client_id, encryption key id and auth source —
everything else is shared. MNRL values are placeholders until C-DOT publishes
that endpoint.
"""

def _clean_url(raw: str | None, default: str = "") -> str:
    """Normalise a URL that came from a .env file.

    C-DOT publishes these endpoints inside documents, and what lands in .env is
    often not a bare URL: markdown link syntax `[https://x](https://x)`, angle
    brackets, surrounding quotes, a trailing slash, or a stray newline. Every
    one of those produces a 404 or a connection error that looks like a server
    problem rather than a config typo, so they are stripped here once.
    """
    text = (raw or default or "").strip().strip('"').strip("'").strip()
    if not text:
        return ""

    # markdown link: [label](target) -> target
    md = re.match(r"^\[.*?\]\((.*?)\)$", text)
    if md:
        text = md.group(1).strip()

    text = text.strip("<>").strip()

    # a doubled URL from a bad paste: https://a...https://a...
    if text.count("https://") > 1:
        text = "https://" + text.split("https://")[-1]

    return text.rstrip("/")

# ── RATE LIMITING ────────────────────────────────────────────────────────────
class _RateLimiter:
    """Sliding-window limiter, one per module.

    MNRL allows 120 requests per 10 minutes. A 10 lakh cycle is ~334 requests at
    3000 per batch, so a large run genuinely does need pacing. A small run must
    never pause: 112 records is a single request and cannot exhaust the budget.

    Two properties matter and both were wrong in an earlier version:

    1. TIMESTAMPS ARE WALL-CLOCK, NOT PROCESS-RELATIVE. time.monotonic() has an
       arbitrary origin, so entries could not be reasoned about or reset
       sensibly. time.time() lets a stale window be recognised and dropped.

    2. THE WINDOW SELF-HEALS. If every recorded call is older than the window,
       the list is cleared outright rather than trimmed. A crashed or abandoned
       run therefore cannot leave a long-lived process believing the budget is
       spent — which is what made a 112-record run pause for 443 seconds.

    A deliberate non-goal: this does not try to account for retries that urllib3
    performs inside session.post(). Those are invisible here, so the session
    used for ATR mounts no automatic retry — see _get_session(). Retries are
    driven by the explicit attempt loop in submit_batch(), which does pass
    through acquire().
    """

    def __init__(self, max_calls: int, per_seconds: int):
        self.max_calls = max_calls
        self.per_seconds = per_seconds
        self._calls: list[float] = []
        self._lock = threading.Lock()

    def _prune(self, now: float) -> None:
        """Drop calls that have fallen out of the window. Caller holds the lock."""
        cutoff = now - self.per_seconds
        if self._calls and self._calls[-1] <= cutoff:
            # Every entry is stale: clear rather than trim, so an abandoned run
            # cannot leave a phantom budget behind.
            self._calls.clear()
        else:
            self._calls = [t for t in self._calls if t > cutoff]

    def acquire(self) -> float:
        waited = 0.0
        while True:
            with self._lock:
                now = time.time()
                self._prune(now)
                if len(self._calls) < self.max_calls:
                    self._calls.append(now)
                    return waited
                # Wait only until the OLDEST call leaves the window
                sleep_for = max(0.0, self._calls[0] + self.per_seconds - now) + 0.05
            print(
                f"[ATR] rate limit reached ({len(self._calls)}/{self.max_calls} "
                f"requests in the last {self.per_seconds}s) — pausing {sleep_for:.0f}s"
            )
            time.sleep(sleep_for)
            waited += sleep_for

    def reset(self) -> None:
        """Forget the window. Called at the start of each run so a crashed or
        abandoned earlier run cannot make a fresh one wait."""
        with self._lock:
            self._calls.clear()

    def snapshot(self) -> dict:
        with self._lock:
            now = time.time()
            self._prune(now)
            return {
                "used": len(self._calls),
                "max_calls": self.max_calls,
                "per_seconds": self.per_seconds,
                "oldest_age_seconds": round(now - self._calls[0], 1) if self._calls else None,
            }


_limiters: dict[str, _RateLimiter] = {}
_limiter_lock = threading.Lock()


def _limiter(module: str) -> _RateLimiter | None:
    cfg = spec(module)["rate_limit"]
    if not cfg:
        return None
    with _limiter_lock:
        if module.upper() not in _limiters:
            _limiters[module.upper()] = _RateLimiter(*cfg)
        return _limiters[module.upper()]


def reset_rate_limiter(module: str) -> None:
    """Clear a module's window. Used at run start."""
    lim = _limiter(module)
    if lim:
        lim.reset()


def rate_limit_status(module: str) -> dict | None:
    lim = _limiter(module)
    return lim.snapshot() if lim else None


def effective_batch_size(module: str, requested: int | None = None) -> int:
    """Batch size for a module: its published cap unless explicitly overridden.

    Defaulting to the cap rather than a generic 1000 matters for MNRL, which
    allows 3000 entries per request but only 120 requests per 10 minutes. At
    1000 per batch a 10 lakh cycle needs ~1000 requests and spends most of its
    time paused; at 3000 it needs ~334.
    """
    cap = spec(module)["max_batch"]
    if requested:
        return max(1, min(requested, cap))
    return cap


# ── DB ───────────────────────────────────────────────────────────────────────
_pool: psycopg2.pool.ThreadedConnectionPool | None = None


def _get_pool() -> psycopg2.pool.ThreadedConnectionPool:
    global _pool
    if _pool is None or _pool.closed:
        _pool = psycopg2.pool.ThreadedConnectionPool(
            minconn=1, maxconn=ATR_PG_MAXCONN, dsn=PG_DSN
        )
    return _pool


class _cursor:
    def __init__(self, dict_cursor: bool = False):
        self.dict_cursor = dict_cursor
        self._conn = None
        self._cur = None

    def __enter__(self):
        # getconn() and cursor() are two steps. If the second one throws, the
        # connection is already out of the pool and __exit__ will NOT run —
        # Python only calls it when __enter__ returned. Without this guard each
        # such failure permanently removed one connection from the pool, and a
        # handful of them exhausted it until the process restarted.
        self._conn = _get_pool().getconn()
        try:
            factory = psycopg2.extras.RealDictCursor if self.dict_cursor else None
            self._cur = self._conn.cursor(cursor_factory=factory) if factory else self._conn.cursor()
        except Exception:
            try:
                _get_pool().putconn(self._conn)
            except Exception:
                pass
            self._conn = None
            raise
        return self._cur

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self._conn is None:
            return False
        try:
            self._conn.rollback() if exc_type else self._conn.commit()
        except Exception:
            pass
        try:
            if self._cur:
                self._cur.close()
        except Exception:
            pass
        try:
            _get_pool().putconn(self._conn)
        except Exception:
            pass
        self._conn = self._cur = None
        return False


# ── CONFIG VALIDATION ────────────────────────────────────────────────────────
_KEY_ID_PATTERN = re.compile(r"^[A-Za-z0-9_]{1,64}$")


def _redact(text: str) -> str:
    """Strip anything that looks like key material out of a message.

    Errors end up in logs, in atr_runs.error and on screen. A base64 RSA key
    has no business in any of those.
    """
    text = str(text)
    text = re.sub(r"-----BEGIN[^-]*-----.*?-----END[^-]*-----", "<key redacted>",
                  text, flags=re.S)
    text = re.sub(r"[A-Za-z0-9+/=]{64,}", "<redacted>", text)
    return text


def _key_id(module: str) -> str:
    """The configured key NAME, with a readable error if it holds a key."""
    raw = ATR_KEY_ID[module]
    var = f"{module}_ATR_KEY_ID"

    if not raw:
        raise ValueError(
            f"{var} is not set. It should be a short name such as "
            f"'{module}_ATR', with the key itself in {module}_ATR_PUBLIC_KEY."
        )

    if not _KEY_ID_PATTERN.match(raw):
        looks_like_key = len(raw) > 64 or "BEGIN" in raw or raw.endswith("==")
        if looks_like_key:
            raise ValueError(
                f"{var} contains the public key itself, but it should contain a "
                f"NAME. Set {var}={module}_ATR and move the key into "
                f"{module}_ATR_PUBLIC_KEY."
            )
        raise ValueError(
            f"{var} must be letters, digits and underscores only "
            f"(for example {module}_ATR)."
        )
    return raw


def preflight(module: str) -> dict:
    """Check everything needed to send, WITHOUT sending.

    Called before a run is created so a misconfiguration fails immediately
    rather than after a run row, a queue scan and a half-built batch.
    """
    module = module.upper()
    problems: list[str] = []

    try:
        key_id = _key_id(module)
        if not os.getenv(f"{key_id}_PUBLIC_KEY"):
            problems.append(
                f"{key_id}_PUBLIC_KEY is not set — that env var should hold the "
                f"base64 RSA public key C-DOT issued for ATR."
            )
    except ValueError as exc:
        key_id = None
        problems.append(str(exc))

    cfg = spec(module)
    if not ATR_CLIENT_ID.get(module):
        problems.append(
            f"{module} {cfg['id_field']} is not set "
            f"({'FRI_CLIENT_ID' if module == 'FRI' else 'MNRL_BANK_ID'})."
        )

    # MNRL requires investigation_details on every record, and it comes from the
    # per-classification defaults. Missing it fails every number in the batch,
    # so catch it here rather than at send time.
    if cfg["investigation_required"]:
        for row in get_defaults(module):
            if not row.get("investigation_details"):
                problems.append(
                    f"investigation_details is empty for {module} / "
                    f"{row['segment']} — it is mandatory on every MNRL record."
                )
            elif len(str(row["investigation_details"])) > cfg["max_investigation"]:
                problems.append(
                    f"investigation_details for {module} / {row['segment']} is "
                    f"longer than the {cfg['max_investigation']} characters MNRL allows."
                )

    url = ATR_URL.get(module) or ""
    if not url:
        problems.append(f"{module}_ATR_URL is not set.")
    elif "REPLACE-ME" in url:
        problems.append(
            f"{module}_ATR_URL is still the placeholder. C-DOT has not published "
            f"this endpoint yet."
        )

    return {
        "module": module,
        "ready": not problems,
        "key_id": key_id,
        "key_present": bool(key_id and os.getenv(f"{key_id}_PUBLIC_KEY")),
        "client_id_present": bool(ATR_CLIENT_ID.get(module)),
        "id_field": cfg["id_field"],
        "max_batch": cfg["max_batch"],
        "rate_limit": cfg["rate_limit"],
        "url": url,
        "problems": problems,
    }


# ── HTTP SESSION ─────────────────────────────────────────────────────────────
_session: requests.Session | None = None


def _get_session() -> requests.Session:
    global _session
    if _session is not None:
        return _session
    session = requests.Session()
    # NO automatic retry on the ATR session, deliberately.
    #
    # urllib3 retries happen INSIDE session.post(), so they reach the gateway
    # without passing through the rate limiter. That makes the limiter's count
    # diverge from the number of requests actually sent — the cause of a
    # 112-record run pausing for 443 seconds.
    #
    # Retries are instead driven by the explicit attempt loop in submit_batch(),
    # which calls limiter.acquire() before each one. Connection-level failures
    # are retried there too, so nothing is lost.
    retry = Retry(total=0, read=0, connect=0, status=0, backoff_factor=0)
    adapter = HTTPAdapter(max_retries=retry, pool_connections=10, pool_maxsize=10)
    session.mount("http://", adapter)
    session.mount("https://", adapter)
    _session = session
    return _session

def get_defaults(module: str, segment: str | None = None,
                 case_type: str | None = None) -> list[dict]:
    """ATR defaults, keyed by (module, segment, case_type).

    A compliant number needs a different action code from an open one, so both
    sets live here and a run picks the one matching what it is reporting.
    """
    sql = "SELECT * FROM atr_defaults WHERE module = %s"
    params: list[Any] = [module.upper()]
    if segment:
        sql += " AND segment = %s"
        params.append(segment)
    if case_type:
        sql += " AND case_type = %s"
        params.append(case_type)
    sql += " ORDER BY case_type, segment"

    with _cursor(dict_cursor=True) as cur:
        cur.execute(sql, params)
        rows = [dict(r) for r in cur.fetchall()]

    for row in rows:
        row["updated_at"] = row["updated_at"].isoformat() if row.get("updated_at") else None
        for key in ("total_money_saved",):
            if row.get(key) is not None:
                row[key] = float(row[key])
    return rows


def save_defaults(module: str, segment: str, payload: dict,
                  updated_by: str | None = None, case_type: str = CASE_OPEN) -> dict:
    """Upsert one (module, segment, case_type) default set."""
    if ATR_DEFAULTS_LOCKED:
        raise PermissionError(
            "ATR action codes are locked. They are a compliance decision and "
            "cannot be changed from the application. Set ATR_DEFAULTS_LOCKED=0 "
            "on the server to allow edits."
        )
    module = module.upper()
    cfg = spec(module)
    action_taken = str(payload.get("action_taken", "")).strip()
    action_type = str(payload.get("action_type", "")).strip().upper() or None

    if cfg["uses_action_type"]:
        if action_type not in VALID_ACTION_TYPES:
            raise ValueError("action_type must be BOA or BOT")
        allowed = BOA_ACTION_CODES if action_type == "BOA" else BOT_ACTION_CODES
        if action_taken not in allowed:
            raise ValueError(
                f"action_taken '{action_taken}' is not valid for {action_type}. "
                f"Allowed: {', '.join(sorted(allowed, key=int))}"
            )
    else:
        # MNRL has no action_type; one flat table of codes 1-10
        action_type = None
        if action_taken not in cfg["action_codes"]:
            raise ValueError(
                f"action_taken '{action_taken}' is not a valid MNRL code. "
                f"Allowed: {', '.join(sorted(cfg['action_codes'], key=int))}"
            )

    grievance = str(payload.get("grievance_received", "No")).strip().title()
    if grievance not in ("Yes", "No"):
        raise ValueError("grievance_received must be Yes or No")

    investigation = payload.get("investigation_details") or None
    if cfg["investigation_required"] and not investigation:
        raise ValueError(
            "investigation_details is mandatory for MNRL and must be at most "
            f"{cfg['max_investigation']} characters"
        )
    if investigation and len(str(investigation)) > cfg["max_investigation"]:
        raise ValueError(
            f"investigation_details must be at most {cfg['max_investigation']} "
            f"characters for {module}"
        )

    money = payload.get("total_money_saved")
    declined = payload.get("transaction_declined")
    if action_type == "BOT" and (money is None or declined is None):
        raise ValueError("BOT requires total_money_saved and transaction_declined")

    if not cfg["supports_remarks"]:
        payload = {**payload, "remarks": None}

    reactivation = str(payload.get("reactivation_id") or "").strip() or None
    if reactivation and reactivation not in MNRL_REACTIVATION_REASONS:
        raise ValueError(
            f"reactivation_id '{reactivation}' is not valid. Allowed: "
            f"{', '.join(sorted(MNRL_REACTIVATION_REASONS))}"
        )
    # Only meaningful on an MNRL reactivated report
    if not (module == "MNRL" and case_type == CASE_CLEARED):
        reactivation = None

    with _cursor() as cur:
        cur.execute(
            """
            INSERT INTO atr_defaults (
                module, segment, case_type, action_type, action_taken,
                investigation_details, grievance_received, remarks,
                total_money_saved, transaction_declined, reactivation_id,
                is_active, updated_by, updated_at
            )
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, NOW())
            ON CONFLICT (module, segment, case_type) DO UPDATE SET
                action_type           = EXCLUDED.action_type,
                action_taken          = EXCLUDED.action_taken,
                investigation_details = EXCLUDED.investigation_details,
                grievance_received    = EXCLUDED.grievance_received,
                remarks               = EXCLUDED.remarks,
                reactivation_id       = EXCLUDED.reactivation_id,
                total_money_saved     = EXCLUDED.total_money_saved,
                transaction_declined  = EXCLUDED.transaction_declined,
                is_active             = EXCLUDED.is_active,
                updated_by            = EXCLUDED.updated_by,
                updated_at            = NOW()
            """,
            (
                module, segment, case_type, action_type, action_taken,
                (payload.get("investigation_details") or None),
                grievance,
                (payload.get("remarks") or None),
                money, declined, reactivation,
                bool(payload.get("is_active", True)),
                updated_by,
            ),
        )
    return get_defaults(module, segment, case_type)[0]


# ── RECORD BUILDING ──────────────────────────────────────────────────────────
def _atr_mobile(value: Any) -> str | None:
    """C-DOT accepts 10 digits, or 91 + 10 digits. Anything else is rejected."""
    digits = re.sub(r"\D", "", str(value or ""))
    if len(digits) == 10:
        return digits
    if len(digits) == 12 and digits.startswith("91"):
        return digits
    if len(digits) == 13 and digits.startswith("091"):
        return digits[1:]
    return None


def _mnrl_reason(row: dict) -> str:
    """disconnectionreason_id for one number.

    Mandatory on every MNRL record and sourced from the feed, not the defaults —
    it is a property of why THAT number was revoked. Values outside 1-6 fall
    back to 6 (Others) rather than failing the number, since a rejected record
    helps nobody and the reason is descriptive rather than an instruction.
    """
    raw = str(row.get("disconnection_reason_id") or "").strip()
    return raw if raw in MNRL_DISCONNECTION_REASONS else MNRL_DEFAULT_DISCONNECTION_REASON


def _mnrl_reactivation_reason(row: dict, defaults: dict) -> str:
    """reactivation_id for one number.

    The number's OWN value wins: it is a property of why that number came back,
    captured from the reactivation feed at reconcile time, exactly like
    disconnectionreason_id. The configured default is only a fallback for rows
    whose feed did not carry one. Anything outside 1-5 falls back rather than
    failing the number.
    """
    raw = str(row.get("reactivation_id") or "").strip()
    if raw in MNRL_REACTIVATION_REASONS:
        return raw
    configured = str(defaults.get("reactivation_id") or "").strip()
    if configured in MNRL_REACTIVATION_REASONS:
        return configured
    return MNRL_DEFAULT_REACTIVATION_REASON


def build_record(row: dict, defaults: dict, date_of_action: str,
                 module: str = "FRI", case_type: str = "open") -> dict:
    """One atr_data entry, shaped to the module's contract.

    Raises ValueError if the row cannot be represented; the caller records that
    as a skipped number with the reason attached.
    """
    module = module.upper()
    cfg = spec(module)

    mobile = _atr_mobile(row.get("mobile_no"))
    if not mobile:
        raise ValueError(f"Invalid mobile_no format: {row.get('mobile_no')!r}")

    action_taken = str(defaults["action_taken"])
    grievance = defaults.get("grievance_received") or "No"
    investigation = defaults.get("investigation_details")

    if module == "MNRL":
        # MNRL: flat action table, no action_type, and investigation_details
        # plus disconnectionreason_id are both mandatory.
        if not investigation:
            raise ValueError(
                "investigation_details is mandatory for MNRL — set it in the "
                "ATR defaults for this classification"
            )
        record = {
            "mobile_no": mobile,
            "action_taken": action_taken,
            "investigation_details": str(investigation)[: cfg["max_investigation"]],
            "date_of_action": date_of_action,
            "disconnectionreason_id": _mnrl_reason(row),
            "grievance_received": grievance,
        }
        # Reactivated numbers additionally carry WHY they came back.
        if case_type == CASE_CLEARED and MNRL_SEND_REACTIVATION_ID:
            record["reactivation_id"] = _mnrl_reactivation_reason(row, defaults)
        return record

    # FRI: action_type drives which fields apply
    record = {
        "mobile_no": mobile,
        "action_type": defaults["action_type"],
        "action_taken": action_taken,
        "date_of_action": date_of_action,
        "grievance_received": grievance,
    }
    if investigation:
        record["investigation_details"] = str(investigation)[: cfg["max_investigation"]]

    remarks = defaults.get("remarks")
    if remarks:
        record["remarks"] = str(remarks)[: cfg["max_remarks"]]

    upi = row.get("upi_id") or defaults.get("upi_id")
    if upi:
        record["upi_id"] = str(upi)

    if defaults["action_type"] == "BOT":
        record["total_money_saved"] = str(defaults.get("total_money_saved") or "0")
        record["transaction_declined"] = str(defaults.get("transaction_declined") or "0")

    return record


# ── TRANSPORT ────────────────────────────────────────────────────────────────
class PayloadTooLarge(Exception):
    """HTTP 413 — caller should halve the batch and retry."""


def submit_batch(module: str, records: list[dict]) -> dict:
    """POST one encrypted batch. Returns a normalised result.

        {"outcome": "ok"|"partial"|"failed",
         "errors": {mobile_no: message}, "request_id": str|None,
         "http_status": int, "message": str}
    """
    module = module.upper()
    cfg = spec(module)
    url = ATR_URL[module]
    client_id = ATR_CLIENT_ID[module]
    if not client_id:
        raise ValueError(
            f"{module} {cfg['id_field']} is not configured — set "
            f"{'FRI_CLIENT_ID' if module == 'FRI' else 'MNRL_BANK_ID'}"
        )

    if len(records) > cfg["max_batch"]:
        raise ValueError(
            f"{module} allows at most {cfg['max_batch']} entries per request, "
            f"got {len(records)}"
        )

    # FRI sends client_id, MNRL sends bank_id
    payload = {cfg["id_field"]: str(client_id), "atr_data": records}
    encrypted = encrypt_data(payload, _key_id(module))
    session = _get_session()

    # None for FRI, which publishes no request-frequency limit
    limiter = _limiter(module)

    for attempt in range(1, ATR_MAX_RETRIES + 1):
        token = _token(module, force_refresh=(attempt > 1))
        headers = {"Content-Type": "application/json", "Authorization": token}
        headers.update(cfg["extra_headers"])  # MNRL requires Source: API

        if limiter:
            limiter.acquire()

        try:
            resp = session.post(
                url,
                json=encrypted,
                headers=headers,
                timeout=ATR_TIMEOUT,
                verify=ATR_VERIFY_SSL,
            )
        except (requests.exceptions.ConnectionError, requests.exceptions.Timeout) as exc:
            # The session performs no automatic retry (see _get_session), so
            # transport failures are handled here, where the limiter sees them.
            if attempt < ATR_MAX_RETRIES:
                wait = min(2 ** attempt, 20)
                print(f"[ATR] connection error, retrying in {wait}s: {exc}")
                time.sleep(wait)
                continue
            return {"outcome": "failed", "errors": {}, "request_id": None,
                    "http_status": 0, "message": f"Connection failed: {exc}"}

        if resp.status_code == 413:
            raise PayloadTooLarge(f"{len(records)} records rejected as too large")

        if resp.status_code == 401 and attempt < ATR_MAX_RETRIES:
            time.sleep(1)
            continue

        if resp.status_code == 429 and attempt < ATR_MAX_RETRIES:
            wait = int(resp.headers.get("Retry-After", 10))
            print(f"[ATR] Rate limited, waiting {wait}s")
            time.sleep(wait)
            continue

        try:
            body = resp.json() if resp.text else {}
        except ValueError:
            body = {"message": (resp.text or "")[:300]}

        # The two gateways disagree on capitalisation: FRI documents
        # requestId / errorIn, MNRL documents RequestId / Errorin. Matching
        # case-insensitively means neither spelling silently loses the
        # per-number rejection reasons.
        lower = {str(k).lower(): v for k, v in body.items()} if isinstance(body, dict) else {}
        code = str(lower.get("response_code") or resp.status_code)
        errors = {str(k): str(v) for k, v in (lower.get("errorin") or {}).items()}
        request_id = lower.get("requestid")
        message = str(lower.get("message") or "")

        if resp.status_code == 200 and code == "200":
            return {"outcome": "ok", "errors": {}, "request_id": request_id,
                    "http_status": 200, "message": message or "Success"}

        if resp.status_code == 200 and code == "422":
            # partial success — errorIn names the rejected numbers
            return {"outcome": "partial", "errors": errors, "request_id": request_id,
                    "http_status": 200, "message": message or "Partial success"}

        if resp.status_code in (500, 502, 503) and attempt < ATR_MAX_RETRIES:
            time.sleep(min(2 ** attempt, 20))
            continue

        # A gateway-level rejection tells the operator nothing unless it says
        # WHICH endpoint was called. "Not Found" on its own looks like a server
        # fault; "404 from <url>" points straight at the config.
        detail = message or (resp.text or "").strip()[:200] or resp.reason or ""
        env_var = f"{module}_ATR_URL"

        if resp.status_code == 404:
            hint = (
                f"404 Not Found from {url}. The gateway is reachable but that "
                f"path does not exist — check {env_var} against the C-DOT "
                f"specification (host AND path, including the /v2/atr suffix)."
            )
        elif resp.status_code == 401:
            hint = (
                f"401 Unauthorized from {url}. The ATR gateway rejected the "
                f"token. C-DOT may issue separate credentials for ATR; check "
                f"{module}_ATR_AUTH_* settings."
            )
        elif resp.status_code == 403:
            hint = (
                f"403 Forbidden from {url}. Credentials were accepted but this "
                f"client is not permitted to submit ATRs — usually an "
                f"onboarding or IP-allowlist issue at C-DOT's end."
            )
        else:
            hint = f"HTTP {resp.status_code} from {url}."
            if detail:
                hint += f" Gateway said: {detail}"

        return {"outcome": "failed", "errors": errors, "request_id": request_id,
                "http_status": resp.status_code, "url": url,
                "message": _redact(hint)}

    return {"outcome": "failed", "errors": {}, "request_id": None,
            "http_status": 0, "message": f"Exhausted {ATR_MAX_RETRIES} attempts"}


# ── RUN LOGGING ──────────────────────────────────────────────────────────────
def _create_run(module, segment, feed_date, defaults, total, batch_size,
                trigger_source="manual", date_to=None, case_type=None) -> str:
    """Open a run row. Records the DATE RANGE covered and how many days that is,
    so the audit shows a fortnight was sent in one go rather than 15 times."""
    run_id = str(uuid.uuid4())
    dfrom = ms._to_date(feed_date)
    dto = ms._to_date(date_to) or dfrom
    days = (dto - dfrom).days + 1 if dfrom and dto else None

    with _cursor() as cur:
        cur.execute(
            """
            INSERT INTO atr_runs (
                id, module, segment, feed_date, date_from, date_to, days_covered,
                case_type, action_type, action_taken,
                total_selected, batch_size, status, trigger_source, started_at
            )
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, 'running', %s, NOW())
            """,
            (run_id, module.upper(), segment, dfrom, dfrom, dto, days, case_type,
             (defaults or {}).get("action_type"), (defaults or {}).get("action_taken"),
             total, batch_size, trigger_source),
        )
    return run_id


def _finish_run(run_id, status, *, sent=0, rejected=0, skipped=0, resent=0,
                cleared=0, batches=0, request_ids=None, error=None) -> None:
    with _cursor() as cur:
        cur.execute(
            """
            UPDATE atr_runs
            SET status = %s, total_sent = %s, total_rejected = %s,
                total_skipped = %s, total_resent = %s, total_cleared = %s,
                batches_sent = %s, request_ids = %s::jsonb, error = %s,
                finished_at = NOW()
            WHERE id = %s
            """,
            (status, sent, rejected, skipped, resent, cleared, batches,
             json.dumps(request_ids or []), error, run_id),
        )


def _log_events(rows: list[tuple]) -> None:
    """Append to atr_events — the single audit table.

    One row per attempt per number, whatever the outcome, holding the exact
    payload sent and the response that came back. This is the only place ATR
    history is kept; the master row carries just the current state.
    """
    if not rows:
        return
    with _cursor() as cur:
        psycopg2.extras.execute_values(
            cur,
            """INSERT INTO atr_events (
                   run_id, module, mobile_no, cust_id, segment, outcome, trigger,
                   action_type, action_taken, request_payload, response_code,
                   response_message, request_id, state_hash, previous_state,
                   current_state, attempt_no, batch_no, case_type, feed_date
               ) VALUES %s""",
            rows,
            template=("(%s, %s, %s, %s, %s, %s, %s, %s, %s, %s::jsonb, %s, %s, %s, "
                      "%s, %s::jsonb, %s::jsonb, %s, %s, %s, %s::date)"),
            page_size=1000,
        )


def list_runs(module: str | None = None, limit: int = 50) -> list[dict]:
    sql = "SELECT * FROM atr_runs"
    params: list[Any] = []
    if module:
        sql += " WHERE module = %s"
        params.append(module.upper())
    sql += " ORDER BY started_at DESC LIMIT %s"
    params.append(limit)

    with _cursor(dict_cursor=True) as cur:
        cur.execute(sql, params)
        rows = [dict(r) for r in cur.fetchall()]

    for row in rows:
        row["id"] = str(row["id"])
        row["feed_date"] = str(row["feed_date"]) if row.get("feed_date") else None
        for key in ("started_at", "finished_at"):
            row[key] = row[key].isoformat() if row.get(key) else None
    return rows


def list_events(run_id: str | None = None, module: str | None = None,
                mobile_no: str | None = None, outcome: str | None = None,
                limit: int = 500) -> list[dict]:
    """Query the audit table. By run, by number, or by outcome."""
    clauses, params = [], []
    if run_id:
        clauses.append("run_id = %s")
        params.append(run_id)
    if module:
        clauses.append("module = %s")
        params.append(module.upper())
    if mobile_no:
        clauses.append("mobile_no = %s")
        params.append(ms._norm_mobile(mobile_no))
    if outcome:
        clauses.append("outcome = %s")
        params.append(outcome)
    where = " AND ".join(clauses) or "TRUE"
    params.append(limit)

    with _cursor(dict_cursor=True) as cur:
        cur.execute(
            f"""SELECT id, run_id, module, mobile_no, cust_id, segment, case_type,
                       feed_date, outcome,
                       trigger, action_type, action_taken, request_payload,
                       response_code, response_message, request_id,
                       previous_state, current_state, attempt_no, batch_no, created_at
                FROM atr_events WHERE {where}
                ORDER BY created_at DESC, id DESC LIMIT %s""",
            params,
        )
        rows = [dict(r) for r in cur.fetchall()]

    for row in rows:
        row["run_id"] = str(row["run_id"])
        row["created_at"] = row["created_at"].isoformat()
        row["feed_date"] = str(row["feed_date"]) if row.get("feed_date") else None
        row["case_label"] = CASE_LABEL.get(
            (row.get("module"), row.get("case_type")), row.get("case_type"))
    return rows


def number_history(mobile_no: str) -> dict:
    """Everything ever reported for one number, newest first.

    The answer to "when did we send an ATR for this number, what did we send,
    what came back, and was it ever re-sent because the status changed".
    """
    mobile = ms._norm_mobile(mobile_no)
    events = list_events(mobile_no=mobile, limit=200)
    return {
        "mobile_no": mobile,
        "master": ms.lookup(mobile).get("records", []),
        "events": events,
        "send_count": sum(1 for e in events if e["outcome"] == "sent"),
        "resend_count": max(0, sum(1 for e in events if e["outcome"] == "sent") - 1),
    }


# ── SCHEDULE ─────────────────────────────────────────────────────────────────
def get_schedule(module: str | None = None) -> list[dict]:
    sql = "SELECT * FROM atr_schedule"
    params: list[Any] = []
    if module:
        sql += " WHERE module = %s"
        params.append(module.upper())
    sql += " ORDER BY module"

    with _cursor(dict_cursor=True) as cur:
        cur.execute(sql, params)
        rows = [dict(r) for r in cur.fetchall()]

    for row in rows:
        for key in ("next_run_at", "last_run_at", "updated_at"):
            row[key] = row[key].isoformat() if row.get(key) else None
        row["last_run_id"] = str(row["last_run_id"]) if row.get("last_run_id") else None
    return rows


def run_now(module: str) -> dict:
    """Bring the next cycle forward to now. The ticker picks it up within
    ATR_SCHEDULER_TICK_SECONDS; pressing Send does the same thing immediately."""
    with _cursor() as cur:
        cur.execute(
            "UPDATE atr_schedule SET next_run_at = NOW(), updated_at = NOW() WHERE module = %s",
            (module.upper(),),
        )
    return get_schedule(module)[0]


def save_schedule(module: str, enabled: bool | None = None,
                  interval_days: int | None = None,
                  run_at_hour: int | None = None,
                  next_run_at: Any = None,
                  _internal: bool = False) -> dict:
    """Change the cycle settings.

    Locked by default. `_internal=True` is for the scheduler's own bookkeeping
    (advancing next_run_at after a run), which must keep working regardless.
    """
    if ATR_SCHEDULE_LOCKED and not _internal:
        raise PermissionError(
            "The ATR schedule is fixed and cannot be changed from the "
            "application. Set ATR_SCHEDULE_LOCKED=0 on the server to change it."
        )
    module = module.upper()
    sets, params = [], []

    if enabled is not None:
        sets.append("enabled = %s")
        params.append(bool(enabled))
    if interval_days is not None:
        if not 1 <= int(interval_days) <= 90:
            raise ValueError("Interval must be between 1 and 90 days.")
        sets.append("interval_days = %s")
        params.append(int(interval_days))
    if run_at_hour is not None:
        if not 0 <= int(run_at_hour) <= 23:
            raise ValueError("Hour must be between 0 and 23.")
        sets.append("run_at_hour = %s")
        params.append(int(run_at_hour))
        # Re-anchor the pending run to the new hour, otherwise changing it has
        # no effect until after the next cycle has already fired.
        sets.append("""next_run_at = (
            date_trunc('day', (next_run_at AT TIME ZONE timezone))
            + (%s || ' hours')::interval
        ) AT TIME ZONE timezone""")
        params.append(int(run_at_hour))
    if next_run_at:
        sets.append("next_run_at = %s")
        params.append(next_run_at)

    if not sets:
        return get_schedule(module)[0]

    sets.append("updated_at = NOW()")
    params.append(module)
    with _cursor() as cur:
        cur.execute(f"UPDATE atr_schedule SET {', '.join(sets)} WHERE module = %s", params)
    return get_schedule(module)[0]


def _advance_schedule(module: str, run_id: str | None, status: str) -> None:
    """Move next_run_at forward by one interval, anchored to the hour setting."""
    with _cursor() as cur:
        cur.execute(
            """
            UPDATE atr_schedule
            SET last_run_at = NOW(), last_run_id = %s, last_status = %s,
                -- run_at_hour is a LOCAL hour. Computing this in the server
                -- timezone would mean "2 AM" fired at 07:30 IST on a UTC server.
                next_run_at = (
                    date_trunc('day', (NOW() AT TIME ZONE timezone))
                    + (interval_days || ' days')::interval
                    + (run_at_hour || ' hours')::interval
                ) AT TIME ZONE timezone,
                updated_at  = NOW()
            WHERE module = %s
            """,
            (run_id, status, module.upper()),
        )


def due_modules() -> list[str]:
    with _cursor() as cur:
        cur.execute(
            "SELECT module FROM atr_schedule WHERE enabled AND next_run_at <= NOW() ORDER BY module"
        )
        return [r[0] for r in cur.fetchall()]


def run_due_cycles() -> list[dict]:
    """Run whatever the schedule says is due. Safe to call from anywhere.

    A Postgres advisory lock means only one process does the work even when
    several web workers tick at the same moment; the rest return immediately.
    """
    results = []
    for module in due_modules():
        # A DEDICATED connection, not one from the pool.
        #
        # A session-level advisory lock lives on its connection, so it has to be
        # held for the whole cycle — which for MNRL can be half an hour. Holding
        # a pooled connection that long starves every web request behind it, so
        # the lock gets its own short-lived connection instead.
        try:
            conn = psycopg2.connect(PG_DSN)
        except Exception as exc:
            print(f"[ATR] could not open a lock connection for {module}: {exc}")
            continue
        try:
            with conn.cursor() as cur:
                cur.execute("SELECT pg_try_advisory_lock(%s)", (_lock_key(module),))
                if not cur.fetchone()[0]:
                    print(f"[ATR] {module} cycle already running elsewhere, skipping")
                    continue
            conn.commit()
            try:
                result = run_atr(module, trigger_source="scheduled")
                _advance_schedule(module, result.get("run_id"), result.get("status", "done"))
                results.append(result)
            except Exception as exc:
                _advance_schedule(module, None, "failed")
                print(f"[ATR] scheduled {module} cycle failed: {exc}")
                results.append({"module": module, "status": "failed", "error": str(exc)})
            finally:
                try:
                    with conn.cursor() as cur:
                        cur.execute("SELECT pg_advisory_unlock(%s)", (_lock_key(module),))
                    conn.commit()
                except Exception:
                    pass  # closing the connection releases the lock anyway
        finally:
            try:
                conn.close()
            except Exception:
                pass
    return results


def _lock_key(module: str) -> int:
    """Stable 32-bit advisory lock id per module."""
    return 990000 + (1 if module.upper() == "FRI" else 2)


_scheduler_thread: threading.Thread | None = None


def start_scheduler() -> None:
    """Start the fortnightly cycle ticker. Call once at app start.

    Deliberately a plain thread with a DB advisory lock rather than a scheduler
    library: with several gunicorn workers every process would otherwise fire
    the same cycle. The lock makes running it everywhere harmless.
    """
    global _scheduler_thread
    if not ATR_SCHEDULER_ENABLED:
        print("[ATR] Scheduler disabled (ATR_SCHEDULER_ENABLED=0)")
        return
    if _scheduler_thread and _scheduler_thread.is_alive():
        return

    def _tick():
        while True:
            try:
                ran = run_due_cycles()
                for r in ran:
                    print(f"[ATR] scheduled cycle: {r}")
            except Exception as exc:
                print(f"[ATR] scheduler tick failed: {exc}")
            time.sleep(ATR_SCHEDULER_TICK_SECONDS)

    _scheduler_thread = threading.Thread(target=_tick, name="atr-scheduler", daemon=True)
    _scheduler_thread.start()
    print(
        f"[ATR] Scheduler started, ticking every {ATR_SCHEDULER_TICK_SECONDS}s"
    )


# ── MAIN ENTRY POINT ─────────────────────────────────────────────────────────
def run_atr(
    module: str,
    feed_date: Any = None,
    segment: str | None = None,
    batch_size: int | None = None,
    limit: int | None = None,
    trigger_source: str = "manual",
    date_to: Any = None,
    case_type: str | None = None,
) -> dict:
    """Send reports for every number whose state differs from the last one sent.

    Scoped to one classification at a time because the action code differs per
    classification; segment=None loops the ones that have work.
    """
    module = module.upper()

    # A run is scoped to one (segment, case_type) because the action code
    # differs for each. Anything unspecified fans out over the combinations
    # that actually have work.
    if not segment or str(segment).upper() == "ALL" or not case_type:
        results = []
        segs = ([segment] if segment and str(segment).upper() != "ALL"
                else [ms.SEGMENT_IWPB, ms.SEGMENT_NON_IWPB])
        cases = [case_type] if case_type else list(CASE_TYPES)
        for ct in cases:
            for seg in segs:
                defaults = get_defaults(module, seg, ct)
                if not defaults:
                    continue
                d = defaults[0]
                if not d.get("is_active", True):
                    continue
                if ms.count_atr_eligible(module, feed_date, seg,
                                         d["action_type"], str(d["action_taken"]),
                                         case_type=ct, date_to=date_to):
                    results.append(run_atr(module, feed_date, seg, batch_size,
                                           limit, trigger_source, date_to, ct))
        if not results:
            return {
                "status": "nothing_to_send",
                "module": module,
                "runs": [],
                "sent": 0,
                "rejected": 0,
                "message": (
                    "Every reconciled number has already been reported in its "
                    "current state. Nothing to send."
                ),
            }
        return {
            "status": "done",
            "module": module,
            "runs": results,
            "sent": sum(r.get("sent", 0) for r in results),
            "rejected": sum(r.get("rejected", 0) for r in results),
            "resent": sum(r.get("resent", 0) for r in results),
            "date_from": str(ms._to_date(feed_date)) if feed_date else None,
            "date_to": str(ms._to_date(date_to) or ms._to_date(feed_date)) if feed_date else None,
        }

    case_type = case_type or CASE_OPEN
    defaults_list = get_defaults(module, segment, case_type)
    if not defaults_list:
        raise ValueError(
            f"No ATR defaults are configured for {module} / {segment} / {case_type}."
        )
    defaults = defaults_list[0]
    if not defaults.get("is_active", True):
        raise ValueError(f"ATR defaults for {module} / {segment} are switched off.")

    action_type = defaults["action_type"]
    action_taken = str(defaults["action_taken"])
    batch_size = effective_batch_size(module, batch_size)

    # A previous run in this process may have left entries in the rate-limit
    # window — including one that crashed. Clearing it means a small run never
    # inherits a phantom budget and pauses for no reason.
    reset_rate_limiter(module)

    # Free anything a crashed run left claimed, or it would sit stuck forever
    released = ms.release_stuck_claims(module)
    if released:
        print(f"[ATR] released {released} stuck claim(s) from an earlier run")

    eligible = ms.count_atr_eligible(module, feed_date, segment, action_type,
                                     action_taken, case_type=case_type, date_to=date_to)
    cleared = ms.count_cleared_in_window(module, feed_date, segment)

    if eligible == 0:
        return {
            "status": "nothing_to_send",
            "module": module,
            "segment": segment,
            "sent": 0, "rejected": 0, "resent": 0,
            "cleared_skipped": cleared,
            "message": (
                f"Nothing to send for {segment}. Every number has already been "
                f"reported in its current state"
                + (f", and {cleared:,} turned compliant before this run."
                   if cleared else ".")
            ),
        }

    run_id = _create_run(module, segment, feed_date, defaults, eligible,
                         batch_size, trigger_source, date_to=date_to,
                         case_type=case_type)

    date_of_action = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    sent = rejected = skipped = resent = batches = 0
    request_ids: list[str] = []
    fatal_error: str | None = None

    print(
        f"[ATR] {module}/{segment} ({case_type}) run={run_id} eligible={eligible:,} "
        f"action={action_type}/{action_taken} source={trigger_source}"
    )
    print(f"[ATR] endpoint: {ATR_URL.get(module)}")

    # Safety net. Each pass must consume rows, so the number of passes cannot
    # exceed the number of eligible rows. If it ever does, something is wrong
    # with the claim predicate and stopping beats hammering C-DOT.
    max_passes = max(1, -(-eligible // max(1, batch_size))) + 5
    aborted_reason: str | None = None

    try:
        while True:
            if limit is not None and sent + rejected >= limit:
                break

            if batches >= max_passes:
                aborted_reason = (
                    f"Stopped after {batches} batches, more than the "
                    f"{max_passes} this queue should need. No further requests sent."
                )
                print(f"[ATR] X {aborted_reason}")
                break

            take = batch_size
            if limit is not None:
                take = min(batch_size, limit - (sent + rejected))

            rows = ms.claim_atr_batch(module, run_id, action_type, action_taken,
                                      feed_date, segment, take,
                                      case_type=case_type, date_to=date_to)
            if not rows:
                break
            batches += 1

            by_mobile: dict[str, dict] = {}
            records: list[dict] = []
            failed: dict[int, tuple[str, str]] = {}
            events: list[tuple] = []

            for row in rows:
                trigger = ms.classify_trigger(row, action_type, action_taken)
                row["_trigger"] = trigger
                row["_current"] = ms.state_snapshot(row, module, action_type, action_taken)
                row["_previous"] = {
                    "action_type": row.get("atr_action_type"),
                    "action_taken": row.get("atr_action_taken"),
                    "reported_before": bool(row.get("atr_state_hash")),
                    "previous_status": row.get("atr_status"),
                } if row.get("atr_state_hash") else None

                try:
                    record = build_record(row, defaults, date_of_action,
                                          module, case_type)
                except ValueError as exc:
                    failed[row["id"]] = ("skipped", str(exc))
                    events.append((
                        run_id, module, ms._norm_mobile(row["mobile_no"]),
                        row.get("cust_id"), row.get("segment"), "skipped", trigger,
                        action_type, action_taken, None, None, str(exc), None,
                        None, json.dumps(row["_previous"]) if row["_previous"] else None,
                        json.dumps(row["_current"]),
                        int(row.get("atr_send_count", 0)) + 1, batches,
                        case_type, row.get("cleared_on") or row.get("last_reported_on"),
                    ))
                    continue
                records.append(record)
                by_mobile[record["mobile_no"]] = row

            if not records:
                ms.complete_atr_batch(module, run_id, action_type, action_taken,
                                      failed=failed)
                _log_events(events)
                skipped += len(failed)
                continue

            chunks = [records]
            while chunks:
                chunk = chunks.pop(0)
                try:
                    result = submit_batch(module, chunk)
                except PayloadTooLarge:
                    if len(chunk) <= ATR_MIN_BATCH_SIZE:
                        raise
                    mid = len(chunk) // 2
                    print(f"[ATR] 413 on {len(chunk)} records — splitting to {mid}")
                    chunks = [chunk[:mid], chunk[mid:]] + chunks
                    continue

                if result["request_id"]:
                    request_ids.append(str(result["request_id"]))

                errors = result["errors"]
                by_number = {r["mobile_no"]: r for r in chunk}
                ok_hashes: dict[int, str] = {}

                for mobile, record in by_number.items():
                    row = by_mobile.get(mobile)
                    if row is None:
                        continue
                    attempt = int(row.get("atr_send_count", 0)) + 1
                    prev = json.dumps(row["_previous"]) if row["_previous"] else None
                    cur_state = json.dumps(row["_current"])

                    if result["outcome"] == "failed":
                        # Not a per-number rejection: the gateway refused the
                        # whole batch (auth, config, malformed payload). Every
                        # other batch would fail the same way, so the run stops
                        # after this one rather than sending the queue into a
                        # wall — and the numbers stay eligible for a retry.
                        outcome, message = "failed", result["message"]
                        failed[row["id"]] = ("rejected", message)
                        rejected += 1
                        fatal_error = message
                    elif mobile in errors:
                        outcome, message = "rejected", errors[mobile]
                        failed[row["id"]] = ("rejected", message)
                        rejected += 1
                    else:
                        outcome, message = "sent", result["message"]
                        ok_hashes[row["id"]] = row["new_hash"]
                        sent += 1
                        if row["_trigger"] != "first_send":
                            resent += 1

                    events.append((
                        run_id, module, ms._norm_mobile(mobile), row.get("cust_id"),
                        row.get("segment"), outcome, row["_trigger"],
                        action_type, action_taken, json.dumps(record),
                        str(result.get("http_status")), message,
                        result.get("request_id"),
                        row["new_hash"] if outcome == "sent" else None,
                        prev, cur_state, attempt, batches,
                        case_type, row.get("cleared_on") or row.get("last_reported_on"),
                    ))

                ms.complete_atr_batch(module, run_id, action_type, action_taken,
                                      sent=ok_hashes, failed=failed)
                failed = {}

            _log_events(events)
            skipped += sum(1 for e in events if e[5] == "skipped")

            if fatal_error:
                aborted_reason = (
                    f"C-DOT refused the batch: {fatal_error}. Stopped after "
                    f"{batches} batch(es); the remaining numbers are untouched."
                )
                print(f"[ATR] X {aborted_reason}")
                break

            if ATR_THROTTLE_MS:
                time.sleep(ATR_THROTTLE_MS / 1000)

    except Exception as exc:
        _finish_run(run_id, "failed", sent=sent, rejected=rejected, skipped=skipped,
                    resent=resent, cleared=cleared, batches=batches,
                    request_ids=request_ids, error=str(exc))
        ms.release_stuck_claims(module, minutes=0)
        print(f"[ATR] X run {run_id} failed: {exc}")
        raise

    if fatal_error and sent == 0:
        status = "failed"
    elif rejected or fatal_error:
        status = "partial"
    else:
        status = "done"
    _finish_run(run_id, status, sent=sent, rejected=rejected, skipped=skipped,
                resent=resent, cleared=cleared, batches=batches,
                request_ids=request_ids, error=aborted_reason or fatal_error)

    summary = {
        "status": status,
        "run_id": run_id,
        "module": module,
        "segment": segment,
        "feed_date": str(ms._to_date(feed_date)) if feed_date else None,
        "date_from": str(ms._to_date(feed_date)) if feed_date else None,
        "date_to": str(ms._to_date(date_to) or ms._to_date(feed_date)) if feed_date else None,
        "case_type": case_type,
        "case_label": CASE_LABEL.get((module, case_type), case_type),
        "action_type": action_type,
        "action_taken": action_taken,
        "selected": eligible,
        "sent": sent,
        "resent": resent,
        "rejected": rejected,
        "skipped": skipped,
        "cleared_skipped": cleared,
        "batches": batches,
        "trigger_source": trigger_source,
        "message": aborted_reason,
    }
    print(f"[ATR] Done: {summary}")
    return summary

        "action_taken": action_taken,

def count_report_rows(run_id: str | None = None, module: str | None = None,
                      date_from: Any = None, date_to: Any = None,
                      outcome: str | None = None) -> int:
    where, params = _report_filter(run_id, module, date_from, date_to, outcome)
    with _cursor() as cur:
        cur.execute(f"SELECT COUNT(*) FROM atr_events WHERE {where}", params)
        return int(cur.fetchone()[0])


def _report_filter(run_id, module, date_from, date_to, outcome) -> tuple[str, list]:
    clauses, params = [], []
    if run_id:
        clauses.append("run_id = %s")
        params.append(run_id)
    if module:
        clauses.append("module = %s")
        params.append(module.upper())
    if outcome and outcome.lower() != "all":
        clauses.append("outcome = %s")
        params.append(outcome)
    if date_from:
        clauses.append("created_at >= %s")
        params.append(ms._to_date(date_from))
    if date_to:
        clauses.append("created_at < (%s::date + 1)")
        params.append(ms._to_date(date_to))
    return (" AND ".join(clauses) or "TRUE"), params


def iter_report_rows(run_id: str | None = None, module: str | None = None,
                     date_from: Any = None, date_to: Any = None,
                     outcome: str | None = None) -> Iterator[dict]:
    """Stream audit rows as report rows.

    Server-side cursor, same as the master export: a fortnightly run covering
    10 lakh numbers produces 10 lakh audit rows, and the report must not build
    that as a list.
    """
    where, params = _report_filter(run_id, module, date_from, date_to, outcome)
    conn = _get_pool().getconn()
    try:
        cur = conn.cursor(name=f"atr_report_{os.getpid()}_{id(params)}",
                          cursor_factory=psycopg2.extras.RealDictCursor)
        cur.itersize = 5000
        cur.execute(
            f"""SELECT module, mobile_no, cust_id, segment, case_type, feed_date,
                       outcome, trigger, action_type, action_taken, request_payload,
                       response_code, response_message, request_id, attempt_no,
                       previous_state, created_at
                FROM atr_events WHERE {where}
                ORDER BY created_at DESC, id DESC""",
            params,
        )
        for row in cur:
            yield _report_row(dict(row))
        cur.close()
        conn.commit()
    except Exception:
        try:
            conn.rollback()
        except Exception:
            pass
        raise
    finally:
        try:
            _get_pool().putconn(conn)
        except Exception:
            pass


# ── CONNECTIVITY DIAGNOSTICS ─────────────────────────────────────────────────
def describe_config(module: str) -> dict:
    """What this process will actually use for a module. Nothing secret.

    Exists because a 404 is indistinguishable from a dozen config mistakes when
    all you can see is the word "Not Found".
    """
    module = module.upper()
    cfg = spec(module)
    url = ATR_URL.get(module) or ""
    key_id = ATR_KEY_ID.get(module) or ""
    key_env = f"{key_id}_PUBLIC_KEY" if key_id else None

    parsed = {}
    try:
        from urllib.parse import urlparse
        u = urlparse(url)
        parsed = {"scheme": u.scheme, "host": u.netloc, "path": u.path}
    except Exception:
        pass

    return {
        "module": module,
        "url": url,
        "url_parts": parsed,
        "url_env_var": f"{module}_ATR_URL",
        "key_id": key_id,
        "key_env_var": key_env,
        "public_key_present": bool(key_env and os.getenv(key_env)),
        "id_field": cfg["id_field"],
        "id_env_var": "FRI_CLIENT_ID" if module == "FRI" else "MNRL_BANK_ID",
        "id_present": bool(ATR_CLIENT_ID.get(module)),
        "extra_headers": cfg["extra_headers"],
        "max_batch": cfg["max_batch"],
        "rate_limit": cfg["rate_limit"],
        "verify_ssl": ATR_VERIFY_SSL,
        # Fail-soft: preflight reads atr_defaults, and diagnostics has to stay
        # usable when the database is exactly what is broken.
        "problems": _safe_problems(module),
    }


def _safe_problems(module: str) -> list[str]:
    try:
        return preflight(module).get("problems", [])
    except Exception as exc:
        return [f"Could not read ATR defaults from the database: {_redact(exc)}"]


def probe_endpoint(module: str) -> dict:
    """Check whether the configured ATR path exists, without submitting anything.

    A GET on a POST-only endpoint should come back 405 Method Not Allowed —
    which proves the path is right and only the method is wrong. A 404 means the
    URL itself is wrong. That single distinction separates "my config is broken"
    from "the gateway is broken", and costs nothing to check.
    """
    module = module.upper()
    url = ATR_URL.get(module) or ""
    if not url:
        return {"ok": False, "reason": f"{module}_ATR_URL is not set."}

    out = {"url": url}
    try:
        resp = _get_session().get(url, timeout=20, verify=ATR_VERIFY_SSL,
                                  headers=spec(module)["extra_headers"])
        out["http_status"] = resp.status_code
        out["reason"] = resp.reason
        body = (resp.text or "").strip()[:200]
        out["body_preview"] = _redact(body) if body else ""

        if resp.status_code in (405, 401, 403, 400, 422):
            out["ok"] = True
            out["verdict"] = (
                f"The path exists (HTTP {resp.status_code} to a GET is expected "
                f"on a POST-only endpoint). The URL is correct."
            )
        elif resp.status_code == 404:
            out["ok"] = False
            out["verdict"] = (
                "404 — the host is reachable but this path does not exist. "
                f"Check {module}_ATR_URL against the C-DOT specification."
            )
        else:
            out["ok"] = True
            out["verdict"] = f"Host reachable, returned HTTP {resp.status_code} to a GET."
    except requests.exceptions.SSLError as exc:
        out.update(ok=False, verdict=f"TLS failure: {exc}. Try ATR_VERIFY_SSL=0 for the test gateway.")
    except requests.exceptions.ConnectionError as exc:
        out.update(ok=False, verdict=(
            f"Cannot reach the host at all: {exc}. Check the hostname, DNS, and "
            f"whether this server needs a proxy to reach sancharsaathi.gov.in."
        ))
    except Exception as exc:
        out.update(ok=False, verdict=f"Probe failed: {_redact(exc)}")
    return out
