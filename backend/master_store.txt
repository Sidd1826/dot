"""
master_store.py  (v3)
=====================

Master store for FRI / MNRL. Written only by the reconcile stage, read by the
download and ATR stages.

Design rules
------------
1. RECONCILE GATE   A number enters a master table only if it matched HSBC
                    `mobilecleanup`. The match also supplies cust_id and
                    gho_code, so segment (IWPB / Non-IWPB) is known from the
                    moment the row is created — no later join.

2. SAME-DAY VETO    A number appearing on BOTH sides of one feed date is ignored
                    entirely: not added, not cleared, and retracted if an earlier
                    job already applied it.
                      FRI  add = status != compliant  | clear = compliant
                      MNRL add = Normal feed          | clear = Reactivated feed
                    Detected via dot_feed_ledger, so the two sides may arrive in
                    either order and in separate jobs.

3. NOTHING IS MATERIALISED   Reads for export and ATR are streamed — server-side
                    cursor for exports, keyset pagination for the ATR queue.
                    A 10 lakh row download never builds a 10 lakh element list.

Public API
----------
    ingest_fri_reconciled(...) / ingest_mnrl_reconciled(...)
    iter_active(module, feed_date, segment)             -> generator of dicts
    count_active(module, feed_date, segment)            -> int
    segment_counts(module, feed_date)                   -> live per-class counts
    iter_atr_queue(module, ...)                         -> generator of batches
    mark_atr_result(...) / count_atr_pending(...)
    list_feed_dates(module) / list_gho_codes(module)  (diagnostics only)
    master_stats(module) / conflicts(...) / lookup(...) / purge_expired()
"""

# ── CONNECTION POOL ──────────────────────────────────────────────────────────
_pool: psycopg2.pool.ThreadedConnectionPool | None = None


def _get_pool() -> psycopg2.pool.ThreadedConnectionPool:
    global _pool
    if _pool is None or _pool.closed:
        _pool = psycopg2.pool.ThreadedConnectionPool(minconn=1, maxconn=MAX_CONN, dsn=PG_DSN)
    return _pool


class _cursor:
    """Borrow a connection, yield a cursor, commit/rollback, return it."""

    def __init__(self, dict_cursor: bool = False):
        self.dict_cursor = dict_cursor
        self._conn = None
        self._cur = None

    def __enter__(self):
        # getconn() and cursor() are two steps. If the second one throws, the
        # connection is already out of the pool and __exit__ will NOT run —
        # Python only calls it when __enter__ returned. Without this guard each
        # such failure permanently removed one connection from the pool, and a
        # handful of them exhausted it until the process restarted.
        self._conn = _get_pool().getconn()
        try:
            factory = psycopg2.extras.RealDictCursor if self.dict_cursor else None
            self._cur = self._conn.cursor(cursor_factory=factory) if factory else self._conn.cursor()
        except Exception:
            try:
                _get_pool().putconn(self._conn)
            except Exception:
                pass
            self._conn = None
            raise
        return self._cur

    def __exit__(self, exc_type, exc_val, exc_tb):
        if self._conn is None:
            return False
        try:
            self._conn.rollback() if exc_type else self._conn.commit()
        except Exception:
            pass
        try:
            if self._cur:
                self._cur.close()
        except Exception:
            pass
        try:
            _get_pool().putconn(self._conn)
        except Exception:
            pass
        self._conn = self._cur = None
        return False


# ── VALUE / HEADER HELPERS ───────────────────────────────────────────────────
_DATE_FORMATS = (
    "%Y-%m-%d", "%d-%m-%Y", "%d/%m/%Y", "%Y/%m/%d",
    "%d-%b-%Y", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S",
)


def _canon(name: Any) -> str:
    """'Mobile No' -> 'mobile_no'. Excel headers are Title Cased by build_excel;
    records straight off the API are snake_case. One path reads both."""
    return re.sub(r"[^a-z0-9]+", "_", str(name).strip().lower()).strip("_")


def _header_map(rows: list[dict]) -> dict[str, str]:
    return {_canon(k): k for k in rows[0].keys()} if rows else {}


def _pick(row: dict, hmap: dict[str, str], *names: str) -> Any:
    for name in names:
        actual = hmap.get(_canon(name))
        if actual is not None:
            value = row.get(actual)
            if value is not None and str(value).strip() != "":
                return value
    return None


def _norm_mobile(value: Any) -> str:
    if value is None:
        return ""
    return re.sub(r"\D", "", str(value)).strip()


def _to_date(value: Any, fallback: date | None = None) -> date | None:
    if value is None or value == "":
        return fallback
    if isinstance(value, datetime):
        return value.date()
    if isinstance(value, date):
        return value
    text = str(value).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(text[: len(fmt) + 6], fmt).date()
        except ValueError:
            continue
    try:
        return datetime.fromisoformat(text).date()
    except ValueError:
        return fallback


def _txt(value: Any, limit: int | None = None) -> str | None:
    if value is None:
        return None
    s = str(value).strip()
    if not s:
        return None
    return s[:limit] if limit else s


def _payload(record: dict) -> psycopg2.extras.Json | None:
    return psycopg2.extras.Json(record) if STORE_PAYLOAD else None


def segment_for(gho_code: Any) -> str:
    """Classify a row from its GHO code. Always returns IWPB or Non-IWPB.

    A blank or unrecognised code falls back to DEFAULT_SEGMENT rather than
    creating a third bucket — a row parked in an 'unknown' segment would never
    be picked up by a classification-scoped ATR run and would sit pending
    forever. `master_stats()['missing_gho']` counts these so they stay visible.
    """
    code = _txt(gho_code)
    if not code:
        return DEFAULT_SEGMENT

    upper = code.upper()
    if upper in IWPB_GHO_CODES:
        return SEGMENT_IWPB
    if upper in NON_IWPB_GHO_CODES:
        return SEGMENT_NON_IWPB
    return DEFAULT_SEGMENT


# ── SQL BUILDERS ─────────────────────────────────────────────────────────────
_FRESHER = "(m.cleared_on IS NULL OR EXCLUDED.last_reported_on > m.cleared_on)"
_UPSERT_CACHE: dict[str, tuple[str, str]] = {}


def _upsert_sql(module: str) -> tuple[str, str]:
    if module in _UPSERT_CACHE:
        return _UPSERT_CACHE[module]

    table = _TABLES[module]
    mcols = _MODULE_COLS[module]

    insert_cols = (
        ["mobile_no", "cust_id", "gho_code", "segment"]
        + CUSTOMER_COLS
        + ["status", "first_reported_on", "last_reported_on", "report_count"]
        + [c for c, _ in mcols]
        + ["payload"]
    )
    template = (
        "(%s, %s, %s, %s, "
        + ", ".join(["%s"] * len(CUSTOMER_COLS))
        + ", 'active', %s::date, %s::date, 1, "
        + ", ".join(p for _, p in mcols)
        + ", %s)"
    )
    feed_updates = "\n".join(
        f"    {col} = COALESCE(EXCLUDED.{col}, m.{col}),"
        for col, _ in mcols
    )
    # Refresh the customer snapshot on every sighting, but never overwrite a
    # known value with a blank one.
    customer_updates = "\n".join(
        f"    {col} = COALESCE(EXCLUDED.{col}, m.{col})," for col in CUSTOMER_COLS
    )

    sql = f"""
INSERT INTO {table} AS m ({", ".join(insert_cols)})
VALUES %s
ON CONFLICT (mobile_no) DO UPDATE SET
    -- a sighting NEWER than the clearance re-opens the number;
    -- an older or replayed file can never resurrect a cleared row
    status       = CASE WHEN {_FRESHER} THEN 'active' ELSE m.status END,
    clear_reason = CASE WHEN {_FRESHER} THEN NULL     ELSE m.clear_reason END,
    cleared_at   = CASE WHEN {_FRESHER} THEN NULL     ELSE m.cleared_at END,
    cleared_on   = CASE WHEN {_FRESHER} THEN NULL     ELSE m.cleared_on END,

    -- Deliberately NOT resetting atr_status here. Whether a fresh report is
    -- owed is decided by the state fingerprint, so a number re-reported with an
    -- unchanged status stays settled instead of being sent again.

    first_reported_on = LEAST(m.first_reported_on, EXCLUDED.first_reported_on),
    last_reported_on  = GREATEST(m.last_reported_on, EXCLUDED.last_reported_on),
    report_count = m.report_count
        + CASE WHEN EXCLUDED.last_reported_on > m.last_reported_on THEN 1 ELSE 0 END,

    cust_id  = COALESCE(EXCLUDED.cust_id, m.cust_id),
{customer_updates}
    -- keep a known code rather than overwriting it with a blank one
    gho_code = COALESCE(EXCLUDED.gho_code, m.gho_code),
    segment  = CASE WHEN EXCLUDED.gho_code IS NULL THEN m.segment ELSE EXCLUDED.segment END,
{feed_updates}
    payload    = COALESCE(EXCLUDED.payload, m.payload),
    updated_at = NOW()
RETURNING (xmax::text::bigint = 0) AS was_insert
"""
    _UPSERT_CACHE[module] = (sql, template)
    return sql, template


def _clear_sql(module: str) -> tuple[str, str]:
    table = _TABLES[module]
    reason = _CLEAR_REASON[module]

    if module == MODULE_MNRL:
        return (
            f"""
UPDATE {table} AS m
SET status          = 'cleared',
    clear_reason    = '{reason}',
    cleared_on      = v.clear_date,
    cleared_at      = NOW(),
    reactivation_id = COALESCE(v.reactivation_id, m.reactivation_id),
    updated_at      = NOW()
FROM (VALUES %s) AS v(mobile_no, clear_date, reactivation_id)
WHERE m.mobile_no = v.mobile_no
  AND (m.cleared_on IS NULL OR v.clear_date > m.cleared_on)
RETURNING m.mobile_no
""",
            "(%s, %s::date, %s)",
        )

    return (
        f"""
UPDATE {table} AS m
SET status       = 'cleared',
    clear_reason = '{reason}',
    cleared_on   = v.clear_date,
    cleared_at   = NOW(),
    updated_at   = NOW()
FROM (VALUES %s) AS v(mobile_no, clear_date)
WHERE m.mobile_no = v.mobile_no
  AND (m.cleared_on IS NULL OR v.clear_date > m.cleared_on)
RETURNING m.mobile_no
""",
        "(%s, %s::date)",
    )

# ── LOW-LEVEL WRITES ─────────────────────────────────────────────────────────
def _write_events(module, mobiles, event, event_date, source_file) -> None:
    rows = [(module, m, event, event_date, source_file) for m in mobiles]
    if not rows:
        return
    with _cursor() as cur:
        psycopg2.extras.execute_values(cur, _EVENT_SQL, rows, page_size=PAGE_SIZE)


def _record_ledger(module, feed_date, kind, mobiles, source_file) -> int:
    rows = [(module, feed_date, kind, m, source_file) for m in set(mobiles)]
    if not rows:
        return 0
    with _cursor() as cur:
        psycopg2.extras.execute_values(cur, _LEDGER_SQL, rows, page_size=PAGE_SIZE)
    return len(rows)


def _ledger_conflicts(module, feed_date, mobiles) -> set[str]:
    candidates = list({m for m in mobiles if m})
    if not candidates:
        return set()
    with _cursor() as cur:
        cur.execute(_CONFLICT_SQL, (module, feed_date, candidates))
        return {r[0] for r in cur.fetchall()}


_HISTORY_SQL = """
INSERT INTO master_history
    (module, mobile_no, cust_id, segment, event_type, feed_date, status_value, source_file)
VALUES %s
ON CONFLICT (module, mobile_no, event_type, feed_date) DO NOTHING
"""


def _record_history(module: str, rows: list[tuple]) -> int:
    """Append lifecycle events. Append-only and idempotent.

    The master row holds current state; this holds the timeline. When a number
    flagged on the 10th turns compliant on the 14th, the master row is updated
    but BOTH events survive here, so "when was this disconnected, when did it
    become compliant" is always answerable.

    feed_date is the DATA date, never the day the reconcile happened to run.
    """
    if not rows:
        return 0
    with _cursor() as cur:
        psycopg2.extras.execute_values(cur, _HISTORY_SQL, rows, page_size=PAGE_SIZE)
    return len(rows)


def number_timeline(mobile_no: str, module: str | None = None) -> list[dict]:
    """Every recorded state for one number, oldest first."""
    mobile = _norm_mobile(mobile_no)
    if not mobile:
        return []
    sql = """SELECT module, mobile_no, cust_id, segment, event_type, feed_date,
                    status_value, source_file, recorded_at
             FROM master_history WHERE mobile_no = %s"""
    params: list[Any] = [mobile]
    if module:
        sql += " AND module = %s"
        params.append(module.upper())
    sql += " ORDER BY feed_date, id"

    with _cursor(dict_cursor=True) as cur:
        cur.execute(sql, params)
        rows = [dict(r) for r in cur.fetchall()]
    for r in rows:
        r["feed_date"] = str(r["feed_date"])
        r["recorded_at"] = r["recorded_at"].isoformat()
    return rows


def _upsert(module: str, rows: list[tuple]) -> dict:
    if not rows:
        return {"received": 0, "inserted": 0, "updated": 0}
    sql, template = _upsert_sql(module)
    with _cursor() as cur:
        result = psycopg2.extras.execute_values(
            cur, sql, rows, template=template, page_size=PAGE_SIZE, fetch=True
        )
    inserted = sum(1 for r in result if r[0])
    return {"received": len(rows), "inserted": inserted, "updated": len(result) - inserted}


_CLEARED_UPSERT_CACHE: dict[str, tuple[str, str]] = {}


def _cleared_upsert_sql(module: str) -> tuple[str, str]:
    """Insert-or-update a number as CLEARED.

    A number that reconciles with HSBC and arrives already compliant has never
    been in the master, so there is nothing to update. It still belongs there:
    the reconciled report has to be able to say "these turned compliant today".
    This inserts it in the cleared state, and updates an existing row the same
    way _clear would.
    """
    if module in _CLEARED_UPSERT_CACHE:
        return _CLEARED_UPSERT_CACHE[module]

    table = _TABLES[module]
    reason = _CLEAR_REASON[module]
    extra_col = ", reactivation_id" if module == MODULE_MNRL else ""
    extra_val = ", %s" if module == MODULE_MNRL else ""
    extra_upd = ("\n    reactivation_id = COALESCE(EXCLUDED.reactivation_id, m.reactivation_id),"
                 if module == MODULE_MNRL else "")

    cols = (["mobile_no", "cust_id", "gho_code", "segment"] + CUSTOMER_COLS
            + ["status", "clear_reason", "cleared_on", "cleared_at",
               "first_reported_on", "last_reported_on", "report_count", "atr_status"])

    template = (
        "(%s, %s, %s, %s, "
        + ", ".join(["%s"] * len(CUSTOMER_COLS))
        + f", 'cleared', '{reason}', %s::date, NOW(), %s::date, %s::date, 1, 'not_required'"
        + extra_val + ")"
    )
    customer_updates = "\n".join(
        f"    {c} = COALESCE(EXCLUDED.{c}, m.{c})," for c in CUSTOMER_COLS
    )

    sql = f"""
INSERT INTO {table} AS m ({", ".join(cols)}{extra_col})
VALUES %s
ON CONFLICT (mobile_no) DO UPDATE SET
    -- Only a clearance NEWER than the one on file wins, so replaying an old
    -- file can never un-clear or re-date a number.
    status       = CASE WHEN m.cleared_on IS NULL OR EXCLUDED.cleared_on > m.cleared_on
                        THEN 'cleared' ELSE m.status END,
    clear_reason = CASE WHEN m.cleared_on IS NULL OR EXCLUDED.cleared_on > m.cleared_on
                        THEN '{reason}' ELSE m.clear_reason END,
    cleared_on   = CASE WHEN m.cleared_on IS NULL OR EXCLUDED.cleared_on > m.cleared_on
                        THEN EXCLUDED.cleared_on ELSE m.cleared_on END,
    cleared_at   = CASE WHEN m.cleared_on IS NULL OR EXCLUDED.cleared_on > m.cleared_on
                        THEN NOW() ELSE m.cleared_at END,
    -- A number already reported keeps its ATR history; one that cleared before
    -- any report was due is marked so it never enters a cycle.
    atr_status   = CASE WHEN m.atr_send_count > 0 THEN m.atr_status ELSE 'not_required' END,
    first_reported_on = LEAST(m.first_reported_on, EXCLUDED.first_reported_on),
    cust_id  = COALESCE(EXCLUDED.cust_id, m.cust_id),
    gho_code = COALESCE(EXCLUDED.gho_code, m.gho_code),
    segment  = CASE WHEN EXCLUDED.segment = 'UNKNOWN' THEN m.segment ELSE EXCLUDED.segment END,
{customer_updates}{extra_upd}
    updated_at = NOW()
RETURNING m.mobile_no, (xmax::text::bigint = 0) AS was_insert
"""
    _CLEARED_UPSERT_CACHE[module] = (sql, template)
    return sql, template

    if not rows:
        return {"received": 0, "flagged": 0, "inserted": 0}
    sql, template = _cleared_upsert_sql(module)
    with _cursor() as cur:
        result = psycopg2.extras.execute_values(
            cur, sql, rows, template=template, page_size=PAGE_SIZE, fetch=True
        )
    mobiles = [r[0] for r in result]
    inserted = sum(1 for r in result if r[1])
    _write_events(module, mobiles, _CLEAR_REASON[module], feed_date, source_file)
    return {"received": len(rows), "flagged": len(mobiles), "inserted": inserted}


def _clear(module: str, values: list[tuple], feed_date: date, source_file) -> dict:
    if not values:
        return {"received": 0, "flagged": 0, "absent_or_already_cleared": 0}
    sql, template = _clear_sql(module)
    with _cursor() as cur:
        flagged = psycopg2.extras.execute_values(
            cur, sql, values, template=template, page_size=PAGE_SIZE, fetch=True
        )
    mobiles = [r[0] for r in flagged]
    _write_events(module, mobiles, _CLEAR_REASON[module], feed_date, source_file)
    return {
        "received": len(values),
        "flagged": len(mobiles),
        "absent_or_already_cleared": len(values) - len(mobiles),
    }


def _retract(module: str, mobiles: list[str], feed_date: date, source_file) -> dict:
    """Undo this feed date's effect for numbers now known to be contradicted."""
    if not mobiles:
        return {"deleted": 0, "unclear": 0}

    table = _TABLES[module]
    with _cursor() as cur:
        cur.execute(
            f"""DELETE FROM {table}
                WHERE mobile_no = ANY(%s) AND first_reported_on = %s AND report_count = 1""",
            (mobiles, feed_date),
        )
        deleted = cur.rowcount
        cur.execute(
            f"""UPDATE {table}
                SET status = 'active', clear_reason = NULL, cleared_on = NULL,
                    cleared_at = NULL, updated_at = NOW()
                WHERE mobile_no = ANY(%s) AND cleared_on = %s""",
            (mobiles, feed_date),
        )
        uncleared = cur.rowcount

    if deleted or uncleared:
        _write_events(module, mobiles, "retracted", feed_date, source_file)
    return {"deleted": deleted, "unclear": uncleared}


def purge_expired(retention_days: int = RETENTION_DAYS,
                  ledger_days: int = LEDGER_RETENTION_DAYS,
                  grace_hours: int = PURGE_GRACE_HOURS) -> dict:
    """Drop rows that have fallen out of the rolling window.

    The `grace_hours` guard is load-bearing. The feed date comes from the source
    file's date folder, so reconciling an export older than the retention window
    would otherwise delete its own rows the instant they landed — the run logs a
    healthy insert count and the table comes back empty. Anything written or
    touched in the last `grace_hours` is protected; it becomes eligible on the
    next run, by which point the count is visible in the reconcile result.
    """
    out: dict[str, int] = {}
    with _cursor() as cur:
        for module, table in _TABLES.items():
            cur.execute(
                f"""DELETE FROM {table}
                    WHERE last_reported_on < (CURRENT_DATE - %s::int)
                      AND (cleared_on IS NULL OR cleared_on < (CURRENT_DATE - %s::int))
                      AND updated_at < NOW() - (%s::int * INTERVAL '1 hour')""",
                (retention_days, retention_days, grace_hours),
            )
            out[f"{module.lower()}_purged"] = cur.rowcount
        cur.execute(
            f"""DELETE FROM {LEDGER_TABLE}
                WHERE feed_date < (CURRENT_DATE - %s::int)
                  AND seen_at < NOW() - (%s::int * INTERVAL '1 hour')""",
            (ledger_days, grace_hours),
        )
        out["ledger_purged"] = cur.rowcount
    return out


def retention_window_check(feed_date: Any) -> dict:
    """Is this feed date already outside the rolling window?

    Reconciling such a file still works, but the rows are purge-eligible once
    the grace period ends. The reconcile result carries this so the UI can say
    so plainly instead of the data quietly vanishing overnight.
    """
    fdate = _to_date(feed_date)
    if not fdate:
        return {"outside_window": False, "age_days": None}
    age = (date.today() - fdate).days
    return {
        "outside_window": age > RETENTION_DAYS,
        "age_days": age,
        "retention_days": RETENTION_DAYS,
        "grace_hours": PURGE_GRACE_HOURS,
    }


def _customer_values(hsbc: dict) -> tuple:
    """The mobilecleanup name/address/phone snapshot, in CUSTOMER_COLS order."""
    return tuple(
        _txt(hsbc.get(col), CUSTOMER_COL_LIMITS.get(col)) for col in CUSTOMER_COLS
    )


def _dedupe_rows(rows: Iterable[tuple]) -> list[tuple]:
    """Last-wins dedupe on mobile_no (element 0).

    Required: ON CONFLICT DO UPDATE raises if one statement touches the same
    target row twice, and daily feeds do contain repeats.
    """
    seen: dict[str, tuple] = {}
    for row in rows:
        seen[row[0]] = row
    return list(seen.values())


# ── INGEST: FRI ──────────────────────────────────────────────────────────────
def ingest_fri_reconciled(
    feed_date: Any,
    rows: list[dict],
    matched_map: dict[str, dict],
    source_file: str | None = None,
    cust_id_key: str = "cust_id",
    gho_key: str = "gho_code",
) -> dict:
    """Apply a reconciled FRI feed to fri_master.

    rows        every row read from the FRI export, compliant ones included —
                the store needs both sides to spot a same-day contradiction
    matched_map mobile -> mobilecleanup row. Membership IS the reconcile gate.
    """
    fdate = _to_date(feed_date) or date.today()
    hmap = _header_map(rows)
    matched = {_norm_mobile(k): v for k, v in (matched_map or {}).items()}

    add_records: dict[str, dict] = {}
    clear_dates: dict[str, date] = {}
    invalid = 0

    for record in rows:
        mobile = _norm_mobile(_pick(record, hmap, "mobile_no", "mob_num", "mobile"))
        if not mobile:
            invalid += 1
            continue
        status = str(_pick(record, hmap, "status") or "").strip().lower()
        if status == FRI_COMPLIANT_STATUS:
            clear_dates[mobile] = fdate
        else:
            add_records[mobile] = record

    # ledger FIRST, so conflicts are visible to any later job for this date
    _record_ledger(MODULE_FRI, fdate, KIND_ADD, add_records.keys(), source_file)
    _record_ledger(MODULE_FRI, fdate, KIND_CLEAR, clear_dates.keys(), source_file)

    conflicted = _ledger_conflicts(
        MODULE_FRI, fdate, list(add_records.keys()) + list(clear_dates.keys())
    )
    conflicted |= set(add_records) & set(clear_dates)
    for mobile in conflicted:
        add_records.pop(mobile, None)
        clear_dates.pop(mobile, None)
    if conflicted:
        _write_events(MODULE_FRI, conflicted, "conflict_ignored", fdate, source_file)

    # reconcile gate
    reconciled_rows, unreconciled = [], 0
    reported_history: list[tuple] = []
    segment_counts = {s: 0 for s in SEGMENTS}
    for mobile, record in add_records.items():
        hsbc = matched.get(mobile)
        if hsbc is None:
            unreconciled += 1
            continue
        gho = _txt(hsbc.get(gho_key), 20)
        seg = segment_for(gho)
        segment_counts[seg] += 1
        # Timeline entry, built from named values rather than tuple positions
        reported_history.append((
            MODULE_FRI, mobile, _txt(hsbc.get(cust_id_key), 100), seg,
            "reported", fdate, _txt(_pick(record, hmap, "status"), 50), source_file,
        ))
        reconciled_rows.append((
            mobile,
            _txt(hsbc.get(cust_id_key), 100),
            gho,
            seg,
            *_customer_values(hsbc),
            fdate, fdate,
            _txt(_pick(record, hmap, "fri"), 50),
            _txt(_pick(record, hmap, "status"), 50),
            _txt(_pick(record, hmap, "lsa_id"), 20),
            _txt(_pick(record, hmap, "tsp_id"), 20),
            _payload(record),
        ))

    upsert_stats = _upsert(MODULE_FRI, _dedupe_rows(reconciled_rows))

    # Timeline: this number was on the risk feed on this DATA date.
    _record_history(MODULE_FRI, reported_history)

    # Compliant numbers. Those that reconcile with HSBC are inserted-or-updated
    # as cleared rows so the report can list what turned compliant today; the
    # rest can only update a row that already exists.
    cleared_rows, unmatched_clears = [], []
    cleared_history: list[tuple] = []
    for mobile, clear_date in clear_dates.items():
        hsbc = matched.get(mobile)
        if hsbc is None:
            unmatched_clears.append((mobile, clear_date))
            continue
        gho = _txt(hsbc.get(gho_key), 20)
        cleared_rows.append((
            mobile,
            _txt(hsbc.get(cust_id_key), 100),
            gho,
            segment_for(gho),
            *_customer_values(hsbc),
            clear_date, clear_date, clear_date,
        ))
        cleared_history.append((
            MODULE_FRI, mobile, _txt(hsbc.get(cust_id_key), 100), segment_for(gho),
            "compliant", clear_date, "compliant", source_file,
        ))

    cleared_stats = _upsert_cleared(MODULE_FRI, _dedupe_rows(cleared_rows),
                                    fdate, source_file)

    # Timeline: and this number became compliant on this DATA date. The earlier
    # 'reported' row is untouched, so the disconnection record survives.
    _record_history(MODULE_FRI, cleared_history)
    clear_stats = _clear(MODULE_FRI, unmatched_clears, fdate, source_file)
    retract_stats = _retract(MODULE_FRI, sorted(conflicted), fdate, source_file)
    purged = purge_expired()

    return {
        "module": MODULE_FRI,
        "feed_date": str(fdate),
        "rows_in_file": len(rows),
        "invalid_mobile": invalid,
        "conflicts_ignored": len(conflicted),
        # Two clear paths, and BOTH must be counted: reconciled numbers are
        # inserted/updated as cleared rows by _upsert_cleared, unreconciled ones
        # can only update a row that already exists. Reporting just the second
        # made a run that cleared 14 reconciled numbers log "cleared=0".
        "cleared": cleared_stats["flagged"] + clear_stats["flagged"],
        "cleared_reconciled": cleared_stats["flagged"],
        "cleared_new": cleared_stats["inserted"],
        "clear_unmatched": clear_stats["absent_or_already_cleared"],
        "unreconciled": unreconciled,
        "master_inserted": upsert_stats["inserted"],
        "master_updated": upsert_stats["updated"],
        "by_segment": segment_counts,
        "retracted_deleted": retract_stats["deleted"],
        "retracted_uncleared": retract_stats["unclear"],
        "purged": purged,
        "retention": retention_window_check(fdate),
    }


# ── INGEST: MNRL ─────────────────────────────────────────────────────────────
def ingest_mnrl_reconciled(
    feed_date: Any,
    data_type: str,
    rows: list[dict],
    matched_map: dict[str, dict],
    source_file: str | None = None,
    cust_id_key: str = "cust_id",
    gho_key: str = "gho_code",
) -> dict:
    """data_type 'Normal' -> additions (reconcile-gated); 'Reactivated' -> clears."""
    fdate = _to_date(feed_date) or date.today()
    hmap = _header_map(rows)
    matched = {_norm_mobile(k): v for k, v in (matched_map or {}).items()}
    is_react = str(data_type).strip().lower().startswith("react")
    kind = KIND_CLEAR if is_react else KIND_ADD

    feed: dict[str, dict] = {}
    invalid = 0
    for record in rows:
        mobile = _norm_mobile(_pick(record, hmap, "mobile_no", "mob_num", "mobile"))
        if not mobile:
            invalid += 1
            continue
        feed[mobile] = record

    _record_ledger(MODULE_MNRL, fdate, kind, feed.keys(), source_file)
    conflicted = _ledger_conflicts(MODULE_MNRL, fdate, feed.keys())
    for mobile in conflicted:
        feed.pop(mobile, None)
    if conflicted:
        _write_events(MODULE_MNRL, conflicted, "conflict_ignored", fdate, source_file)

    upsert_stats = {"inserted": 0, "updated": 0}
    clear_stats = {"flagged": 0, "absent_or_already_cleared": 0}
    unreconciled = 0
    segment_counts = {s: 0 for s in SEGMENTS}

    cleared_stats = {"flagged": 0, "inserted": 0}

    if is_react:
        # Same as FRI compliance: a reconciled reactivated number is written to
        # the master as a cleared row so it shows up in the day's report.
        cleared_rows, unmatched = [], []
        cleared_history: list[tuple] = []
        for mobile, record in feed.items():
            clear_date = _to_date(_pick(record, hmap, "date_of_reactivation"), fdate)
            react_id = _txt(_pick(record, hmap, "reactivation_id"), 50)
            hsbc = matched.get(mobile)
            if hsbc is None:
                unmatched.append((mobile, clear_date, react_id))
                continue
            gho = _txt(hsbc.get(gho_key), 20)
            cleared_history.append((
                MODULE_MNRL, mobile, _txt(hsbc.get(cust_id_key), 100),
                segment_for(gho), "reactivated", clear_date,
                react_id or "reactivated", source_file,
            ))
            cleared_rows.append((
                mobile,
                _txt(hsbc.get(cust_id_key), 100),
                gho,
                segment_for(gho),
                *_customer_values(hsbc),
                clear_date, clear_date, clear_date,
                react_id,
            ))
        cleared_stats = _upsert_cleared(MODULE_MNRL, _dedupe_rows(cleared_rows),
                                        fdate, source_file)
        _record_history(MODULE_MNRL, cleared_history)
        clear_stats = _clear(MODULE_MNRL, unmatched, fdate, source_file)
    else:
        reconciled_rows = []
        reported_history: list[tuple] = []
        for mobile, record in feed.items():
            hsbc = matched.get(mobile)
            if hsbc is None:
                unreconciled += 1
                continue
            gho = _txt(hsbc.get(gho_key), 20)
            seg = segment_for(gho)
            segment_counts[seg] += 1
            reported_history.append((
                MODULE_MNRL, mobile, _txt(hsbc.get(cust_id_key), 100), seg,
                "reported", fdate,
                _txt(_pick(record, hmap, "disconnectionreason_id",
                           "disconnection_reason_id"), 20),
                source_file,
            ))
            reconciled_rows.append((
                mobile,
                _txt(hsbc.get(cust_id_key), 100),
                gho,
                seg,
                *_customer_values(hsbc),
                fdate, fdate,
                _to_date(_pick(record, hmap, "date_of_disconnection")),
                _txt(_pick(record, hmap, "disconnectionreason_id",
                           "disconnection_reason_id"), 20),
                _txt(_pick(record, hmap, "lsa_id"), 20),
                _txt(_pick(record, hmap, "tsp_id"), 20),
                _payload(record),
            ))
        upsert_stats = _upsert(MODULE_MNRL, _dedupe_rows(reconciled_rows))
        _record_history(MODULE_MNRL, reported_history)

    retract_stats = _retract(MODULE_MNRL, sorted(conflicted), fdate, source_file)
    purged = purge_expired()

    return {
        "module": MODULE_MNRL,
        "data_type": "Reactivated" if is_react else "Normal",
        "feed_date": str(fdate),
        "rows_in_file": len(rows),
        "invalid_mobile": invalid,
        "conflicts_ignored": len(conflicted),
        # Two clear paths, and BOTH must be counted: reconciled numbers are
        # inserted/updated as cleared rows by _upsert_cleared, unreconciled ones
        # can only update a row that already exists. Reporting just the second
        # made a run that cleared 14 reconciled numbers log "cleared=0".
        "cleared": cleared_stats["flagged"] + clear_stats["flagged"],
        "cleared_reconciled": cleared_stats["flagged"],
        "cleared_new": cleared_stats["inserted"],
        "clear_unmatched": clear_stats["absent_or_already_cleared"],
        "unreconciled": unreconciled,
        "master_inserted": upsert_stats["inserted"],
        "master_updated": upsert_stats["updated"],
        "by_segment": segment_counts,
        "retracted_deleted": retract_stats["deleted"],
        "retracted_uncleared": retract_stats["unclear"],
        "purged": purged,
        "retention": retention_window_check(fdate),
    }


# ── FILTERED READS ───────────────────────────────────────────────────────────
def _filter_clause(feed_date: Any = None, segment: str | None = None,
                   include_cleared: bool = False,
                   record_status: str = "active",
                   date_to: Any = None) -> tuple[str, list]:
    """Shared WHERE for download / ATR / count.

    record_status
        'active'   only numbers still requiring action
        'cleared'  only numbers that turned compliant / were reactivated
        'all'      both

    Dates. `feed_date` alone matches a single day; `feed_date` + `date_to`
    matches an inclusive range, which is what a fortnightly cycle needs — one
    selection instead of fifteen. Date matching follows the status: an active
    row belongs to the day it was REPORTED, a cleared row to the day it
    CLEARED, and 'all' matches either.
    """
    clauses: list[str] = []
    params: list[Any] = []

    if not include_cleared and record_status == "active":
        clauses.append("status = 'active'")
    elif record_status == "cleared":
        clauses.append("status = 'cleared'")

    dfrom = _to_date(feed_date)
    dto = _to_date(date_to) or dfrom

    if dfrom:
        if record_status == "active":
            clauses.append("last_reported_on BETWEEN %s AND %s")
            params.extend([dfrom, dto])
        elif record_status == "cleared":
            clauses.append("cleared_on BETWEEN %s AND %s")
            params.extend([dfrom, dto])
        else:
            clauses.append(
                "(last_reported_on BETWEEN %s AND %s OR cleared_on BETWEEN %s AND %s)"
            )
            params.extend([dfrom, dto, dfrom, dto])

    if segment and segment.upper() not in ("ALL", ""):
        clauses.append("segment = %s")
        params.append(segment)

    return (" AND ".join(clauses) or "TRUE"), params


def count_active(module: str, feed_date: Any = None, segment: str | None = None,
                 record_status: str = "active", date_to: Any = None) -> int:
    table = _TABLES[module.upper()]
    where, params = _filter_clause(feed_date, segment, record_status=record_status,
                                   date_to=date_to)
    with _cursor() as cur:
        cur.execute(f"SELECT COUNT(*) FROM {table} WHERE {where}", params)
        return int(cur.fetchone()[0])


def iter_active(module: str, feed_date: Any = None, segment: str | None = None,
                columns: list[str] | None = None,
                record_status: str = "active", date_to: Any = None) -> Iterator[dict]:
    """Stream master rows for export.

    Uses a NAMED (server-side) cursor: Postgres holds the result set and ships
    it in blocks of STREAM_SIZE. A 10 lakh row download therefore costs one
    block of memory, not a 10 lakh element list. Caller must consume or close
    the generator.
    """
    module = module.upper()
    table = _TABLES[module]
    cols = columns or EXPORT_COLUMNS[module]
    where, params = _filter_clause(feed_date, segment, record_status=record_status,
                                   date_to=date_to)

    conn = _get_pool().getconn()
    name = f"stream_{module.lower()}_{os.getpid()}_{id(params)}"
    try:
        cur = conn.cursor(name=name, cursor_factory=psycopg2.extras.RealDictCursor)
        cur.itersize = STREAM_SIZE
        cur.execute(
            f"SELECT {', '.join(cols)} FROM {table} WHERE {where} "
            f"ORDER BY GREATEST(last_reported_on, COALESCE(cleared_on, last_reported_on)) DESC, id",
            params,
        )
        for row in cur:
            yield {k: ("" if v is None else str(v)) for k, v in dict(row).items()}
        cur.close()
        conn.commit()
    except Exception:
        try:
            conn.rollback()
        except Exception:
            pass
        raise
    finally:
        try:
            _get_pool().putconn(conn)
        except Exception:
            pass


def segment_counts(module: str, feed_date: Any = None) -> dict:
    """Live counts per classification, for the segmented control.

    There are only ever two classifications — IWPB and Non-IWPB (CIB) — so this
    replaces the old GHO-code dropdown outright: the control shows every
    possible value along with what it would return.

    `unclassified` counts rows whose mobilecleanup GHO code was blank. They are
    included in 'both' but belong to neither classification, so a
    classification-scoped ATR run will never pick them up.
    """
    module = module.upper()
    table = _TABLES[module]
    where, params = _filter_clause(feed_date)

    with _cursor(dict_cursor=True) as cur:
        cur.execute(
            f"""
            SELECT segment,
                   COUNT(*)                                        AS total,
                   COUNT(*) FILTER (WHERE atr_status = 'pending')  AS atr_pending,
                   COUNT(*) FILTER (WHERE gho_code IS NULL
                                      OR gho_code = '')            AS no_gho_code
            FROM {table} WHERE {where}
            GROUP BY segment
            """,
            params,
        )
        rows = {r["segment"]: dict(r) for r in cur.fetchall()}

    out = {
        "module": module,
        "feed_date": str(_to_date(feed_date)) if feed_date else None,
        "segments": [],
        "total": 0,
        "atr_pending": 0,
        "unclassified": 0,
    }

    for seg in SEGMENTS:
        row = rows.get(seg, {})
        entry = {
            "segment": seg,
            "label": SEGMENT_LABELS.get(seg, seg),
            "count": int(row.get("total", 0) or 0),
            "atr_pending": int(row.get("atr_pending", 0) or 0),
        }
        out["segments"].append(entry)
        out["total"] += entry["count"]
        out["atr_pending"] += entry["atr_pending"]

    out["unclassified"] = sum(int(r.get("no_gho_code", 0) or 0) for r in rows.values())
    return out


def list_gho_codes(module: str | None = None) -> list[dict]:
    """Raw GHO codes seen on reconciled rows. Diagnostics only.

    The UI no longer offers a GHO filter — with just IWPB and CIB in play, the
    classification control already covers every value. Kept so an operator can
    confirm what the source data actually contains.
    """
    out: list[dict] = []
    with _cursor(dict_cursor=True) as cur:
        if module:
            table = _TABLES[module.upper()]
            cur.execute(
                f"""SELECT gho_code, segment, COUNT(*) AS record_count
                    FROM {table} WHERE status = 'active' AND gho_code IS NOT NULL
                    GROUP BY gho_code, segment ORDER BY gho_code"""
            )
        else:
            cur.execute(
                f"""SELECT gho_code, COUNT(*) AS record_count
                    FROM {MOBILECLEANUP_TABLE} WHERE gho_code IS NOT NULL
                    GROUP BY gho_code ORDER BY gho_code"""
            )
        for row in cur.fetchall():
            d = dict(row)
            d.setdefault("segment", segment_for(d.get("gho_code")))
            out.append(d)
    return out


def list_feed_dates(module: str) -> list[dict]:
    """Dates available for download, newest first."""
    table = _TABLES[module.upper()]
    with _cursor(dict_cursor=True) as cur:
        cur.execute(
            f"""SELECT last_reported_on AS feed_date, COUNT(*) AS record_count
                FROM {table} WHERE status = 'active'
                GROUP BY last_reported_on ORDER BY last_reported_on DESC LIMIT 120"""
        )
        return [{"feed_date": str(r["feed_date"]), "record_count": r["record_count"]}
                for r in cur.fetchall()]


# ── ATR QUEUE ────────────────────────────────────────────────────────────────
#
# Duplicate protection lives here, in two layers.
#
#   1. STATE FINGERPRINT. atr_state_hash captures everything that determines the
#      report: classification, the feed status, and the action codes in force.
#      A row is eligible only when the stored hash differs from the hash computed
#      now. Clicking Send twice therefore sends once: the second run recomputes
#      the same hash for every row and finds nothing to do. A genuine status
#      change alters the hash and makes the number eligible again by itself.
#
#   2. ATOMIC CLAIM. Batches are taken with FOR UPDATE SKIP LOCKED and flipped to
#      'sending' in the same statement, so two concurrent runs — a scheduled
#      cycle and an impatient operator, say — can never claim the same row.
#
# Cleared numbers are excluded throughout by _filter_clause's status='active'.
# That is what stops a number that turned compliant mid-fortnight from being
# reported: clearing it removes it from the queue with no special case.

# The feed column whose value counts as "the status" for fingerprint purposes
_STATUS_COL = {MODULE_FRI: "fri_status", MODULE_MNRL: "disconnection_reason_id"}

# Rows claimed but never completed (process died mid-run) are released after this
ATR_STUCK_MINUTES = int(os.environ.get("ATR_STUCK_MINUTES", "30"))


def _hash_expr(module: str, prefix: str = "") -> str:
    """SQL computing a row's current state fingerprint.

    INCLUDES status and clear_reason. A number flagged on the 10th and compliant
    on the 14th keeps the same fri_status ('flagged_for_reverification') — only
    `status` flips to 'cleared'. Fingerprinting the feed status alone therefore
    made compliance invisible and the follow-up report was never sent. That is
    the whole point of the second report, so both columns are in the hash.

    Still EXCLUDES dates and report_count: a number re-reported with an
    unchanged status is not a new state and must not trigger a second report.
    """
    status_col = _STATUS_COL[module]
    p = f"{prefix}." if prefix else ""
    return (
        f"md5(concat_ws('|', {p}segment, {p}status, coalesce({p}clear_reason, ''), "
        f"coalesce({p}{status_col}, ''), "
        "%s::text, %s::text))"
    )


def state_snapshot(row: dict, module: str, action_type: str, action_taken: str) -> dict:
    """The human-readable form of what the fingerprint covers."""
    return {
        "segment": row.get("segment"),
        "record_status": row.get("status"),
        "clear_reason": row.get("clear_reason"),
        "feed_status": row.get(_STATUS_COL[module]),
        "action_type": action_type,
        "action_taken": str(action_taken),
    }


def release_stuck_claims(module: str, minutes: int = ATR_STUCK_MINUTES) -> int:
    """Return rows abandoned mid-send to the queue. Run before each cycle."""
    table = _TABLES[module.upper()]
    with _cursor() as cur:
        cur.execute(
            f"""UPDATE {table}
                SET atr_status = 'pending', atr_claimed_at = NULL, updated_at = NOW()
                WHERE atr_status = 'sending'
                  AND atr_claimed_at < NOW() - (%s || ' minutes')::interval""",
            (minutes,),
        )
        return cur.rowcount


def count_atr_eligible(module: str, feed_date: Any = None, segment: str | None = None,
                       action_type: str = "", action_taken: str = "",
                       case_type: str = "open", date_to: Any = None) -> int:
    """How many numbers would actually be reported right now.

    Counts rows whose state differs from the one last reported — not rows
    marked 'pending', which is why pressing Send twice shows zero the second
    time instead of re-sending.
    """
    module = module.upper()
    table = _TABLES[module]
    # 'open' cases are the active rows; 'cleared' cases are the compliant /
    # reactivated ones, which need their own report with a different action code.
    record_status = "cleared" if case_type == "cleared" else "active"
    where, params = _filter_clause(feed_date, segment, record_status=record_status,
                                   date_to=date_to)
    with _cursor() as cur:
        cur.execute(
            f"""SELECT COUNT(*) FROM {table}
                WHERE {where}
                  AND atr_status <> 'sending'
                  AND atr_state_hash IS DISTINCT FROM {_hash_expr(module)}""",
            params + [action_type, action_taken],
        )
        return int(cur.fetchone()[0])


def count_atr_pending(module: str, feed_date: Any = None, segment: str | None = None,
                      case_type: str = "open", date_to: Any = None) -> int:
    """Rows never successfully reported. Used for dashboard display only."""
    table = _TABLES[module.upper()]
    record_status = "cleared" if case_type == "cleared" else "active"
    where, params = _filter_clause(feed_date, segment, record_status=record_status,
                                   date_to=date_to)
    with _cursor() as cur:
        cur.execute(
            f"""SELECT COUNT(*) FROM {table}
                WHERE {where} AND atr_status IN ('pending', 'rejected')""",
            params,
        )
        return int(cur.fetchone()[0])


def count_cleared_in_window(module: str, feed_date: Any = None,
                            segment: str | None = None) -> int:
    """Numbers that cleared before ever being reported.

    Surfaced so a cycle can say "42 numbers turned compliant before this run and
    were left out" rather than the operator wondering where they went.
    """
    module = module.upper()
    table = _TABLES[module]
    where, params = _filter_clause(feed_date, segment, include_cleared=True)
    with _cursor() as cur:
        cur.execute(
            f"""SELECT COUNT(*) FROM {table}
                WHERE {where} AND status = 'cleared' AND atr_send_count = 0""",
            params,
        )
        return int(cur.fetchone()[0])


def claim_atr_batch(module: str, run_id: str, action_type: str, action_taken: str,
                    feed_date: Any = None, segment: str | None = None,
                    batch_size: int = 1000, case_type: str = "open",
                    date_to: Any = None) -> list[dict]:
    """Atomically take the next batch of eligible rows and mark them 'sending'.

    SKIP LOCKED means a concurrent run steps over anything this one holds rather
    than blocking or double-claiming. Returns [] when the queue is exhausted,
    which is the loop's termination condition.
    """
    module = module.upper()
    table = _TABLES[module]
    record_status = "cleared" if case_type == "cleared" else "active"
    where, params = _filter_clause(feed_date, segment, record_status=record_status,
                                   date_to=date_to)
    hash_expr = _hash_expr(module)
    status_col = _STATUS_COL[module]

    select_cols = ", ".join(
        ["id", "mobile_no", "cust_id", "segment", "gho_code", status_col,
         "status", "clear_reason", "cleared_on", "last_reported_on",
         "atr_state_hash", "atr_send_count", "atr_status", "atr_action_type",
         "atr_action_taken"]
    )

    with _cursor(dict_cursor=True) as cur:
        cur.execute(
            f"""
            WITH claimed AS (
                SELECT id FROM {table}
                WHERE {where}
                  AND atr_status <> 'sending'
                  AND atr_state_hash IS DISTINCT FROM {hash_expr}
                  -- Never re-claim a row this run has already attempted.
                  --
                  -- A rejected row deliberately keeps a NULL hash so the NEXT
                  -- cycle retries it. Without this predicate the CURRENT run
                  -- immediately re-claims it: every batch that fails comes
                  -- straight back, and the loop only ends when the process is
                  -- killed. That is how 112 numbers became 13,320 attempts.
                  AND atr_run_id IS DISTINCT FROM %s::uuid
                ORDER BY id
                LIMIT %s
                FOR UPDATE SKIP LOCKED
            )
            UPDATE {table} AS m
            SET atr_status = 'sending', atr_claimed_at = NOW(), atr_run_id = %s
            FROM claimed c
            WHERE m.id = c.id
            RETURNING {", ".join("m." + c for c in select_cols.split(", "))},
                      {_hash_expr(module, "m")} AS new_hash
            """,
            params + [action_type, action_taken, run_id, batch_size, run_id,
                      action_type, action_taken],
        )
        return [dict(r) for r in cur.fetchall()]


def complete_atr_batch(module: str, run_id: str, action_type: str, action_taken: str,
                       sent: dict[int, str] | None = None,
                       failed: dict[int, tuple[str, str]] | None = None) -> None:
    """Write the outcome back and, for successes, store the new fingerprint.

    Storing the hash is what makes the send idempotent: until the number's state
    changes again it will not be selected by any future run.

    sent   {row_id: state_hash}
    failed {row_id: (status, error)} where status is 'rejected' or 'skipped'
    """
    table = _TABLES[module.upper()]
    with _cursor() as cur:
        if sent:
            cur.execute(
                f"""UPDATE {table} AS m
                    SET atr_status = 'sent',
                        atr_state_hash = v.hash,
                        atr_send_count = m.atr_send_count + 1,
                        atr_last_sent_at = NOW(),
                        atr_submitted_at = NOW(),
                        atr_claimed_at = NULL,
                        atr_run_id = %s,
                        atr_action_type = %s,
                        atr_action_taken = %s,
                        atr_error = NULL,
                        updated_at = NOW()
                    FROM (SELECT unnest(%s::bigint[]) AS row_id,
                                 unnest(%s::text[])   AS hash) AS v
                    WHERE m.id = v.row_id""",
                (run_id, action_type, action_taken,
                 list(sent.keys()), list(sent.values())),
            )
        if failed:
            # Deliberately does NOT store the hash: a rejected number stays
            # eligible so the next cycle retries it.
            cur.execute(
                f"""UPDATE {table} AS m
                    SET atr_status = v.state,
                        atr_claimed_at = NULL,
                        atr_run_id = %s,
                        atr_action_type = %s,
                        atr_action_taken = %s,
                        atr_error = v.err,
                        updated_at = NOW()
                    FROM (SELECT unnest(%s::bigint[]) AS row_id,
                                 unnest(%s::text[])   AS state,
                                 unnest(%s::text[])   AS err) AS v
                    WHERE m.id = v.row_id""",
                (run_id, action_type, action_taken,
                 list(failed.keys()),
                 [v[0] for v in failed.values()],
                 [v[1] for v in failed.values()]),
            )


def classify_trigger(row: dict, action_type: str, action_taken: str) -> str:
    """Why this number is being reported now — recorded on every event."""
    if not row.get("atr_state_hash"):
        return "first_send"
    if row.get("atr_status") == "rejected":
        return "retry_after_reject"
    prev = (row.get("atr_action_type"), str(row.get("atr_action_taken") or ""))
    if prev != (action_type, str(action_taken)):
        return "action_config_change"
    return "status_change"


def reset_atr_status(module: str, feed_date: Any = None, segment: str | None = None,
                     only_failed: bool = True, date_to: Any = None,
                     record_status: str = "all") -> int:
    """Put numbers back in the queue so failures can be retried.

    Clears the fingerprint too — otherwise the row would be marked pending but
    still filtered out as unchanged.
    """
    table = _TABLES[module.upper()]
    where, params = _filter_clause(feed_date, segment, record_status=record_status,
                                   date_to=date_to)
    extra = "AND atr_status IN ('rejected', 'skipped')" if only_failed else ""
    with _cursor() as cur:
        cur.execute(
            f"""UPDATE {table}
                SET atr_status = 'pending', atr_state_hash = NULL,
                    atr_error = NULL, atr_claimed_at = NULL, updated_at = NOW()
                WHERE {where} {extra}""",
            params,
        )
        return cur.rowcount

def master_stats(module: str) -> dict:
    module = module.upper()
    table = _TABLES[module]
    with _cursor(dict_cursor=True) as cur:
        cur.execute(
            f"""SELECT COUNT(*)                                          AS total,
                       COUNT(*) FILTER (WHERE status = 'active')         AS active,
                       COUNT(*) FILTER (WHERE status = 'cleared')        AS cleared,
                       COUNT(*) FILTER (WHERE segment = 'IWPB'
                                          AND status = 'active')         AS iwpb,
                       COUNT(*) FILTER (WHERE segment = 'Non-IWPB'
                                          AND status = 'active')         AS non_iwpb,
                       COUNT(*) FILTER (WHERE status = 'active'
                                          AND (gho_code IS NULL
                                               OR gho_code = ''))        AS missing_gho,
                       COUNT(*) FILTER (WHERE status = 'active'
                                          AND atr_status = 'pending')    AS atr_pending,
                       COUNT(*) FILTER (WHERE atr_status = 'sent')       AS atr_sent,
                       COUNT(*) FILTER (WHERE atr_status = 'rejected')   AS atr_rejected,
                       COUNT(DISTINCT cust_id)                           AS customers,
                       MIN(first_reported_on)                            AS oldest_report,
                       MAX(last_reported_on)                             AS latest_report
                FROM {table}"""
        )
        row = dict(cur.fetchone() or {})
        cur.execute(
            f"""SELECT COUNT(*) AS conflicts_last_7d FROM {EVENTS_TABLE}
                WHERE module = %s AND event = 'conflict_ignored'
                  AND event_date >= CURRENT_DATE - 7""",
            (module,),
        )
        row.update(dict(cur.fetchone() or {}))

    for key in ("oldest_report", "latest_report"):
        row[key] = str(row[key]) if row.get(key) else None
    row["module"] = module
    row["retention_days"] = RETENTION_DAYS
    return row


def conflicts(module: str, feed_date: Any = None, limit: int = 500) -> list[dict]:
    module = module.upper()
    params: list[Any] = [module]
    where = "module = %s AND event = 'conflict_ignored'"
    fdate = _to_date(feed_date)
    if fdate:
        where += " AND event_date = %s"
        params.append(fdate)
    params.append(limit)

    with _cursor(dict_cursor=True) as cur:
        cur.execute(
            f"""SELECT module, mobile_no, event_date, source_file, created_at
                FROM {EVENTS_TABLE} WHERE {where}
                ORDER BY event_date DESC, created_at DESC LIMIT %s""",
            params,
        )
        rows = cur.fetchall()

    out = []
    for row in rows:
        d = dict(row)
        d["event_date"] = str(d["event_date"])
        d["created_at"] = d["created_at"].isoformat()
        out.append(d)
    return out


def lookup(mobile_no: str) -> dict:
    """Everything both masters know about one number."""
    mobile = _norm_mobile(mobile_no)
    if not mobile:
        return {"records": [], "events": []}

    records = []
    with _cursor(dict_cursor=True) as cur:
        for module, table in _TABLES.items():
            cur.execute(
                f"""SELECT status, clear_reason, cleared_on, cust_id, gho_code, segment,
                           first_reported_on, last_reported_on, report_count,
                           atr_status, atr_action_taken, atr_submitted_at, atr_error
                    FROM {table} WHERE mobile_no = %s""",
                (mobile,),
            )
            row = cur.fetchone()
            if row:
                d = dict(row)
                d["module"] = module
                records.append(d)

        cur.execute(
            f"""SELECT module, event, event_date, source_file, created_at
                FROM {EVENTS_TABLE} WHERE mobile_no = %s
                ORDER BY created_at DESC LIMIT 50""",
            (mobile,),
        )
        events = [dict(r) for r in cur.fetchall()]

    return {"mobile_no": mobile, "records": records, "events": events}


def atr_preview_counts(module: str, feed_date: Any = None, segment: str | None = None,
                       date_to: Any = None,
                       defaults: dict[tuple[str, str], dict] | None = None) -> list[dict]:
    """Eligible / never-sent counts for every (case_type, segment), in ONE query.

    The Reports page previously issued a dozen separate queries to build this —
    two per combination plus the defaults lookups — which is both slow and, on a
    threaded server, a good way to exhaust the connection pool. A single grouped
    scan does the same work on one connection.

    `defaults` maps (case_type, segment) -> the action codes in force, which the
    fingerprint depends on. Combinations with no configured defaults are still
    returned, with zero counts.
    """
    module = module.upper()
    table = _TABLES[module]
    status_col = _STATUS_COL[module]
    defaults = defaults or {}

    segs = [segment] if segment and segment.upper() != "ALL" else [SEGMENT_IWPB, SEGMENT_NON_IWPB]

    # The action codes vary per (case_type, segment), so they arrive as a small
    # VALUES list joined to the master rather than as scalar parameters.
    combos, params = [], []
    for case_type in ("open", "cleared"):
        for seg in segs:
            d = defaults.get((case_type, seg)) or {}
            combos.append("(%s, %s, %s, %s)")
            params.extend([case_type, seg,
                           str(d.get("action_type") or ""),
                           str(d.get("action_taken") or "")])

    dfrom = _to_date(feed_date)
    dto = _to_date(date_to) or dfrom
    date_params = [dfrom, dto, dfrom, dto] if dfrom else []
    date_filter = (
        """AND ((c.case_type = 'open'    AND m.last_reported_on BETWEEN %s AND %s)
             OR (c.case_type = 'cleared' AND m.cleared_on        BETWEEN %s AND %s))"""
        if dfrom else ""
    )

    sql = f"""
        WITH combos(case_type, segment, action_type, action_taken) AS (VALUES {", ".join(combos)})
        SELECT c.case_type, c.segment,
               COUNT(m.id) FILTER (
                   WHERE m.atr_status <> 'sending'
                     AND m.atr_state_hash IS DISTINCT FROM
                         md5(concat_ws('|', m.segment, m.status, coalesce(m.clear_reason, ''),
                                       coalesce(m.{status_col}, ''),
                                       c.action_type, c.action_taken))
               ) AS eligible,
               COUNT(m.id) FILTER (WHERE m.atr_status IN ('pending', 'rejected')) AS never_sent,
               COUNT(m.id) AS in_window
        FROM combos c
        LEFT JOIN {table} m
          ON m.segment = c.segment
         AND m.status = CASE WHEN c.case_type = 'cleared' THEN 'cleared' ELSE 'active' END
         {date_filter}
        GROUP BY c.case_type, c.segment
        ORDER BY c.case_type DESC, c.segment
    """

    with _cursor(dict_cursor=True) as cur:
        cur.execute(sql, params + date_params)
        return [dict(r) for r in cur.fetchall()]
