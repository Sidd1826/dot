const POLL = {
  FAST_POLL_MS:   3_000,
  SLOW_POLL_MS:  20_000,
  MS_PER_RECORD:    1.2,
  FAST_THRESHOLD: 30_000,
};

function slowPhaseDuration(count) {
  if (!count || count <= POLL.FAST_THRESHOLD) return 0;
  return Math.round(count * POLL.MS_PER_RECORD);
}

    .join("&");

// JOB POLLER
// ─────────────────────────────────────────────
function useJobPoller({ jobId, recordCount = 0, onDone, onFailed }) {
  const timerRef    = useRef(null);
  const startedAt   = useRef(null);
  const isFast      = useRef(false);
  const slowPhaseMs = useRef(0);

  useEffect(() => {
    if (!jobId) return;

    startedAt.current   = Date.now();
    isFast.current      = false;
    slowPhaseMs.current = slowPhaseDuration(recordCount);

    const stop = () => { if (timerRef.current) clearTimeout(timerRef.current); };

    const scheduleNext = () => {
      const elapsed = Date.now() - startedAt.current;
      const goFast  = slowPhaseMs.current === 0 || elapsed >= slowPhaseMs.current;
      if (goFast) isFast.current = true;
      timerRef.current = setTimeout(tick, goFast ? POLL.FAST_POLL_MS : POLL.SLOW_POLL_MS);
    };

    const tick = async () => {
      try {
        const data = await api.pollJobStatus(jobId);
        if      (data.status === "done")   { stop(); onDone(data.result); }
        else if (data.status === "failed") { stop(); onFailed(data.error || "The job failed."); }
        else                               { scheduleNext(); }
      } catch (e) { stop(); onFailed(e.message); }
    };

    tick();
    return stop;
  }, [jobId]); // eslint-disable-line
}

const fmt = (n) => (n != null ? Number(n).toLocaleString("en-IN") : "—");
// Every timestamp is rendered in IST explicitly. The backend sends UTC with an
// offset, so without a timeZone the browser's own zone decides — a machine set
// to GMT showed times five and a half hours out.
const IST = "Asia/Kolkata";
const fmtDate = (d) =>
  d ? new Date(d).toLocaleDateString("en-IN",
    { day: "numeric", month: "short", year: "numeric", timeZone: IST }) : "—";
const fmtDateTime = (d) =>
  d ? new Date(d).toLocaleString("en-IN", {
    day: "numeric", month: "short", year: "numeric",
    hour: "2-digit", minute: "2-digit", timeZone: IST,
  }) : "—";
// A plain YYYY-MM-DD data date. No parsing through Date(): these are calendar
// dates, not instants, and putting them through a timezone can shift the day.
const fmtDataDate = (d) => {
  if (!d) return "—";
  const [y, m, day] = String(d).split("T")[0].split("-");
  if (!y || !m || !day) return String(d);
  const months = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
  return `${Number(day)} ${months[Number(m) - 1]} ${y}`;
};
const today = () => new Date().toISOString().split("T")[0];
const triggerBlobDownload = (url, filename) => {
  const a = document.createElement("a");
  a.href = url; a.download = filename; a.click();
};

const Paginated = ({ items, renderRow, columns, emptyTitle, emptyHint, loading, accent = "blue" }) => {
  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE);
  useEffect(() => { setVisibleCount(PAGE_SIZE); }, [items]);

  if (loading) return (
    <div className="flex flex-col items-center py-14 gap-3">
      <Loader2 size={26} className="animate-spin text-slate-300" />
      <p className="text-sm text-slate-400">Loading…</p>
    </div>
  );
  if (!items.length) return (
    <div className="flex flex-col items-center py-14 gap-3">
      <div className="w-12 h-12 rounded-xl bg-slate-100 flex items-center justify-center">
        <Cloud size={22} className="text-slate-400" />
      </div>
      <p className="text-slate-600 font-medium text-sm">{emptyTitle}</p>
      {emptyHint && <p className="text-slate-400 text-xs">{emptyHint}</p>}
    </div>
  );

  const visible = items.slice(0, visibleCount);
  const hasMore = visibleCount < items.length;

  return (
    <>
      <div className="overflow-x-auto rounded-xl border border-slate-200">
        <table className="w-full text-sm">
          <thead>
            <tr className="bg-slate-50 border-b border-slate-200">
              {columns.filter(Boolean).map((h) => (
                <th key={h} className="px-4 py-3 text-left text-xs font-semibold text-slate-500 uppercase tracking-wider whitespace-nowrap">
                  {h}
                </th>
              ))}
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">{visible.map(renderRow)}</tbody>
        </table>
      </div>
      <div className="flex items-center justify-between mt-3 px-1">
        <p className="text-xs text-slate-400">
          Showing <span className="font-semibold text-slate-600">{visible.length}</span> of{" "}
          <span className="font-semibold text-slate-600">{items.length}</span>
        </p>
        {hasMore && (
          <button onClick={() => setVisibleCount((c) => c + PAGE_SIZE)}
            className={`flex items-center gap-1.5 px-4 py-2 text-xs font-semibold text-${accent}-600
              bg-${accent}-50 hover:bg-${accent}-100 border border-${accent}-200 rounded-lg transition-all`}>
            Load more
            <span className={`ml-0.5 px-1.5 py-0.5 bg-${accent}-200 text-${accent}-700 rounded-full text-xs`}>
              {items.length - visibleCount} left
            </span>
          </button>
        )}
      </div>
    </>
  );
};

    try { const r = await api.fetchFRICount(d); setCount(r.count); } catch { /* silent */ }
  };

  const handleDownload = async () => {
    setStarting(true); setAlert(null);
    try {
      const r = await api.fetchFRICount(date);
      setCount(r.count);
      if (r.count === 0) { showAlert("warning", "Nothing published for that date."); return; }
      setRC(r.count);
      const job = await api.startFRIExport(date);
      setJobId(job.job_id);
    } catch (e) { showAlert("error", e.message); }
    finally { setStarting(false); }
  };

  useJobPoller({
    jobId, recordCount,
    onDone: async (result) => {
      setJobId(null); await loadFiles();
      showAlert("success", `Downloaded ${fmt(result.record_count)} records. Reconcile them next.`);
    },
    onFailed: (err) => { setJobId(null); showAlert("error", err); },
  });

  useEffect(() => { loadFiles(); }, [loadFiles]);
  useEffect(() => { handleDateChange(date); }, []); // eslint-disable-line

  return (
    <div className="space-y-6">
      <SectionCard icon={Database} iconBg="bg-blue-600" title="FRI data"
        subtitle="Pull a day's Financial Fraud Risk Indicator list and store it">
        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-5">
          <DatePicker value={date} onChange={handleDateChange} label="Data date" />
          <CountBadge label="Available" count={count} color="blue" />
          <div className="rounded-xl border border-slate-200 p-4 flex flex-col gap-2 justify-center bg-slate-50">
            <div className="flex items-center gap-2 text-slate-400 text-xs font-medium"><Cloud size={14} /> Stored in Azure Blob</div>
            <div className="flex items-center gap-2 text-slate-400 text-xs font-medium"><HardDrive size={14} /> Indexed in PostgreSQL</div>
            <div className="flex items-center gap-2 text-slate-400 text-xs font-medium"><Shield size={14} /> Time-limited download links</div>
          </div>
        </div>
        <PrimaryBtn onClick={handleDownload} loading={starting}
          disabled={count === 0 || starting || !!jobId} label="Download FRI data" />
        {jobId && <RunningBanner title="Downloading and saving…" subtitle="Running in the background" />}
        {alert && <Alert {...alert} onClose={() => setAlert(null)} />}
      </SectionCard>

      <SectionCard icon={BarChart3} iconBg="bg-blue-500" title="FRI archive"
        subtitle={`${files.length} file${files.length === 1 ? "" : "s"}`}
        action={
          <button onClick={loadFiles} disabled={loadingFiles}
            className="flex items-center gap-1.5 text-xs text-blue-600 hover:text-blue-800 font-medium px-3 py-1.5 hover:bg-blue-50 rounded-lg transition-colors">
            <RefreshCw size={13} className={loadingFiles ? "animate-spin" : ""} /> Refresh
          </button>
        }>
        <FilesTable files={files} loading={loadingFiles}
          onDownload={async (f) => {
            try {
              const { url, file_name } = await api.getDownloadUrl(f.id, "FRI");
              triggerBlobDownload(url, file_name);
            } catch (e) { showAlert("error", `Download failed: ${e.message}`); }
          }} />
      </SectionCard>
    </div>
  );
};

    const count   = dataType === "Normal" ? normalCount : reactCount;
    const setRc   = dataType === "Normal" ? setRcN : setRcR;
    const setJob  = dataType === "Normal" ? setJN : setJR;

    setBusy(true); setAlert(null);
    try {
      if (count === 0) { showAlert("warning", `Nothing published for ${dataType.toLowerCase()} on that date.`); return; }

  return (
    <div className="space-y-6">
      <SectionCard icon={Database} iconBg="bg-teal-600" title="MNRL data"
        subtitle="Pull a day's revoked and reactivated number lists">
        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-5">
          <DatePicker value={date} onChange={(d) => { setDate(d); setNC(null); setRC(null); fetchCounts(d); }} label="Data date" />
          <CountBadge label="Normal" count={normalCount} color="teal" />
          <CountBadge label="Reactivated" count={reactCount} color="violet" />
        </div>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
          <div>
            <PrimaryBtn onClick={() => start("Normal")} loading={startingN}
              disabled={normalCount === 0 || busy}
              gradient="linear-gradient(135deg, #0d9488, #0891b2)" label="Download normal" />
            {jobNormal && <RunningBanner title="Downloading normal list…" subtitle="Running in the background" />}
          </div>
          <div>
            <PrimaryBtn onClick={() => start("Reactivated")} loading={startingR}
              disabled={reactCount === 0 || busy}

        </div>
        {alert && <Alert {...alert} onClose={() => setAlert(null)} />}
      </SectionCard>

      <SectionCard icon={BarChart3} iconBg="bg-teal-500" title="MNRL archive"
        subtitle={`${files.length} file${files.length === 1 ? "" : "s"}`}
        action={
          <button onClick={loadFiles} disabled={loadingFiles}
            className="flex items-center gap-1.5 text-xs text-teal-600 hover:text-teal-800 font-medium px-3 py-1.5 hover:bg-teal-50 rounded-lg transition-colors">
            <RefreshCw size={13} className={loadingFiles ? "animate-spin" : ""} /> Refresh
          </button>
        }>
        <FilesTable files={files} loading={loadingFiles} showDataType
          onDownload={async (f) => {
            try {
              const { url, file_name } = await api.getDownloadUrl(f.id, "MNRL");
              triggerBlobDownload(url, file_name);
            } catch (e) { showAlert("error", `Download failed: ${e.message}`); }
          }} />
      </SectionCard>

// Classification options carry their own counts, so there is nothing left for a

const classificationOptions = (segCounts) => {

  return [
    { value: "ALL", label: "Both", count: segCounts?.total },
    { value: "IWPB", label: "IWPB", count: by["IWPB"]?.count },

const UnclassifiedNote = ({ count, module }) => {
  if (!count) return null;
  return (
    <div className="md:col-span-full flex items-start gap-2 text-xs text-amber-700 bg-amber-50 border border-amber-200 rounded-xl px-4 py-3">
      <AlertCircle size={14} className="text-amber-500 shrink-0 mt-0.5" />
      <span>
        {fmt(count)} {module} record{count === 1 ? " has" : "s have"} no GHO code in
        the customer data, so {count === 1 ? "it sits" : "they sit"} outside both
        classifications and won't be picked up by a report run. They're still
        included when you pick "Both".
      </span>
    </div>
  );
};

                         dates, segCounts, loading, disabled }) => {
  const dateOptions = [
    { value: "ALL", label: "All dates in the window" },
    ...dates.map((d) => ({ value: d.feed_date, label: `${d.feed_date} · ${fmt(d.record_count)}` })),
  ];

  return (
    <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
      <SegmentedControl label="Source" value={module} onChange={setModule} disabled={disabled}
        options={[{ value: "FRI", label: "FRI" }, { value: "MNRL", label: "MNRL" }]} />
      <SearchableSelect label="Data date" value={date} onChange={setDate}

        emptyText="No reconciled data yet — reconcile a file above first" />

        disabled={disabled} options={classificationOptions(segCounts)} />

      <UnclassifiedNote count={segCounts?.unclassified} module={module} />

          <Info size={14} className="text-slate-400 shrink-0 mt-0.5" />

            file above and the dates will appear here.
          </span>
        </div>
      )}
    </div>
  );
};

const ReconcileModule = () => {
  // reconcile a file
  const [container, setContainer]   = useState("mnrl-exports");
  const [folders, setFolders]       = useState([]);
  const [srcDate, setSrcDate]       = useState("");
  const [srcFiles, setSrcFiles]     = useState([]);
  const [srcFile, setSrcFile]       = useState("");
  const [loadingFolders, setLFo]    = useState(false);
  const [loadingSrcFiles, setLFi]   = useState(false);
  const [startingRun, setSR]        = useState(false);
  const [runJob, setRunJob]         = useState(null);
  const [lastRun, setLastRun]       = useState(null);

  // download from master
  const [module, setModule]   = useState("MNRL");
  const [mDate, setMDate]     = useState("ALL");
  const [segment, setSegment] = useState("ALL");
  const [dates, setDates]     = useState([]);
  const [segCounts, setSegC]  = useState(null);
  const [loadingMeta, setLM]  = useState(false);
  const [preview, setPreview] = useState(null);
  const [dlJob, setDlJob]     = useState(null);
  const [startingDl, setSD]   = useState(false);
  const [lastFile, setLastFile] = useState(null);

  const [runs, setRuns]       = useState([]);
  const [loadingRuns, setLR]  = useState(true);
  const [alert, setAlert]     = useState(null);
  const [recStatus, setRecStatus] = useState("all");
  const [recFiles, setRecFiles] = useState([]);
  const [loadingRecFiles, setLRF] = useState(true);

  const showAlert = (type, message) => {
    setAlert({ type, message });
    if (type === "success" || type === "info") setTimeout(() => setAlert(null), 9000);
  };

  // Reconciled downloads are listed here, not in the FRI/MNRL tabs — those show
  // only what was pulled from the source APIs.
  const loadRecFiles = useCallback(async (mod) => {
    setLRF(true);
    try { setRecFiles(await api.listFiles(mod, "reconciled")); }
    catch (e) { showAlert("error", `Couldn't load reconciled files: ${e.message}`); }
    finally { setLRF(false); }
  }, []);

  const loadFolders = useCallback(async (c) => {
    setLFo(true); setFolders([]); setSrcDate(""); setSrcFiles([]); setSrcFile("");
    try {
      const list = (await api.fetchReconcileFolders(c)).folders || [];
      setFolders(list);
      if (list.length) setSrcDate(list[0]);
    } catch (e) { showAlert("error", `Couldn't list dates: ${e.message}`); }
    finally { setLFo(false); }
  }, []);

  const loadSrcFiles = useCallback(async (c, d) => {
    if (!d) return;
    setLFi(true); setSrcFiles([]); setSrcFile("");
    try {
      const list = (await api.fetchReconcileFiles(c, d)).files || [];
      setSrcFiles(list);
      if (list.length) setSrcFile(list[0]);
    } catch (e) { showAlert("error", `Couldn't list files: ${e.message}`); }
    finally { setLFi(false); }
  }, []);

  const loadMeta = useCallback(async (mod, dateFilter) => {
    setLM(true);
    try {
      const [d, s] = await Promise.all([
        api.fetchMasterDates(mod),
        api.fetchMasterSegments(mod, dateFilter === "ALL" ? "" : dateFilter),
      ]);
      setDates(d.dates || []);
      setSegC(s);
    } catch (e) { showAlert("error", `Couldn't load filters: ${e.message}`); }
    finally { setLM(false); }
  }, []);

  const loadRuns = useCallback(async () => {
    setLR(true);
    try { setRuns(await api.listReconcileHistory()); }
    catch (e) { showAlert("error", `Couldn't load history: ${e.message}`); }
    finally { setLR(false); }
  }, []);

  const refreshPreview = useCallback(async () => {
    try {
      setPreview(await api.fetchMasterCount({
        type: module, date: mDate === "ALL" ? "" : mDate, segment,
        record_status: recStatus,
      }));
    } catch { setPreview(null); }
  }, [module, mDate, segment, recStatus]);

  useEffect(() => { loadFolders(container); loadRuns(); }, []); // eslint-disable-line
  useEffect(() => { if (srcDate) loadSrcFiles(container, srcDate); }, [srcDate]); // eslint-disable-line
  useEffect(() => {
    setMDate("ALL"); setSegment("ALL");
    loadMeta(module, "ALL"); loadRecFiles(module);
  }, [module, loadMeta, loadRecFiles]);
  useEffect(() => { loadMeta(module, mDate); }, [mDate]); // eslint-disable-line
  useEffect(() => { refreshPreview(); }, [refreshPreview]);

  const handleReconcile = async () => {
    if (!srcFile) { showAlert("warning", "Pick a file to reconcile."); return; }
    setSR(true); setAlert(null); setLastRun(null);
    try {
      const job = await api.startReconcile(container, srcDate, srcFile);
      setRunJob(job.job_id);
    } catch (e) { showAlert("error", e.message); }
    finally { setSR(false); }
  };

  useJobPoller({
    jobId: runJob, recordCount: 0,
    onDone: async (result) => {
      setRunJob(null); setLastRun(result);
      await Promise.all([loadRuns(), loadMeta(module, mDate), refreshPreview()]);

      // A file where nothing matches is a valid outcome, not a failure.
      // A file that matches nothing, or one that was entirely compliant, is a
      // valid outcome rather than a failure. Say what happened, in blue.
      if (["no_matches", "all_compliant", "cleared_only"].includes(result.status)) {
        showAlert("info", result.message);
        return;
      }

      const m = result.master || {};
      const base =
        `Reconciled ${fmt(result.matched_count)} of ${fmt(result.input_row_count)} rows. ` +
        `${fmt(m.master_inserted)} new, ${fmt(m.master_updated)} updated in the master.`;

      // An export older than the retention window lands in the master but is
      // purge-eligible once the grace period ends. Say so now rather than
      // letting the rows disappear overnight.
      if (m.retention?.outside_window) {
        showAlert("warning",
          `${base} Note: this file is dated ${m.retention.age_days} days ago, past the ` +
          `${m.retention.retention_days}-day window, so these rows will be cleared in ` +
          `about ${m.retention.grace_hours} hours. Reconcile a recent export to keep them.`);
        return;
      }
      showAlert("success", base);
    },
    onFailed: (e) => { setRunJob(null); showAlert("error", e); },
  });

  const handleDownload = async () => {
    setSD(true); setAlert(null); setLastFile(null);
    try {
      const job = await api.startMasterDownload({
        type: module, date: mDate === "ALL" ? null : mDate, segment,
        record_status: recStatus,
      });
      setDlJob(job.job_id);
    } catch (e) { showAlert("error", e.message); }
    finally { setSD(false); }
  };

  useJobPoller({
    jobId: dlJob, recordCount: preview?.count || 0,
    onDone: (result) => {
      setDlJob(null);
      if (result.status === "empty" || !result.record_count) {
        setLastFile(null);
        showAlert("warning", result.message || "Nothing matched those filters.");
        return;
      }
      setLastFile(result);
      loadRecFiles(module);
      showAlert("success", `Built ${fmt(result.record_count)} rows across ${result.file_count} file(s).`);
    },
    onFailed: (e) => { setDlJob(null); showAlert("error", e); },
  });

  const reconciling = startingRun || !!runJob;
  const downloading = startingDl || !!dlJob;

  return (
    <div className="space-y-6">
      {/* ── Step 1: reconcile a file into the master ── */}
      <SectionCard icon={GitMerge} iconBg="bg-indigo-600" title="Reconcile a file"
        subtitle="Match a downloaded list against HSBC customer data and write it to the master">
        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-5">
          <SearchableSelect label="Source" value={container} disabled={reconciling}
            onChange={(c) => { setContainer(c); loadFolders(c); }}
            options={[
              { value: "fri-exports", label: "FRI downloads" },
              { value: "mnrl-exports", label: "MNRL downloads" },
            ]} />
          <SearchableSelect label="Data date" value={srcDate} onChange={setSrcDate}
            disabled={reconciling} loading={loadingFolders} options={folders}
            placeholder="No dates available" />
          <SearchableSelect label="File" value={srcFile} onChange={setSrcFile}
            disabled={reconciling} loading={loadingSrcFiles} options={srcFiles}
            placeholder="Pick a date first" />
        </div>

        {srcFile && (
          <div className="flex flex-wrap items-center gap-2 text-xs bg-slate-50 rounded-xl px-4 py-3 mb-5 border border-slate-100">
            <span className={`px-2 py-1 rounded font-semibold ${
              container === "fri-exports" ? "bg-blue-100 text-blue-700" : "bg-teal-100 text-teal-700"
            }`}>{container === "fri-exports" ? "FRI" : "MNRL"} · {srcDate}</span>
            <ArrowRight size={12} className="text-slate-300" />
            <span className="px-2 py-1 bg-amber-100 text-amber-700 rounded font-semibold">Match on HSBC data</span>
            <ArrowRight size={12} className="text-slate-300" />
            <span className="px-2 py-1 bg-green-100 text-green-700 rounded font-semibold">
              {container === "fri-exports" ? "fri_master" : "mnrl_master"}
            </span>
          </div>
        )}

        <PrimaryBtn onClick={handleReconcile} loading={startingRun} Icon={GitMerge}
          disabled={!srcFile || reconciling}
          gradient="linear-gradient(135deg, #4f46e5, #6d28d9)" label="Reconcile into master" />

        {runJob && <RunningBanner tone="indigo" title="Matching against HSBC data…"
          subtitle="Writing reconciled numbers to the master table" />}

        {lastRun && !runJob && (
          <div className="mt-4 rounded-xl border border-indigo-200 bg-indigo-50 p-4">
            <div className="flex flex-wrap items-center gap-x-5 gap-y-2">
              <StatChip label="Rows read" value={lastRun.input_row_count} />
              <StatChip label="Matched" value={lastRun.matched_count} tone="indigo" />
              <StatChip label="New" value={lastRun.master?.master_inserted} tone="green" />
              <StatChip label="Updated" value={lastRun.master?.master_updated} tone="blue" />
              <StatChip label="Cleared" value={lastRun.master?.cleared} tone="amber" />
              <StatChip label="Ignored (same-day conflict)" value={lastRun.master?.conflicts_ignored} tone="rose" />
            </div>
            {lastRun.master?.by_segment && (
              <div className="flex flex-wrap items-center gap-x-5 gap-y-2 mt-3 pt-3 border-t border-indigo-200">
                <StatChip label="IWPB" value={lastRun.master.by_segment.IWPB} tone="indigo" />
                <StatChip label="CIB" value={lastRun.master.by_segment["Non-IWPB"]} tone="slate" />
              </div>
            )}
          </div>
        )}
      </SectionCard>

      {/* ── Step 2: download from the master ── */}
      <SectionCard icon={Filter} iconBg="bg-emerald-600" title="Download reconciled data"
        subtitle="Pulled straight from the master table — pick a date and classification">
        <MasterFilters
          module={module} setModule={setModule}
          date={mDate} setDate={setMDate}
          segment={segment} setSegment={setSegment}
          dates={dates} segCounts={segCounts}
          loading={loadingMeta} disabled={downloading}
        />

        {/* Counts here are fixed totals: picking one does not change the others.
            Each label says what it is, so "Everything" always reads as the sum. */}
        <div className="mt-4">
          <SegmentedControl label="Include" value={recStatus} onChange={setRecStatus}
            disabled={downloading}
            options={[
              { value: "all", label: "Everything", count: preview?.total },
              { value: "active", label: module === "FRI" ? "Still flagged" : "Still disconnected",
                count: preview?.active },
              { value: "cleared", label: module === "FRI" ? "Now compliant" : "Now reactivated",
                count: preview?.cleared },
            ]} />
          <p className="text-xs text-slate-400 mt-1.5">
            {preview
              ? `${fmt(preview.total)} records in this window — ${fmt(preview.active)} still open, ${fmt(preview.cleared)} cleared. The three always add up.`
              : "Loading…"}
            {mDate !== "ALL" &&
              " A number belongs to a date if it was on the feed then, or cleared then."}
          </p>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 my-5">
          <CountBadge label="Will download" count={preview?.count} color="slate"
            sub={recStatus === "all" ? "records" : recStatus === "active" ? "still open" : "cleared"} />
          <CountBadge label="Awaiting ATR" count={preview?.atr_pending} color="amber"
            sub={preview
              ? `${fmt(preview.atr_pending_open)} open · ${fmt(preview.atr_pending_cleared)} cleared`
              : "open and cleared"} />
          <div className="rounded-xl border border-slate-200 bg-slate-50 p-4 flex flex-col justify-center gap-1.5">
            <p className="text-xs font-semibold text-slate-500 uppercase tracking-widest">Selection</p>
            <p className="text-sm text-slate-700 font-medium">
              {module} · {segment === "ALL" ? "IWPB and CIB" : SEGMENT_LABEL[segment]}
            </p>
            <p className="text-xs text-slate-400">
              {mDate === "ALL" ? "All dates in the 90-day window" : mDate}
            </p>
          </div>
        </div>

        <PrimaryBtn onClick={handleDownload} loading={startingDl} Icon={Download}
          disabled={downloading || !preview?.count}
          gradient="linear-gradient(135deg, #059669, #047857)" label="Build Excel and download" />

        {dlJob && <RunningBanner title="Building the workbook…"
          subtitle="Streamed from the master; large sets split across files" />}

// ── Date range with presets ──────────────────────────────────────────────────
// A fortnight is one click. Sending day by day meant 15 selections and 15 sends.
const rangeFor = (days) => {
  const to = new Date();
  const from = new Date();
  from.setDate(to.getDate() - (days - 1));
  const iso = (d) => d.toISOString().split("T")[0];
  return { from: iso(from), to: iso(to) };
};

const DateRangePicker = ({ from, to, onChange, disabled, days }) => {
  const presets = [
    { label: "Last 14 days", days: 14 },
    { label: "Last 30 days", days: 30 },
    { label: "Last 90 days", days: 90 },
  ];
  const activePreset = presets.find((p) => {
    const r = rangeFor(p.days);
    return r.from === from && r.to === to;
  });

  return (
    <div className="flex flex-col gap-3">
      <div className="flex flex-wrap items-center gap-2">
        {presets.map((p) => (
          <button key={p.days} type="button" disabled={disabled}
            onClick={() => { const r = rangeFor(p.days); onChange(r.from, r.to); }}
            className={`px-3 py-1.5 rounded-lg text-xs font-semibold transition-all
              disabled:opacity-40 disabled:cursor-not-allowed ${
              activePreset?.days === p.days
                ? "bg-slate-800 text-white"
                : "bg-slate-100 text-slate-600 hover:bg-slate-200"
            }`}>
            {p.label}
          </button>
        ))}
        <button type="button" disabled={disabled}
          onClick={() => onChange("", "")}
          className={`px-3 py-1.5 rounded-lg text-xs font-semibold transition-all
            disabled:opacity-40 ${
            !from && !to ? "bg-slate-800 text-white" : "bg-slate-100 text-slate-600 hover:bg-slate-200"
          }`}>
          Everything
        </button>
      </div>

      <div className="flex flex-wrap items-end gap-3">
        <div className="flex flex-col gap-1.5">
          <label className="text-xs font-semibold text-slate-500 uppercase tracking-widest">From</label>
          <input type="date" value={from} max={to || today()} disabled={disabled}
            onChange={(e) => onChange(e.target.value, to)}
            className="px-3 py-2.5 rounded-xl border border-slate-200 bg-white text-slate-800
              font-medium text-sm focus:outline-none focus:ring-2 focus:ring-amber-500
              disabled:opacity-50" />
        </div>
        <div className="flex flex-col gap-1.5">
          <label className="text-xs font-semibold text-slate-500 uppercase tracking-widest">To</label>
          <input type="date" value={to} min={from} max={today()} disabled={disabled}
            onChange={(e) => onChange(from, e.target.value)}
            className="px-3 py-2.5 rounded-xl border border-slate-200 bg-white text-slate-800
              font-medium text-sm focus:outline-none focus:ring-2 focus:ring-amber-500
              disabled:opacity-50" />
        </div>
        {days ? (
          <span className="px-2.5 py-1 mb-2 bg-amber-100 text-amber-700 rounded-full text-xs font-semibold">
            {days} day{days === 1 ? "" : "s"}
          </span>
        ) : (
          <span className="px-2.5 py-1 mb-2 bg-slate-100 text-slate-500 rounded-full text-xs font-semibold">
            All dates
          </span>
        )}
      </div>
    </div>
  );
};

// ── Schedule strip ───────────────────────────────────────────────────────────
// ── Schedule strip — read only ───────────────────────────────────────────────
// The cycle is a compliance-governed setting, not an operational one, so this
// reports what is configured and offers no way to change it. The interval, hour
// and enabled flag are locked server-side by ATR_SCHEDULE_LOCKED; a UI that
// appeared to edit them would only produce a 403.
const ScheduleStrip = ({ schedule }) => {
  if (!schedule) return null;
  const next = schedule.next_run_at ? new Date(schedule.next_run_at) : null;
  const daysAway = next ? Math.max(0, Math.ceil((next - Date.now()) / 86400000)) : null;

  const Fact = ({ label, value }) => (
    <div className="flex flex-col">
      <span className="text-xs text-slate-400 uppercase tracking-widest">{label}</span>
      <span className="text-sm font-semibold text-slate-700 tabular-nums">{value}</span>
    </div>
  );

  return (
    <div className="rounded-xl border border-slate-200 bg-slate-50 px-4 py-3">
      <div className="flex flex-wrap items-center gap-x-8 gap-y-3">
        <div className="flex items-center gap-2.5 flex-1 min-w-[220px]">
          <CalendarClock size={16}
            className={schedule.enabled ? "text-emerald-600" : "text-slate-400"} />
          <div>
            <p className="text-sm font-semibold text-slate-700">
              {schedule.enabled
                ? `Runs automatically every ${schedule.interval_days} days`
                : "Automatic sending is paused"}
            </p>
            <p className="text-xs text-slate-400 mt-0.5">
              {schedule.enabled && next
                ? `Next run ${fmtDateTime(schedule.next_run_at)}`
                  + (daysAway !== null ? ` · in ${daysAway} day${daysAway === 1 ? "" : "s"}` : "")
                : "Reports only go when you press Send"}
            </p>
          </div>
        </div>

        <Fact label="Every" value={`${schedule.interval_days} days`} />
        <Fact label="At" value={`${String(schedule.run_at_hour).padStart(2, "0")}:00 IST`} />
        <Fact label="Last run"
          value={schedule.last_run_at ? fmtDateTime(schedule.last_run_at) : "—"} />
        <span className={`px-2.5 py-1 rounded-full text-xs font-semibold ${
          schedule.enabled ? "bg-emerald-100 text-emerald-700" : "bg-slate-200 text-slate-600"
        }`}>
          {schedule.enabled ? "Active" : "Paused"}
        </span>
      </div>

      <p className="text-xs text-slate-400 mt-3 pt-3 border-t border-slate-200">
        This schedule is fixed and cannot be changed from the application.
      </p>
    </div>
  );
};

// ── Review rows: one per case type × classification ──────────────────────────
const ReviewGroup = ({ title, subtitle, rows, module, tone }) => {
  const total = rows.reduce((n, r) => n + (r.eligible || 0), 0);
  const tones = {
    amber: { bg: "bg-amber-50", border: "border-amber-200", text: "text-amber-800", pill: "bg-amber-100 text-amber-700" },
    green: { bg: "bg-emerald-50", border: "border-emerald-200", text: "text-emerald-800", pill: "bg-emerald-100 text-emerald-700" },
  }[tone];

  return (
    <div className={`rounded-xl border ${tones.border} ${tones.bg} p-4`}>
      <div className="flex items-baseline justify-between mb-1">
        <p className={`text-sm font-bold ${tones.text}`}>{title}</p>
        <span className={`px-2.5 py-1 rounded-full text-sm font-bold tabular-nums ${tones.pill}`}>
          {fmt(total)}
        </span>
      </div>
      <p className="text-xs text-slate-500 mb-3">{subtitle}</p>

      <div className="space-y-1.5">
        {rows.map((r) => {
          const d = r.defaults;
          const label = d
            ? codeLabels(module, d.action_type)[String(d.action_taken)] || "custom code"
            : null;
          return (
            <div key={`${r.case_type}-${r.segment}`}
              className="flex items-center justify-between gap-3 bg-white rounded-lg px-3 py-2 border border-white/60">
              <span className="text-xs font-semibold text-slate-700 w-20 shrink-0">
                {SEGMENT_LABEL[r.segment] || r.segment}
              </span>
              <span className="text-xs text-slate-500 flex-1 truncate">
                {d ? <>
                  <span className="font-mono text-slate-600">
                    {d.action_type ? `${d.action_type}·` : ""}{d.action_taken}
                  </span>
                  {" — "}{label}
                </> : <span className="text-rose-500">No action code set</span>}
              </span>
              <span className="text-xs font-bold tabular-nums text-slate-700 w-14 text-right">
                {fmt(r.eligible)}
              </span>
            </div>
          );
        })}
      </div>
    </div>
  );
};

// ── Settings slide-over ──────────────────────────────────────────────────────
// The action-code editor lived here. Removed deliberately: these codes are a
// compliance decision, locked server-side by ATR_DEFAULTS_LOCKED, so shipping a
// UI that appears to edit them would only produce a 403. The codes in force are
// still shown on each review row.

const ATRModule = () => {
  const [module, setModule]   = useState("FRI");
  const [segment, setSegment] = useState("ALL");
  const initial = rangeFor(14);
  const [dateFrom, setDateFrom] = useState(initial.from);
  const [dateTo, setDateTo]     = useState(initial.to);

  const [defaults, setDefaults] = useState([]);

  const [preview, setPreview]   = useState(null);
  const [loadingPreview, setLP] = useState(false);
  const [starting, setStarting] = useState(false);
  const [jobId, setJobId]       = useState(null);
  const [lastRun, setLastRun]   = useState(null);

  const [schedule, setSchedule]     = useState(null);

  const [runs, setRuns]         = useState([]);
  const [loadingRuns, setLR]    = useState(true);
  const [events, setEvents]     = useState(null);
  const [reportJob, setRepJob]  = useState(null);
  const [buildingRep, setBuildingRep] = useState(false);
  const [alert, setAlert]       = useState(null);

  const showAlert = (type, message) => {
    setAlert({ type, message });
    if (type !== "error") setTimeout(() => setAlert(null), 9000);
  };

  const loadMeta = useCallback(async (mod) => {
    try {
      const [df, sc] = await Promise.all([
        api.fetchAtrDefaults(mod), api.fetchAtrSchedule(mod),
      ]);
      setDefaults(df.defaults || []);
      setSchedule((sc.schedule || [])[0] || null);
    } catch (e) { showAlert("error", `Couldn't load settings: ${e.message}`); }
  }, []);

  const loadRuns = useCallback(async () => {
    setLR(true);
    try { setRuns((await api.fetchAtrRuns()).runs || []); }
    catch (e) { showAlert("error", `Couldn't load the log: ${e.message}`); }
    finally { setLR(false); }
  }, []);

  const refreshPreview = useCallback(async () => {
    setLP(true);
    try {
      setPreview(await api.fetchAtrPreview({
        type: module, date_from: dateFrom, date_to: dateTo, segment,
      }));
    } catch { setPreview(null); }
    finally { setLP(false); }
  }, [module, dateFrom, dateTo, segment]);

  useEffect(() => { loadMeta(module); }, [module, loadMeta]);
  useEffect(() => { loadRuns(); }, [loadRuns]);
  useEffect(() => { refreshPreview(); }, [refreshPreview]);

  const handleSend = async () => {
    setStarting(true); setAlert(null); setLastRun(null);
    try {
      const job = await api.submitAtr({
        type: module, date_from: dateFrom || null, date_to: dateTo || null, segment,
      });
      setJobId(job.job_id);
    } catch (e) { showAlert("error", e.message); }
    finally { setStarting(false); }
  };

  const handleBuildReport = async (runId) => {
    setBuildingRep(true); setAlert(null);
    try {
      const job = await api.buildAtrReport(runId ? { run_id: runId } : { type: module });
      setRepJob(job.job_id);
    } catch (e) { showAlert("error", e.message); }
    finally { setBuildingRep(false); }
  };

  const handleRetry = async () => {
    try {
      const r = await api.retryAtr({
        type: module, date_from: dateFrom || null, date_to: dateTo || null, segment,
      });
      await refreshPreview();
      showAlert("success", `${fmt(r.requeued)} number(s) put back in the queue.`);
    } catch (e) { showAlert("error", e.message); }
  };

  useJobPoller({
    jobId, recordCount: preview?.total_eligible || 0,
    onDone: async (result) => {
      setJobId(null); setLastRun(result);
      await Promise.all([loadRuns(), refreshPreview()]);
      const sent = result.total_sent ?? result.sent ?? 0;
      const rej  = result.total_rejected ?? result.rejected ?? 0;
      if (result.status === "nothing_to_send") showAlert("info", result.message);
      else showAlert(rej > 0 ? "warning" : "success",
        rej > 0 ? `Sent ${fmt(sent)}, ${fmt(rej)} rejected by C-DOT.`
                : `Sent ${fmt(sent)} action taken reports.`);
    },
    onFailed: (e) => { setJobId(null); showAlert("error", e); },
  });

  useJobPoller({
    jobId: reportJob, recordCount: 0,
    onDone: (result) => {
      setRepJob(null);
      if (result.status === "empty") { showAlert("info", result.message); return; }
      triggerBlobDownload(result.sas_url, result.file_name);
      showAlert("success", `Report ready — ${fmt(result.record_count)} entries.`);
    },
    onFailed: (e) => { setRepJob(null); showAlert("error", e); },
  });

  const running = starting || !!jobId;
  const openRows    = (preview?.breakdown || []).filter((b) => b.case_type === "open");
  const clearedRows = (preview?.breakdown || []).filter((b) => b.case_type === "cleared");
  const total = preview?.total_eligible ?? 0;

  return (
    <div className="space-y-6">
      {/* ── Step 1 + 2: what, and what will go ── */}
      <SectionCard icon={Send} iconBg="bg-amber-600" title="Action taken reports"
        subtitle="Tell C-DOT what your team did — and what has since been cleared"
        action={
          /* Action codes are a compliance decision and are locked server-side,
             so there is no edit control here. The codes in force are still
             visible on each review row below. */
          null
        }>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-5">
          <SegmentedControl label="Source" value={module} onChange={setModule} disabled={running}
            options={[{ value: "FRI", label: "FRI" }, { value: "MNRL", label: "MNRL" }]} />
          <SegmentedControl label="Classification" value={segment} onChange={setSegment}
            disabled={running}
            options={[
              { value: "ALL", label: "Both" },
              { value: "IWPB", label: "IWPB" },
              { value: "Non-IWPB", label: "CIB" },
            ]} />
        </div>

        <div className="mb-5 p-4 rounded-xl border border-slate-200 bg-slate-50">
          <p className="text-xs font-semibold text-slate-500 uppercase tracking-widest mb-3">
            Data dates to cover
          </p>
          <DateRangePicker from={dateFrom} to={dateTo} days={preview?.days_covered}
            disabled={running}
            onChange={(f, t) => { setDateFrom(f); setDateTo(t); }} />
        </div>

        {/* Review */}
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-5">
          <ReviewGroup tone="amber" rows={openRows} module={module}
            title={module === "FRI" ? "Still flagged" : "Still disconnected"}
            subtitle="Action your team applied" />
          <ReviewGroup tone="green" rows={clearedRows} module={module}
            title={module === "FRI" ? "Now compliant" : "Now reactivated"}
            subtitle="Cleared since the last report" />
        </div>

        <div className="flex items-center justify-between px-4 py-3 rounded-xl bg-slate-800 text-white mb-4">
          <div>
            <p className="text-sm font-semibold">Ready to send</p>
            <p className="text-xs text-slate-300 mt-0.5">
              {loadingPreview ? "Checking…"
                : total === 0
                  ? "Everything here has already been reported in its current state"
                  : `${preview?.days_covered ? `${preview.days_covered} days` : "All dates"} · already-sent numbers are skipped automatically`}
            </p>
          </div>
          <span className="text-3xl font-bold tabular-nums">{fmt(total)}</span>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
          <button onClick={handleRetry} disabled={running}
            className="flex items-center justify-center gap-2 px-4 py-3 rounded-xl text-sm font-semibold
              border border-slate-300 text-slate-700 bg-white hover:bg-slate-50 transition-all
              disabled:opacity-40 disabled:cursor-not-allowed">
            <RotateCcw size={15} /> Requeue rejected
          </button>
          <div className="md:col-span-2">
            <PrimaryBtn onClick={handleSend} loading={starting} Icon={Send}
              disabled={running || !total}
              gradient="linear-gradient(135deg, #d97706, #b45309)"
              label={total ? `Send ${fmt(total)} reports` : "Nothing to send"} />
          </div>
        </div>

        {jobId && <RunningBanner tone="amber" title="Sending to C-DOT…"
          subtitle="Encrypted batches; every attempt is logged per number" />}

        {lastRun && !jobId && (
          <div className="mt-4 rounded-xl border border-slate-200 bg-slate-50 p-4 flex flex-wrap gap-x-5 gap-y-2">
            <StatChip label="Sent" value={lastRun.total_sent ?? lastRun.sent} tone="green" />
            <StatChip label="Rejected" value={lastRun.total_rejected ?? lastRun.rejected} tone="rose" />
            {(lastRun.resent ?? 0) > 0 &&
              <StatChip label="Re-sent after a status change" value={lastRun.resent} tone="indigo" />}
          </div>
        )}

        {alert && <Alert {...alert} onClose={() => setAlert(null)} />}
      </SectionCard>

      {/* ── Schedule ── */}
      <ScheduleStrip schedule={schedule} />

      {/* ── Log ── */}
      <SectionCard icon={ListChecks} iconBg="bg-slate-600" title="Submission log"
        subtitle={`${runs.length} run${runs.length === 1 ? "" : "s"}`}
        action={
          <div className="flex items-center gap-2">
            <button onClick={() => handleBuildReport(null)} disabled={buildingRep || !!reportJob}
              className="flex items-center gap-1.5 text-xs font-semibold text-emerald-600
                bg-emerald-50 hover:bg-emerald-600 hover:text-white px-3 py-1.5 rounded-lg
                transition-colors disabled:opacity-40">
              {reportJob
                ? <><Loader2 size={13} className="animate-spin" /> Building…</>
                : <><Download size={13} /> Full {module} report</>}
            </button>
            <button onClick={loadRuns} disabled={loadingRuns}
              className="flex items-center gap-1.5 text-xs text-slate-600 hover:text-slate-800
                font-medium px-3 py-1.5 hover:bg-slate-100 rounded-lg transition-colors">
              <RefreshCw size={13} className={loadingRuns ? "animate-spin" : ""} /> Refresh
            </button>
          </div>
        }>
        <Paginated items={runs} loading={loadingRuns} accent="amber"
          emptyTitle="Nothing sent yet"
          emptyHint="Reconcile some data, then send the reports"
          columns={["Module", "Case", "Dates covered", "Action", "Sent", "Rejected", "Status", "Started", ""]}
          renderRow={(run) => (
            <tr key={run.id} className="hover:bg-amber-50/30 transition-colors">
              <td className="px-4 py-3.5">
                <span className={`px-2.5 py-1 rounded-full text-xs font-semibold ${
                  run.module === "FRI" ? "bg-blue-100 text-blue-700" : "bg-teal-100 text-teal-700"
                }`}>{run.module}</span>
                <span className="ml-1.5 text-xs text-slate-400">
                  {SEGMENT_LABEL[run.segment] || run.segment || "Both"}
                </span>
              </td>
              <td className="px-4 py-3.5">
                <span className={`px-2 py-0.5 rounded-full text-xs font-semibold whitespace-nowrap ${
                  run.case_type === "cleared"
                    ? "bg-emerald-100 text-emerald-700" : "bg-amber-100 text-amber-700"
                }`}>{run.case_label || run.case_type || "—"}</span>
              </td>
              <td className="px-4 py-3.5 text-xs text-slate-600 whitespace-nowrap">
                {run.date_from
                  ? <>
                      <span>{fmtDataDate(run.date_from)}</span>
                      {run.date_to && run.date_to !== run.date_from &&
                        <> → <span>{fmtDataDate(run.date_to)}</span></>}
                      {run.days_covered > 1 &&
                        <span className="ml-1.5 px-1.5 py-0.5 bg-slate-100 text-slate-500 rounded text-xs">
                          {run.days_covered} days
                        </span>}
                    </>
                  : <span className="text-slate-400">All dates</span>}
              </td>
              <td className="px-4 py-3.5 text-xs text-slate-600 whitespace-nowrap font-mono">
                {run.action_type ? `${run.action_type}·` : ""}{run.action_taken}
              </td>
              <td className="px-4 py-3.5">
                <span className="px-2 py-0.5 bg-green-100 text-green-700 rounded-full text-xs font-semibold tabular-nums">
                  {fmt(run.total_sent)}
                </span>
              </td>
              <td className="px-4 py-3.5">
                {run.total_rejected > 0
                  ? <span className="px-2 py-0.5 bg-rose-100 text-rose-700 rounded-full text-xs font-semibold tabular-nums">
                      {fmt(run.total_rejected)}
                    </span>
                  : <span className="text-slate-300 text-xs">—</span>}
              </td>
              <td className="px-4 py-3.5">
                <span className={`px-2 py-0.5 rounded-full text-xs font-semibold ${
                  run.status === "done" ? "bg-green-100 text-green-700"
                    : run.status === "partial" ? "bg-amber-100 text-amber-700"
                    : run.status === "failed" ? "bg-rose-100 text-rose-700"
                    : "bg-slate-100 text-slate-600"
                }`}>{run.status}</span>
              </td>
              <td className="px-4 py-3.5 text-slate-400 text-xs whitespace-nowrap">
                {fmtDateTime(run.started_at)}
              </td>
              <td className="px-4 py-3.5">
                <div className="flex items-center gap-1.5">
                  <button onClick={async () => {
                    try { setEvents({ run, rows: (await api.fetchAtrEvents(run.id)).events }); }
                    catch (e) { showAlert("error", e.message); }
                  }}
                    className="px-3 py-1.5 text-xs font-semibold text-slate-600 bg-slate-100
                      hover:bg-slate-700 hover:text-white rounded-lg transition-all">
                    View
                  </button>
                  <button onClick={() => handleBuildReport(run.id)} disabled={buildingRep || !!reportJob}
                    className="flex items-center gap-1 px-3 py-1.5 text-xs font-semibold text-emerald-600
                      bg-emerald-50 hover:bg-emerald-600 hover:text-white rounded-lg transition-all
                      disabled:opacity-40">
                    <Download size={12} /> Excel
                  </button>
                </div>
              </td>
            </tr>
          )} />
      </SectionCard>

      {/* Audit drawer */}
      {events && (
        <div className="fixed inset-0 bg-slate-900/40 flex items-end md:items-center justify-center z-50 p-4"
          onClick={() => setEvents(null)}>
          <div className="bg-white rounded-2xl shadow-2xl max-w-4xl w-full max-h-[85vh] overflow-hidden flex flex-col"
            onClick={(e) => e.stopPropagation()}>
            <div className="flex items-center justify-between px-6 py-4 border-b border-slate-200">
              <div>
                <h3 className="font-bold text-slate-800">Report audit</h3>
                <p className="text-xs text-slate-400 mt-0.5">
                  {events.run.module} · {events.run.case_label} ·{" "}
                  {fmtDataDate(events.run.date_from)}
                  {events.run.date_to !== events.run.date_from && ` → ${fmtDataDate(events.run.date_to)}`}
                  {" "}· {fmt(events.rows.length)} entries
                </p>
              </div>
              <button onClick={() => setEvents(null)} className="text-slate-400 hover:text-slate-600">
                <X size={18} />
              </button>
            </div>
            <div className="overflow-y-auto">
              <table className="w-full text-sm">
                <thead className="sticky top-0 bg-slate-50">
                  <tr className="border-b border-slate-200">
                    {["Mobile", "Data date", "Outcome", "Why sent", "Action", "Response"].map((h) => (
                      <th key={h} className="text-left px-4 py-2.5 text-xs font-semibold text-slate-500 uppercase">
                        {h}
                      </th>
                    ))}
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100">
                  {events.rows.map((e) => (
                    <tr key={e.id} className="hover:bg-slate-50/60">
                      <td className="px-4 py-2.5 font-mono text-xs text-slate-700">{e.mobile_no}</td>
                      <td className="px-4 py-2.5 text-xs text-slate-500">{fmtDataDate(e.feed_date)}</td>
                      <td className="px-4 py-2.5">
                        <span className={`px-2 py-0.5 rounded-full text-xs font-semibold ${
                          e.outcome === "sent" ? "bg-green-100 text-green-700"
                            : e.outcome === "rejected" ? "bg-rose-100 text-rose-700"
                            : "bg-amber-100 text-amber-700"
                        }`}>{e.outcome}</span>
                      </td>
                      <td className="px-4 py-2.5 text-xs text-slate-600">
                        {TRIGGER_LABELS[e.trigger] || e.trigger}
                        {e.attempt_no > 1 && <span className="ml-1 text-slate-400">· #{e.attempt_no}</span>}
                      </td>
                      <td className="px-4 py-2.5 text-xs text-slate-600 whitespace-nowrap font-mono">
                        {e.action_type ? `${e.action_type}·` : ""}{e.action_taken}
                      </td>
                      <td className="px-4 py-2.5 text-xs max-w-[220px]">
                        <span className={e.outcome === "sent" ? "text-slate-500" : "text-rose-600"}>
                          {e.response_message || "—"}
                        </span>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
            <div className="px-6 py-3 border-t border-slate-200 bg-slate-50">
              <p className="text-xs text-slate-400">
                Every attempt is kept, with the exact payload sent to C-DOT.
                Use <strong>Excel</strong> for the full report.
              </p>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
