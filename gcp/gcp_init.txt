from fastapi import APIRouter, HTTPException
from logging import logger

from indp import get_total_count, fetch_batch_with_retry, indp_bq_to_az, print_first_rows
from jobs import start_job, get_job

router = APIRouter(prefix="/indp")


@router.get("/count")
def count():
    logger.info("/count api is called")
    try:
        res = get_total_count()
        return {"Response": res}
    except Exception as e:
        logger.error("Failed fetching count")
        logger.error(e)
        raise HTTPException(status_code=500, detail=f"Failed fetching count: {e}")


@router.get("/data")
def data():
    logger.info("/data api is called")
    try:
        res = fetch_batch_with_retry(0, 10000)
        return {"Response": res}
    except Exception as e:
        logger.error("Failed fetching data")
        logger.error(e)
        raise HTTPException(status_code=500, detail=f"Failed fetching data: {e}")


@router.get("/run")
def run():
    logger.info("/run api is called")
    try:
        job_id = start_job(indp_bq_to_az)
        return {
            "job_id": job_id,
            "status": "accepted",
        }
    except Exception as e:
        logger.error("Failed run")
        logger.error(e)
        raise HTTPException(status_code=500, detail=f"Failed run: {e}")


@router.get("/status/{job_id}")
def job_status(job_id: str):
    job = get_job(job_id)
    if not job:
        raise HTTPException(status_code=404, detail="Job not found")
    return job


@router.get("/preview")
def gcp_conn():
    logger.info("/preview api is called")
    try:
        res = print_first_rows()
        return {"Response": res}
    except Exception as e:
        logger.error("Failed preview")
        logger.error(e)
        raise HTTPException(status_code=500, detail=f"Failed preview: {e}")
