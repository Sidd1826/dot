from datetime import datetime, timezone
import os
import time

import azure.functions as func

from inm_gcp_service import app as fastapi_app
from inm_gcp_service.common.logging import get_logger
from inm_gcp_service.functions.indp import indp_bq_to_az
from inm_gcp_service.functions.indp._consts import get_cleanup_schedule, get_indp_schedule
from inm_gcp_service.functions.indp._jobs import (
    clean_old_jobs,
    mark_stale_jobs_failed,
    start_job,
    wait_for_job,
)

app = func.AsgiFunctionApp(app=fastapi_app, http_auth_level=func.AuthLevel.ANONYMOUS)

logger = get_logger("GCP_function_app")

INDP_SCHEDULE_TIME = get_indp_schedule()

# Configuration constants with fixed default values
MAX_ATTEMPTS = 3
ATTEMPT_TIMEOUT_SECONDS = 40 * 60      # 2400 seconds (40 minutes)
TOTAL_BUDGET_SECONDS = 150 * 60        # 9000 seconds (150 minutes)
RETRY_BACKOFF_SECONDS = [120, 600]     # 2 minutes, then 10 minutes
RAISE_ON_EMPTY_SOURCE = False


def _backoff_for(attempt: int) -> int:
    idx = attempt - 1
    if idx < len(RETRY_BACKOFF_SECONDS):
        return RETRY_BACKOFF_SECONDS[idx]
    return RETRY_BACKOFF_SECONDS[-1] if RETRY_BACKOFF_SECONDS else 0


@app.function_name(name="indp_scheduler")
@app.timer_trigger(schedule=get_indp_schedule(), arg_name="timer", run_on_startup=False)
def indp_function(timer: func.TimerRequest) -> None:
    started = time.monotonic()
    logger.info("Timer fired for indp at %s", datetime.now(timezone.utc).isoformat())

    # Clear rows abandoned by a previous invocation that was killed mid-run.
    try:
        mark_stale_jobs_failed()
    except Exception as exc:
        logger.warning("Stale job sweep failed (continuing): %s", exc)

    last_error: str | None = None
    last_result: dict | None = None

    for attempt in range(1, MAX_ATTEMPTS + 1):
        elapsed = time.monotonic() - started
        remaining = TOTAL_BUDGET_SECONDS - elapsed
        if remaining < 60:
            logger.error(
                "Out of time budget (%.0fs used) before attempt %d - stopping.",
                elapsed, attempt,
            )
            break

        attempt_budget = int(min(ATTEMPT_TIMEOUT_SECONDS, remaining))
        logger.info(
            "INDP attempt %d/%d starting (budget=%ds, %.0fs of %ds total used)",
            attempt, MAX_ATTEMPTS, attempt_budget, elapsed, TOTAL_BUDGET_SECONDS,
        )

        try:
            job_id = start_job(indp_bq_to_az)
        except Exception as exc:
            last_error = f"could not start job: {exc}"
            logger.exception("Attempt %d: %s", attempt, last_error)
            job_id = None

        if job_id:
            logger.info("Attempt %d accepted. job_id=%s", attempt, job_id)
            try:
                job = wait_for_job(job_id, attempt_budget)
            except Exception as exc:
                last_error = f"could not track job {job_id}: {exc}"
                logger.exception("Attempt %d: %s", attempt, last_error)
                job = None

            if job is not None:
                if job["status"] == "done":
                    result = job.get("result") or {}
                    status = result.get("status") if isinstance(result, dict) else None
                    last_result = result if isinstance(result, dict) else None

                    if status == "skipped":
                        # Source empty - retryable, the feed may still be landing.
                        last_error = "BigQuery returned 0 rows (source not landed?)"
                        logger.warning("Attempt %d: %s", attempt, last_error)
                    else:
                        logger.info(
                            "INDP succeeded on attempt %d/%d. job_id=%s inserted=%s",
                            attempt, MAX_ATTEMPTS, job_id,
                            result.get("total_inserted") if isinstance(result, dict) else "?",
                        )
                        return
                else:
                    last_error = job.get("error") or "pipeline failed with no error recorded"
                    logger.error(
                        "Attempt %d/%d FAILED. job_id=%s error=%s",
                        attempt, MAX_ATTEMPTS, job_id, last_error,
                    )

        if attempt < MAX_ATTEMPTS:
            pause = _backoff_for(attempt)
            pause = int(min(pause, max(0, TOTAL_BUDGET_SECONDS - (time.monotonic() - started) - 60)))
            if pause > 0:
                logger.info("Backing off %ds before attempt %d", pause, attempt + 1)
                time.sleep(pause)

    # Every attempt is spent.
    if last_result and last_result.get("status") == "skipped":
        msg = f"INDP found no source rows after {MAX_ATTEMPTS} attempts - check the BigQuery feed."
        if RAISE_ON_EMPTY_SOURCE:
            raise RuntimeError(msg)
        logger.error(msg)
        return

    raise RuntimeError(
        f"INDP pipeline failed after {MAX_ATTEMPTS} attempts. Last error: {last_error}"
    )


@app.function_name(name="indp_clean_old_jobs")
@app.timer_trigger(schedule=get_cleanup_schedule(), arg_name="timer", run_on_startup=False)
def clean_old_jobs_timer(timer: func.TimerRequest) -> None:
    now = datetime.now(timezone.utc).isoformat()
    logger.info("Timer fired for clean_old_jobs at %s", now)

    try:
        mark_stale_jobs_failed()
    except Exception as exc:
        logger.warning("Stale job sweep failed (continuing): %s", exc)

    try:
        removed = clean_old_jobs(hours=24 * 7)  # last week
        logger.info("Cleaned %s old job(s) older than 7 days", removed)
    except Exception as e:
        logger.exception("Failed to clean old jobs: %s", e)
        raise
